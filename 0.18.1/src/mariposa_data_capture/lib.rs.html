3//! OQueue data capture implementation for Mariposa
4//!
5//! This allows efficiently capturing data from OQueues for external processing. The goal is to
6//! capture the data with as little overhead as possible on the running system.
7//!
8//! ## The binary data format used in Mariposa OQueue data capture.
9//!
10//! The format is a large raw file with blocks. Each block must be 4k page aligned and is in the
11//! following format:
12//! * The magic "number": `MARIPOSALDOSDATA` (null terminated).
13//! * A [CBOR](https://cbor.io/) object:
14//!     ```
15//!     { "name": "file.path", "oqueues": ["oqueue.path", ...], "type_name": "TypeName" }
16//!     ```
17//!   The `oqueues` and `length` fields are optional.
18//! * A series of CBOR objects serialized from type `TypeName` using Serde. The CBOR objects are not
19//!   part of a list, instead just being concatenated. In Python, they can be read using repeated
20//!   calls to
21//!   [`cbor2.CBORDecoder.decode`](https://cbor2.readthedocs.io/en/latest/api.html#cbor2.CBORDecoder.decode).
22//!
23//! The set of OQueue paths is only for convenience, so it can be incomplete or missing. This may be
24//! because of set of OQueues was not known when output started.
25//!
26//! Note: The output format is not particularly space efficient because it includes the field names
27//! of structs every time they are serialized. If this becomes a problem there are a few options: 1)
28//! use [`#[serde(rename=...)`](https://serde.rs/field-attrs.html) to give shorter field names in
29//! the output, 2) explore using `serde_cbor`s [packed
30//! format](https://docs.rs/serde_cbor/latest/serde_cbor/struct.Serializer.html#method.packed_format).
31#![no_std]
32#![deny(unsafe_code)]

74#[cfg(all(ktest, not(baseline_asterinas)))]
75mod tests {
76    use alloc::sync::Arc;
77    use core::time::Duration;
78
79    use aster_block::test_utils::MemoryDisk;
80    use ostd::{
81        assertion::sleep,
82        orpc::oqueue::{GenericOQueueRef, OQueue, OQueueBase, OQueueRef, ObservationQuery},
83        path,
84        prelude::*,
85    };
86
87    use crate::{
88        DataCaptureDevice as _, DataCaptureDeviceServer, FileDescriptor, ObserverRegistration,
89    };
90
91    #[ktest]
92    fn capture_server() {
93        // Create memory disk with space for 4 blocks
94        let block_device = Arc::new(MemoryDisk::new(4096 * 4));
95        let device = DataCaptureDeviceServer::new(block_device.clone());
96
97        let path = path!(test_capture);
98        let builder = device
99            .new_file(FileDescriptor {
100                length: 4096 * 2,
101                path: path.clone(),
102            })
103            .unwrap();
104        let server = builder.build();
105
106        // Attach an OQueue to the capture
107        let oqueue: OQueueRef<u32> = GenericOQueueRef::new(4, path.clone());
108        let attachment = ObserverRegistration {
109            path,
110            observer: oqueue
111                .attach_strong_observer(ObservationQuery::new(|x| *x as u8))
112                .unwrap(),
113        };
114        server.register_observer(attachment).unwrap();
115
116        let producer = oqueue.attach_ref_producer().unwrap();
117
118        // Test capturing disabled initially
119        producer.produce_ref(&10);
120        sleep(Duration::from_millis(10));
121
122        // Enable capturing
123        server.start().unwrap();
124        producer.produce_ref(&42);
125        producer.produce_ref(&100);
126        producer.produce_ref(&200);
127        sleep(Duration::from_millis(10));
128
129        // Flush and give time for capture to complete
130        server.sync().unwrap();
131
132        sleep(Duration::from_millis(10));
133
134        let device_data = block_device.data.lock();
135
136        let path_bytes: &[u8] = b"test_capture";
137        assert!(
138            device_data
139                .windows(path_bytes.len())
140                .any(|window| window == path_bytes)
141        );
142
143        // The CBOR encoding of 42, 100, 200
144        let data_bytes = [0x18, 0x2a, 0x18, 0x64, 0x18, 0xc8];
145        assert!(
146            device_data
147                .windows(data_bytes.len())
148                .any(|window| window == data_bytes)
149        );
150    }
