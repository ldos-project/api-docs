3//! An ORPC server which performs data capture into the format described in [the crate
4//! documentation](`crate`) using an extraction function and a set of OQueues to observe. Data from
5//! different OQueues are not distinguished in the output.
6//!
7//! This uses the following terms:
8//! * "Device" refers to a block storage device.
9//! * "File" refers to a range within a device used to store data of a specific type from a set of
10//!   registered OQueues.
11//! * "Chunk" refers to a chunk of data written to the block device together. Chucks are the size of
12//!   a block device block, but the term was just too confusing here.
13//!
14//! For each block device that will be used for capture, the user should construct a
15//! [`DataCaptureDevice`]. This can then be used to construct [`DataCaptureFile`] servers (via a
16//! builder due to the generic type).
17//!
18//! [`DataCaptureFile`] has a thread which will observe all OQueues attach via
19//! [`DataCaptureFile::register_observer`].
20

44/// Registration information for an OQueue observer
45pub struct ObserverRegistration<T: Copy + Send + Serialize> {
46    /// The path of the OQueue this is observing
47    pub path: Path,
48    /// An observer attachment at the appropriate type.
49    pub observer: StrongObserver<T>,
50}
51
52impl<T: Copy + Send + Serialize> core::fmt::Debug for ObserverRegistration<T> {
53    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
54        f.debug_struct("ObserverRegistration")
55            .field("path", &self.path)
56            .finish()
57    }
58}

60#[orpc_trait]
61pub trait DataCaptureFile<T: Copy + Send + Serialize>: Any {
62    /// Attach a new OQueue to the output. If output has already started, then the path will not
63    /// appear in the block header.
64    fn register_observer(&self, attachment: ObserverRegistration<T>) -> Result<(), RPCError>;
65    /// Write values the caller already holds, without routing them through an OQueue.
66    fn write_values(&self, values: Box<dyn Iterator<Item = T> + Send>) -> Result<(), RPCError>;
67    /// Sync writes to disk.
68    fn sync(&self) -> Result<(), RPCError>;
69    /// Enable capturing to this file.
70    fn start(&self) -> Result<(), RPCError>;
71    /// Stop the server
72    fn stop(&self) -> Result<(), RPCError>;
73}
74
75/// Command enum for DataCaptureFile operations
76enum DataCaptureFileCommand<T: Copy + Send + Serialize + 'static> {
77    RegisterObserver(ObserverRegistration<T>),
78    WriteValues(Box<dyn Iterator<Item = T> + Send>),
79    Sync,
80    Stop,
81}
82
83impl<T: Copy + Send + Serialize + 'static> core::fmt::Debug for DataCaptureFileCommand<T> {
84    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
85        match self {
86            Self::RegisterObserver(arg0) => {
87                f.debug_tuple("DataCaptureFileCommand").field(arg0).finish()
88            }
89            Self::WriteValues(_) => write!(f, "WriteValues"),
90            Self::Sync => write!(f, "Sync"),
91            Self::Stop => write!(f, "Stop"),
92        }
93    }
94}

96/// Writes the block header if it has not been written yet, consuming the collected OQueue paths so
97/// later attachments no longer accumulate into it.
98fn write_header_once<T>(
99    writer: &mut ChunkingWriteWrapper,
100    name: &Path,
101    paths: &mut Option<Vec<Path>>,
102) -> Result<(), DataCaptureError> {
103    let Some(collected) = paths.take() else {
104        return Ok(());
105    };
106    writer.write_header::<T>(&name.to_string(), &collected)
107}

109#[orpc_server(DataCaptureFile<T>)]
110struct DataCaptureFileServer<T: Copy + Send + Serialize + 'static> {
111    command_oqueue: ConsumableOQueueRef<DataCaptureFileCommand<T>>,
112    command_producer: ValueProducer<DataCaptureFileCommand<T>>,
113    started: AtomicBool,
114    stopped: AtomicBool,
115    stopped_wait_queue: WaitQueue,
116}
117
118pub(crate) struct DataCaptureFileServerThread<T: Copy + Send + Serialize + 'static> {
119    path: Path,
120    command_consumer: Consumer<DataCaptureFileCommand<T>>,
121    block_device: Arc<dyn BlockDevice>,
122    start_bid: Bid,
123    end_bid: Bid,
124    server: Arc<DataCaptureFileServer<T>>,
125}
126
127impl<T: Copy + Send + Serialize + 'static> DataCaptureFileServerThread<T> {
128    fn run(&self) -> Result<(), DataCaptureError> {
129        let mut data_buf_handler = ChunkingWriteWrapper::new(
130            BLOCK_SIZE * 2,
131            self.block_device.clone(),
132            self.start_bid,
133            self.end_bid,
134        );
135        let mut observers: Vec<StrongObserver<T>> = Default::default();
136        // The paths of the attached OQueues. Once the header is written this is set to None and
137        // paths are no longer collected even if more OQueues are attached.
138        let mut paths = Some(Vec::default());
139        let mut block_handler = BlockOnMany::new();
140
141        loop {
142            let blockers = [(&self.command_consumer) as &dyn Blocker]
143                .into_iter()
144                .chain(observers.iter().map(|o| o as &dyn Blocker));
145            block_handler.block_on(blockers);
146
147            // Handle commands from method calls
148            while let Some(cmd) = self.command_consumer.try_consume() {
149                match cmd {
150                    DataCaptureFileCommand::RegisterObserver(ObserverRegistration {
151                        path,
152                        observer,
153                    }) => {
154                        observers.push(observer);
155                        if let Some(paths) = &mut paths {
156                            paths.push(path);
157                        }
158                    }
159                    DataCaptureFileCommand::WriteValues(values) => {
160                        write_header_once::<T>(&mut data_buf_handler, &self.path, &mut paths)?;
161                        for value in values {
162                            data_buf_handler.write_value(&value);
163                            data_buf_handler.flush_if_needed()?;
164                        }
165                        data_buf_handler.sync()?;
166                    }
167                    DataCaptureFileCommand::Sync => {
168                        data_buf_handler.sync()?;
169                    }
170                    DataCaptureFileCommand::Stop => {
171                        data_buf_handler.sync()?;
172                        self.server
173                            .stopped
174                            .store(true, core::sync::atomic::Ordering::SeqCst);
175                        self.server.stopped_wait_queue.wake_all();
176                        return Ok(());
177                    }
178                }
179            }
180
181            let started = self
182                .server
183                .started
184                .load(core::sync::atomic::Ordering::SeqCst);
185            // Observe and serialize
186            for o in &observers {
187                // We can't skip the try_strong_observe calls when not `capturing` because that
188                // would leave the values in the OQueues and block them.
189                while let Ok(Some(v)) = o.try_strong_observe() {
190                    if started {
191                        write_header_once::<T>(&mut data_buf_handler, &self.path, &mut paths)?;
192
193                        data_buf_handler.write_value(&v);
194                        data_buf_handler.flush_if_needed()?;
195                        if data_buf_handler.current_bid == self.end_bid {
196                            log::warn!("Data capture ran out of space.");
197                        }
198                    }
199                }
200            }
201        }
202    }
203}

205#[orpc_impl]
206impl<T: Copy + Send + Serialize> DataCaptureFile<T> for DataCaptureFileServer<T> {
207    fn register_observer(&self, attachment: ObserverRegistration<T>) -> Result<(), RPCError> {
208        self.command_producer
209            .produce(DataCaptureFileCommand::RegisterObserver(attachment));
210        Ok(())
211    }
212
213    fn write_values(&self, values: Box<dyn Iterator<Item = T> + Send>) -> Result<(), RPCError> {
214        self.command_producer
215            .produce(DataCaptureFileCommand::WriteValues(values));
216        Ok(())
217    }
218
219    fn sync(&self) -> Result<(), RPCError> {
220        self.command_producer.produce(DataCaptureFileCommand::Sync);
221        Ok(())
222    }
223
224    fn stop(&self) -> Result<(), RPCError> {
225        self.command_producer.produce(DataCaptureFileCommand::Stop);
226        self.stopped_wait_queue.wait_until(|| {
227            self.stopped
228                .load(core::sync::atomic::Ordering::SeqCst)
229                .then_some(())
230        });
231        Ok(())
232    }
233
234    fn start(&self) -> Result<(), RPCError> {
235        self.started
236            .store(true, core::sync::atomic::Ordering::SeqCst);
237        Ok(())
238    }
239}

241/// A builder for a [`DataCaptureFile`] provided by a [`DataCaptureDevice`]. This is required
242/// because ORPC methods cannot take type parameters, so this serves to hold all the information
243/// needed to construct the [`DataCaptureFile`] and then take the type parameter that is also
244/// needed.
245///
246/// [`DataCaptureDevice`]: crate::data_capture_device::DataCaptureDevice
247pub struct DataCaptureFileBuilder {
248    pub(crate) block_device: Arc<dyn BlockDevice>,
249    pub(crate) path: Path,
250    pub(crate) start: usize,
251    pub(crate) end: usize,
252    pub(crate) server: Arc<dyn Server>,
253}
254
255impl DataCaptureFileBuilder {
256    /// Construct the [`DataCaptureFile`] for a specific type of data.
257    pub fn build<T>(self) -> Arc<dyn DataCaptureFile<T>>
258    where
259        T: Copy + Send + Serialize + 'static,
260    {
261        // We manually context switch into the server here. This is not something we should
262        // generally do, but this build function is required and can't be on a server (due to the
263        // need to take a type parameter) and it's body should run in the context of the manager
264        // server.
265        self.server
266            .orpc_server_base()
267            .call_in_context(move || -> Result<Arc<DataCaptureFileServer<T>>, RPCError> {
268                let command_oqueue =
269                    ConsumableOQueueRef::new(8, self.path.append(&path!(commands)));
270                let server = new_server!(|_| DataCaptureFileServer {
271                    command_producer: command_oqueue
272                        .attach_value_producer()
273                        .expect("single purpose OQueue failed."),
274                    command_oqueue,
275                    started: AtomicBool::new(false),
276                    stopped: AtomicBool::new(false),
277                    stopped_wait_queue: WaitQueue::new(),
278                });
279
280                spawn_thread(server.clone(), {
281                    let thread = DataCaptureFileServerThread {
282                        path: self.path.clone(),
283                        command_consumer: server
284                            .command_oqueue
285                            .attach_consumer()
286                            .expect("single purpose OQueue failed."),
287                        block_device: self.block_device,
288                        start_bid: Bid::from_offset(self.start),
289                        end_bid: Bid::from_offset(self.end),
290                        server: server.clone(),
291                    };
292                    move || Ok(thread.run()?)
293                });
294
295                Ok(server)
296            })
297            .expect("Closure cannot fail")
298    }
