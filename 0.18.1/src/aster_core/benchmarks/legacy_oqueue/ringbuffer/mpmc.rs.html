34/// A single element of the ringbuffer that tracks the last "turn" that updated it.
35struct Slot<T> {
36    /// The turn for this slot. When it is even, it is writable. If it was previously written to,
37    /// that data has been read by all Consumers and StrongObservers.
38    /// If it is odd, it is readable. It cannot be written to because some Consumer or
39    /// StrongObserver has not yet read the data.
40    turn: CachePadded<AtomicUsize>,
41    /// The data being stored in this slot
42    value: Element<T>,
43}
44
45impl<T> Slot<T> {
46    fn new() -> Self {
47        Slot {
48            turn: CachePadded::new(AtomicUsize::new(0)),
49            value: Element::uninit(),
50        }
51    }
52
53    /// Store `data` in this Slot
54    unsafe fn store(&self, data: T) {
55        unsafe { (*self.value.data.get()).write(data) };
56    }
57
58    /// Get the value present in this Slot. This is only safe if [`Slot::store`] was previously
59    /// called.
60    unsafe fn get(&self) -> T {
61        // SAFETY: This assumes that the self.store has be previously called
62        unsafe { ptr::read(self.value.data.get()).assume_init() }
63    }
64
65    unsafe fn get_maybe_uninit(&self) -> MaybeUninit<T> {
66        // SAFETY: This assumes that the self.store has be previously called
67        unsafe { ptr::read(self.value.data.get()) }
68    }
69}

71/// The sentinel value acts as a lock. When [`MPMCOQueue::head`] is this value or larger, writes to
72/// the queue will be blocked. This is used to ensure that StrongObservers can be initialized with a
73/// valid index and avoid races where the StrongObserver index is a value that has already been
74/// consumed. We pick the value 1<<63 so that when producers attempt to do a fetch_add, they don't
75/// need to first do a load. It is unlikely that we will have 1<<63 concurrent producers (even
76/// try_produce will block on the sentinel) in the time it takes to attach a single observer. This
77/// does reduce the limit of elements in the queue from usize::MAX to (1 << 63 - 1).
78const MPMCOQUEUE_HEAD_SENTINEL: usize = 1 << 63;
79
80/// MPMCOQueue allows conccurrent producers and consumers. For any produced message it is guaranteed
81/// that it will be read by exactly one consumer, all strong observers, and zero or more weak
82/// observers. A consumed message cannot be viewed by a future strong observer, but could be viewed
83/// by a weak observer. Note that if either of STRONG_OBSERVERS or WEAK_OBSERVERS is set to true,
84/// then T must be Copy, otherwise the OQueue trait will not be implemented.
85pub(crate) struct MPMCOQueue<
86    T,
87    const STRONG_OBSERVERS: bool = true,
88    const WEAK_OBSERVERS: bool = true,
89> {
90    this: Weak<Self>,
91    capacity: NonZero<usize>,
92    max_strong_observers: usize,
93    /// A producer "owns" a slot if it successfully incremented head to one index past the slot and
94    /// the slot is marked as ready for writing (even turn). A consumer "owns" a slot if it
95    /// successfully incremented tail to one index past the slot and the slot is marked as ready for
96    /// reading (odd turn).All other slots are "unowned". Note that the turn of a slot will be
97    /// updated from reading to writing only when the slowest consumer or strong observer visits it.
98    slots: Box<[Slot<T>]>,
99    /// The index at which elements are produced to.
100    head: CachePadded<AtomicUsize>,
101    /// The index at which elements are consumed from. Note that unlike StrongObservers, each
102    /// message is read once across all consumers.
103    tail: CachePadded<AtomicUsize>,
104    n_consumers: Mutex<usize>,
105    /// Each tail represents the position of a StrongObserver. Tails may be read by multiple
106    /// handles, but will be written to by exactly one handle. Unlike a Consumer, each message is
107    /// read once by every StrongObserver.
108    strong_observer_tails: Box<[CachePadded<AtomicUsize>]>,
109    n_strong_observers: Mutex<usize>,
110
111    /// Wait queue for threads waiting to put into the queue.
112    put_wait_queue: WaitQueue,
113    /// Wait queue for threads waiting to read data from the queue.
114    read_wait_queue: WaitQueue,
115}

126// Page aligned allocations are needed for large allocations because the buddy allocator in
127// ostd is very simple, and misaligned allocations can lead to fragmentation because the allocator
128// doesn't coalesce often enough.
129fn allocate_page_aligned_array<T>(len: usize) -> Box<[Slot<T>]> {
130    let elem_size = size_of::<Slot<T>>();
131    let total_size = elem_size * len;
132
133    if total_size < 4096 {
134        return (0..(len + 1)).map(|_| Slot::new()).collect();
135    }
136
137    // Layout with page alignment
138    let layout = core::alloc::Layout::from_size_align(total_size, 4096).expect("invalid layout");
139
140    unsafe {
141        let ptr = alloc(layout);
142        if ptr.is_null() {
143            handle_alloc_error(layout);
144        }
145
146        // Initialize each element
147        let mut current = ptr as *mut Slot<T>;
148        for _ in 0..len {
149            ptr::write(current, Slot::new());
150            current = current.add(1);
151        }
152
153        // Turn into a Box<[Slot<T>]> safely
154        let slice = core::slice::from_raw_parts_mut(ptr as *mut Slot<T>, len);
155        Box::from_raw(slice)
156    }
157}

159// MPMC OQueue implementation based on rigtorp
160impl<T, const STRONG_OBSERVERS: bool, const WEAK_OBSERVERS: bool>
161    MPMCOQueue<T, STRONG_OBSERVERS, WEAK_OBSERVERS>
162{
163    /// Create a new MPMCOQueue with a maximum size and a maximum number of strong observers.
164    pub(crate) fn new(capacity: usize, max_strong_observers: usize) -> Arc<Self> {
165        assert!(max_strong_observers == 0 || STRONG_OBSERVERS);
166        assert!(capacity.is_power_of_two());
167        assert!(capacity > 0);
168        assert!(capacity < MPMCOQUEUE_HEAD_SENTINEL);
169
170        Arc::new_cyclic(|this| MPMCOQueue {
171            this: this.clone(),
172            // SAFETY: we check that size > 0 above
173            capacity: unsafe { NonZero::new_unchecked(capacity) },
174            max_strong_observers,
175            slots: allocate_page_aligned_array(capacity),
176            head: CachePadded::new(AtomicUsize::new(0)),
177            tail: CachePadded::new(AtomicUsize::new(usize::MAX)),
178            n_consumers: Mutex::new(0),
179            strong_observer_tails: (0..max_strong_observers)
180                .map(|_| CachePadded::new(AtomicUsize::new(usize::MAX)))
181                .collect(),
182            n_strong_observers: Mutex::new(0),
183            put_wait_queue: Default::default(),
184            read_wait_queue: Default::default(),
185        })
186    }

188    /// Get the capacity of the queue
189    pub(crate) fn capacity(&self) -> usize {
190        self.capacity.into()
191    }

193    /// "turn" is similar to a generation counter with an extra bit for indicating whether we expect
194    /// the slot to be readable or writable.
195    fn turn(&self, index: usize, reading: bool) -> usize {
196        2 * (index / self.capacity) + if reading { 1 } else { 0 }
197    }
198
199    fn next_turn(&self, index: usize, reading: bool) -> usize {
200        self.turn(index, reading) + 2
201    }
202
203    /// The physical index in the ringbuffer.
204    fn idx(&self, index: usize) -> usize {
205        index % self.capacity
206    }
207
208    fn get_slot(&self, position: usize) -> &Slot<T> {
209        // SAFETY: self.idx(position) ensures that the index for `slots` is in bounds
210        unsafe { self.slots.get_unchecked(self.idx(position)) }
211    }
212
213    /// Advance a slot after it has been written to
214    fn mark_slot_as_written(&self, head: usize) {
215        let slot = &self.get_slot(head);
216        // Check if all tails are either uninitialized (in which case they are set to usize::MAX) or
217        // are starting to consume/observe from a position that is in the future from this head. The
218        // latter is possible if a consumer/observer attaches during a produce.
219        if self.tail.load(Ordering::Acquire) > head
220            && self
221                .strong_observer_tails
222                .iter()
223                .all(|t| t.load(Ordering::Acquire) > head)
224        {
225            // Mark the slot as writable so that the produce isn't blocked
226            slot.turn
227                .store(self.next_turn(head, false), Ordering::Release);
228        } else {
229            // Mark the slot as writeable
230            slot.turn.store(self.turn(head, true), Ordering::Release);
231        }
232    }

234    // TODO(arthurp): It may be possible to implement an optimized blocking produce.
235    fn try_produce(&self, data: T) -> Option<T>
236    where
237        T: Send,
238    {
239        loop {
240            // Get the current head position
241            let head = loop {
242                let head = self.head.load(Ordering::Acquire);
243                if head >= MPMCOQUEUE_HEAD_SENTINEL {
244                    while self.head.load(Ordering::Relaxed) >= MPMCOQUEUE_HEAD_SENTINEL {
245                        // TODO(aneesh): Revisit the need for yield - this might only be needed in our tests
246                        // that don't have preemptive scheduling.
247                        Task::yield_now();
248                    }
249                } else {
250                    break head;
251                }
252            };
253            // Get the slot that we'd like to write to
254            let slot = &self.get_slot(head);
255
256            // If it is our turn, attempt to atomically update the counter while checking to see if
257            // another producer has beaten this thread to the slot.
258            if self.turn(head, false) == slot.turn.load(Ordering::Acquire) {
259                // N.B. (aneesh): I'm not entirely sure why SeqCst is used here, but it matches
260                // rigtorp/MPMCQueue
261                if self
262                    .head
263                    .compare_exchange(head, head + 1, Ordering::SeqCst, Ordering::SeqCst)
264                    .is_ok()
265                {
266                    // We have exclusive write access to the slot, safe to write.
267                    unsafe { slot.store(data) };
268                    self.mark_slot_as_written(head);
269                    return None;
270                }
271            } else {
272                // It was not our turn, but we can opportunistically try again.
273                let prev_head = head;
274                let next_head = self.head.load(Ordering::Acquire);
275                // If it is not our turn and the `head` has not moved, the queue must be full.
276                // Return the value back to the user.
277                if next_head == prev_head {
278                    return Some(data);
279                }
280            }
281        }
282    }
283
284    fn mark_slot_as_read<const IS_CONSUMER: bool>(
285        &self,
286        curr_slot_turn: usize,
287        next_slot_turn: usize,
288        tail: usize,
289    ) {
290        let slot = unsafe { &self.slots.get_unchecked(self.idx(tail)) };
291
292        let mut prev_slot_turn = curr_slot_turn;
293        // Check if there's any StrongObservers that have not yet read this slot.
294        if (IS_CONSUMER || self.tail.load(Ordering::Acquire) >= (tail + 1))
295            && (self.strong_observer_tails.is_empty()
296                || self.strong_observer_tails.iter().all(|t| {
297                    // The slowest tail updates the slot. The check for the other tail values
298                    // must be >= and not just >, because it's better for two threads to attempt
299                    // to redundantly update the slot than to have the slot never update.
300                    t.load(Ordering::Acquire) >= (tail + 1)
301                }))
302        {
303            // There might be multiple threads updating this slot at the same time. Do a
304            // compare_exchange instead of store because we don't want to overwrite a
305            // newer value if some other consumer got to it first.
306            // N.B. (aneesh): I'm not entirely sure why SeqCst is used here, but it matches
307            // rigtorp/MPMCQueue
308            while let Err(slot_turn) = slot.turn.compare_exchange(
309                prev_slot_turn,
310                next_slot_turn,
311                Ordering::SeqCst,
312                Ordering::SeqCst,
313            ) {
314                // This slot was updated by another consumer/observer, so we don't need to
315                // update it anymore.
316                if slot_turn >= next_slot_turn {
317                    break;
318                }
319                prev_slot_turn = slot_turn;
320            }
321        }
322    }
323
324    // TODO(arthurp): It may be possible to implement an optimized blocking consume.
325    fn try_consume(&self) -> Option<T> {
326        // Unlike rigtorp/MPMCQueue, we only expose `try_consume` and not a `consume`. This is
327        // because supporting `consume` with sufficient semantics for observation is hard.
328
329        // Get the current tail position
330        let mut tail = self.tail.load(Ordering::Acquire);
331        loop {
332            // Get the slot we intend to read from
333            let slot = &self.get_slot(tail);
334            // If it is our turn, attempt to atomically update the counter while checking to see if
335            // another consumer has beaten this thread to the slot.
336            if self.turn(tail, true) == slot.turn.load(Ordering::Acquire) {
337                if self
338                    .tail
339                    .compare_exchange(tail, tail + 1, Ordering::SeqCst, Ordering::SeqCst)
340                    .is_ok()
341                {
342                    // We have exclusive consume access to the slot
343                    let v = unsafe { self.get_slot(tail).get() };
344                    let prev_slot_turn = slot.turn.load(Ordering::Acquire);
345                    let next_slot_turn = self.next_turn(tail, false);
346
347                    // Check if there's any StrongObservers that have not yet read this slot.
348                    if !STRONG_OBSERVERS {
349                        slot.turn
350                            .store(self.next_turn(tail, false), Ordering::Release);
351                    } else {
352                        self.mark_slot_as_read::<true>(prev_slot_turn, next_slot_turn, tail);
353                    }
354
355                    return Some(v);
356                }
357            } else {
358                let prev_tail = tail;
359                tail = self.tail.load(Ordering::Acquire);
360                if tail == prev_tail {
361                    return None;
362                }
363            }
364        }
365    }

367    fn try_observe(&self, observer_id: usize) -> Option<T>
368    where
369        T: Copy + Send,
370    {
371        // TODO(aneesh) instead of just reading the index @ this observer which is hard to guarantee
372        // on first read (see attach_strong_observer), what if when the index is consumed (turn >
373        // id), we reset the pointer to head and retry?
374        let tail = self.strong_observer_tails[observer_id].load(Ordering::Acquire);
375        let slot = &self.get_slot(tail);
376        if self.turn(tail, true) != slot.turn.load(Ordering::Acquire) {
377            return None;
378        }
379        // SAFETY: the check above guarantees that this slot contains valid data
380        let v = unsafe { slot.get() };
381
382        self.strong_observer_tails[observer_id].fetch_add(1, Ordering::Acquire);
383        let prev_slot_turn = slot.turn.load(Ordering::Acquire);
384        let next_slot_turn = self.next_turn(tail, false);
385        self.mark_slot_as_read::<false>(prev_slot_turn, next_slot_turn, tail);
386
387        Some(v)
388    }

390    fn size(&self) -> usize {
391        self.head
392            .load(Ordering::Relaxed)
393            .saturating_sub(self.tail.load(Ordering::Relaxed))
394    }
395
396    fn empty(&self) -> bool {
397        self.size() == 0
398    }

400    /// Observe a value in the queue based on a cursor. If value isn't available return `None`. This can happen if
401    /// either the value is outside the current range covered by the buffer or the value is overwritten while being
402    /// observed. This is all best effort and there are no specific guarantees about the availability of values.
403    ///
404    /// SAFETY: Must only be called from a single thread at a time with a given observer index. E.i., no object calling
405    /// this can be Sync, but it may be Send.
406    unsafe fn weak_observe(&self, cursor: Cursor) -> Option<T>
407    where
408        T: Copy + Send,
409    {
410        let index = cursor.index();
411
412        if self.empty() && self.head.load(Ordering::Relaxed) == 0 {
413            return None;
414        }
415
416        // Get the current turn of the slot.
417        let slot_turn = self.get_slot(index).turn.load(Ordering::Relaxed);
418        // If the turn of the slot is either (the current turn and readable), or (the next turn and
419        // writable (but not yet written)), then attempt to read.
420        let expected_turn = self.turn(index, true);
421        if slot_turn != expected_turn && slot_turn != (expected_turn + 1) {
422            return None;
423        }
424
425        let v = unsafe { self.get_slot(index).get_maybe_uninit() };
426        // Check if the turn changed while reading. This means that the value we read above is
427        // not guaranteed to be the value at `index`, so we should throw it away and return failure.
428        if self.get_slot(index).turn.load(Ordering::Acquire) != slot_turn {
429            return None;
430        }
431        Some(unsafe { v.assume_init() })
432    }

434    fn get_this(
435        &self,
436    ) -> Result<Arc<MPMCOQueue<T, STRONG_OBSERVERS, WEAK_OBSERVERS>>, OQueueAttachError> {
437        self.this
438            .upgrade()
439            .ok_or_else(|| OQueueAttachError::AllocationFailed {
440                message: "self was removed from original Arc".to_owned(),
441                table_type: "".to_owned(),
442            })

445    /// Attach either a consumer or producer to the queue.
446    /// SAFETY: This method MUST only be called from a single thread at a time.
447    unsafe fn attach_tail(&self, tail: &AtomicUsize) {
448        // If the tail is already initialized, no need to initialize it again
449        if tail.load(Ordering::Acquire) == usize::MAX {
450            // By swapping with the sentinel here we lock all producers, preventing the value at
451            // `obs_pos` from being written to.
452            let obs_pos = self.head.swap(MPMCOQUEUE_HEAD_SENTINEL, Ordering::Acquire);
453            // After this store, consumers/observers cannot consume past the index `obs_pos`. We do
454            // a compare exchange in case some other concurrent call to attach_consumer happened
455            // first. In that case, we can just continue and use the initialized value.
456            let _ = tail.compare_exchange(usize::MAX, obs_pos, Ordering::SeqCst, Ordering::SeqCst);
457            // Allow producers to write to this position again
458            self.head.swap(obs_pos, Ordering::Release);
459        }
460    }

486impl<T: Send + 'static, const STRONG_OBSERVERS: bool, const WEAK_OBSERVERS: bool> Producer<T>
487    for MPMCProducer<T, STRONG_OBSERVERS, WEAK_OBSERVERS>
488{
489    fn produce(&self, data: T) {
490        // self.oqueue.produce(data);
491        // self.oqueue.read_wait_queue.wake_all();
492        let mut data = Some(data);
493        self.oqueue.put_wait_queue.wait_until(|| {
494            if let Some(d) = self.try_produce(data.take().unwrap()) {
495                data = Some(d);
496                None
497            } else {
498                Some(())
499            }
500        })
501    }
502
503    fn try_produce(&self, data: T) -> Option<T> {
504        let res = self.oqueue.try_produce(data);
505        if res.is_none() {
506            self.oqueue.read_wait_queue.wake_all();
507        }
508        res
509    }
510}

519impl<T, const STRONG_OBSERVERS: bool, const WEAK_OBSERVERS: bool> Drop
520    for MPMCConsumer<T, STRONG_OBSERVERS, WEAK_OBSERVERS>
521{
522    fn drop(&mut self) {
523        let mut n_consumers = self.oqueue.n_consumers.lock();
524        *n_consumers -= 1;
525        if *n_consumers == 0 {
526            // Safe to write here because we hold the lock on n_consumers
527            self.oqueue.tail.store(usize::MAX, Ordering::Relaxed);
528        }
529    }
530}

587    fn enqueue(&self, waker: &Arc<Waker>) -> WakerKey {
588        self.oqueue.read_wait_queue.enqueue(waker.clone())
589    }
590
591    fn remove(&self, key: WakerKey) {
592        self.oqueue.read_wait_queue.remove(key);
593    }
594}

596impl<T: Copy + Send, const WEAK_OBSERVERS: bool> StrongObserver<T>
597    for MPMCStrongObserver<T, WEAK_OBSERVERS>
598{
599    fn strong_observe(&self) -> T {
600        self.oqueue
601            .read_wait_queue
602            .wait_until(|| self.try_strong_observe())
603    }
604
605    fn try_strong_observe(&self) -> Option<T> {
606        let res = self.oqueue.try_observe(self.observer_id);
607        if res.is_some() {
608            self.oqueue.put_wait_queue.wake_one();
609        }
610        res
611    }
612}

614/// The weak-observer handle for [`MPMCOQueue`].
615pub(crate) struct MPMCWeakObserver<T, const STRONG_OBSERVERS: bool> {
616    oqueue: Arc<MPMCOQueue<T, STRONG_OBSERVERS, true>>,
617    // Make this Send, but not Sync
618    _phantom: PhantomData<Cell<()>>,
619}
620
621impl<T, const STRONG_OBSERVERS: bool> Blocker for MPMCWeakObserver<T, STRONG_OBSERVERS> {
622    fn should_try(&self) -> bool {
623        self.oqueue.size() > 0
624    }
625
626    fn enqueue(&self, waker: &Arc<Waker>) -> WakerKey {
627        self.oqueue.read_wait_queue.enqueue(waker.clone())
628    }
629
630    fn remove(&self, key: WakerKey) {
631        self.oqueue.read_wait_queue.remove(key);
632    }
633}
634
635impl<T: Copy + Send, const STRONG_OBSERVERS: bool> WeakObserver<T>
636    for MPMCWeakObserver<T, STRONG_OBSERVERS>
637{
638    fn weak_observe(&self, cursor: Cursor) -> Option<T> {
639        // SAFETY: MPMCConsumer is Send, but not Sync, so this can only ever be called from a single thread at a time.
640        unsafe { self.oqueue.weak_observe(cursor) }
641    }
642
643    fn recent_cursor(&self) -> Cursor {
644        // This estimates the most recent element by looking at the head (which is the next slot to write) and subtracting 1.
645        Cursor(self.oqueue.head.load(Ordering::Acquire).saturating_sub(1))
646    }
647
648    fn oldest_cursor(&self) -> Cursor {
649        let Cursor(i) = self.recent_cursor();
650        // Return the most recent - the buffer size or zero if the buffer isn't full yet.
651        if i < self.oqueue.capacity.into() {
652            Cursor(0)
653        } else {
654            Cursor(i - (usize::from(self.oqueue.capacity) - 1))
655        }
656    }
657}

659impl<T: Send + 'static, const STRONG_OBSERVERS: bool, const WEAK_OBSERVERS: bool>
660    MPMCOQueue<T, STRONG_OBSERVERS, WEAK_OBSERVERS>
661{
662    fn attach_producer_internal(&self) -> Result<Box<dyn Producer<T>>, OQueueAttachError> {
663        Ok(Box::new(MPMCProducer {
664            oqueue: self.get_this()?,
665            _phantom: PhantomData,
666        }) as _)
667    }
668
669    fn attach_consumer_internal(&self) -> Result<Box<dyn Consumer<T>>, OQueueAttachError> {
670        let mut n_consumers = self.n_consumers.lock();
671        *n_consumers += 1;
672        // Prevent any strong observers from being added while the tail is initializing. Otherwise
673        // the atomic swap with the head will need some kind of retry.
674        let _ = self.n_strong_observers.lock();
675        // SAFETY: this is safe because we have the lock above.
676        unsafe { self.attach_tail(&self.tail) };
677
678        Ok(Box::new(MPMCConsumer {
679            oqueue: self.get_this()?,
680            _phantom: PhantomData,
681        }) as _)
682    }
683}

685impl<T: Copy + Send + 'static, const WEAK_OBSERVERS: bool> MPMCOQueue<T, true, WEAK_OBSERVERS> {
686    fn attach_strong_observer_internal(
687        &self,
688    ) -> Result<Box<dyn StrongObserver<T>>, OQueueAttachError> {
689        let mut n_observers = self.n_strong_observers.lock();
690        if *n_observers > (self.max_strong_observers + 1) {
691            Err(OQueueAttachError::AllocationFailed {
692                table_type: type_name::<Self>().to_owned(),
693                message: "Maximum observers already attached".to_owned(),
694            })
695        } else {
696            let observer_id = *n_observers;
697            (*n_observers) += 1;
698            // SAFETY: this is safe because we have the lock above.
699            unsafe { self.attach_tail(&self.strong_observer_tails[observer_id]) };
700
701            let oqueue = self.get_this()?;
702            Ok(Box::new(MPMCStrongObserver {
703                oqueue,
704                observer_id,
705                _phantom: PhantomData,
706            }) as _)
707        }
708    }
709}
