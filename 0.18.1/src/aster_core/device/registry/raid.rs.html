3//! RAID-1 logical block device assembly.
4//!
5//! This module assembles the RAID-1 device from its virtio member disks and
6//! registers it with the block layer. The device then gets a devtmpfs node
7//! (`/dev/raid`) like any other block device, so user space can mount a
8//! filesystem on it (e.g., `mount -t ext2 /dev/raid /raid1`).
9

31const RAID_DEVICE_NAME: &str = "raid";

33/// The kernel command-line parameter naming the RAID-1 member disks.
34///
35/// The member disks are configured via `raid.members=<name>,<name>,...`, a
36/// comma-separated list of block device names (e.g. `raid.members=vdc,vdd,vde`).
37/// The list is ordered: the position of each name is used as the member's
38/// logical index. The value is collected during early boot into [`RAID_MEMBERS`]
39/// by the [`aster_cmdline`] registration below.
40const RAID_MEMBERS_PARAM: &str = "raid.members";
41
42/// The kernel command-line parameter naming the read-selection policy.
43///
44/// Set via `raid.selection=<name>` where `<name>` is one of `roundrobin`
45/// (default), `linnos`, `linnos_plus`, `decision_tree`, or `userspace`.
46#[cfg(not(baseline_asterinas))]
47const RAID_SELECTION_PARAM: &str = "raid.selection";
48
49/// The read-selection policy used when `raid.selection` is not set.
50#[cfg(not(baseline_asterinas))]
51const DEFAULT_RAID_SELECTION: &str = "roundrobin";
52
53/// The kernel command-line parameter selecting the admission policy.
54///
55/// Set via `raid.admission=heimdall` to enable the Heimdall performance monitor.
56/// When unset, no admission policy is used and every member is always offered to
57/// the selection policy. Choosing this at runtime keeps the Heimdall code
58/// compiled in every build rather than hidden behind a `cfg`.
59#[cfg(not(baseline_asterinas))]
60const RAID_ADMISSION_PARAM: &str = "raid.admission";
61
62/// Assembles and registers the RAID-1 device.
63///
64/// This must run after the member disks' worker threads have been spawned
65/// (see `block::init_in_first_kthread`), and before `init_in_first_process`
66/// creates the devtmpfs nodes. This is called in `kernel/src/device/registry/mod.rs`
67/// after the block devices have been registered and spawned.
68pub(super) fn init_in_first_kthread() {
69    if let Err(err) = setup_raid1_device() {
70        error!("failed to set up the RAID-1 device: {:?}", err);
71    }
72}

74/// This is the setup_raid1_device() function migrated from the initialization
75/// prior to the big merge.
76fn setup_raid1_device() -> Result<()> {
77    let members = collect_members()?; // Collect member devices
78
79    #[cfg(not(baseline_asterinas))]
80    let init_result = {
81        // Expose the members' completion streams under `/oqueues` regardless of which selection
82        // policy is configured, so the same observation surface is available in every boot.
83        expose_bio_completion_oqueues(&members);
84
85        // The read-selection policy is chosen at runtime via the `raid.selection`
86        // kernel parameter (default: round-robin). Every policy is compiled
87        // unconditionally, so a change to one cannot silently bitrot another and
88        // CI exercises all of them in a single build.
89        let selection = RAID_SELECTION
90            .get()
91            .map(String::as_str)
92            .unwrap_or(DEFAULT_RAID_SELECTION);
93        let selection_policy: Arc<dyn SelectionPolicy> = match selection {
94            "roundrobin" => selection_policies::RoundRobinPolicy::new(members.clone())?,
95            "linnos" => selection_policies::LinnOSPolicy::new(
96                members.clone(),
97                attach_weak_observers(&members),
98            )?,
99            "linnos_plus" => selection_policies::LinnOSPlusPolicy::new(
100                members.clone(),
101                attach_weak_observers(&members),
102            )?,
103            "decision_tree" => selection_policies::DecisionTreePolicy::new(
104                members.clone(),
105                attach_weak_observers(&members),
106            )?,
107            "userspace" => {
108                let (request_producer, decision_consumer) = setup_userspace_policy();
109                selection_policies::UserspacePolicy::new(
110                    members.clone(),
111                    request_producer,
112                    decision_consumer,
113                )?
114            }
115            other => {
116                warn!("unknown raid.selection '{other}'; falling back to round-robin");
117                selection_policies::RoundRobinPolicy::new(members.clone())?
118            }
119        };
120
121        // Build the optional admission layer (`raid.admission` kernel parameter).
122        let admission = build_admission(&members);
123
124        Raid1Device::init(RAID_DEVICE_NAME, members, selection_policy, admission)
125    };
126    #[cfg(baseline_asterinas)]
127    let init_result = Raid1Device::init(RAID_DEVICE_NAME, members);
128
129    // `Raid1DeviceError` is a foreign (component-crate) error, and the kernel's
130    // `Error` is not being extended to carry it as a source, so bridge it here
131    // by picking the errno the caller should see. The descriptive text stays in
132    // sync with the variant's `Display`.
133    let raid_id = init_result.map_err(|err| match err {
134        Raid1DeviceError::NotEnoughMembers => {
135            Error::with_message(Errno::EINVAL, "RAID-1 device requires at least two members")
136        }
137        Raid1DeviceError::Block { .. } => {
138            Error::with_message(Errno::EEXIST, "failed to register the RAID-1 device")
139        }
140    })?;
141
142    spawn_worker_thread(raid_id);
143
144    info!(
145        "RAID-1 device '{}' registered and worker thread spawned",
146        RAID_DEVICE_NAME
147    );
148    Ok(())
149}

151/// Builds the optional admission layer selected at runtime via the
152/// `raid.admission` kernel parameter. Returns `None` (admission disabled) unless
153/// `raid.admission=heimdall`, in which case the Heimdall monitor thread is also
154/// spawned.
155#[cfg(not(baseline_asterinas))]
156fn build_admission(
157    members: &[Arc<dyn aster_block::BlockDevice>],
158) -> Option<Arc<aster_raid::heimdall::Heimdall>> {
159    match RAID_ADMISSION.get().map(String::as_str) {
160        // Default: no admission policy.
161        None => None,
162        Some("heimdall") => Some(setup_heimdall(members)),
163        Some(other) => {
164            warn!("unknown raid.admission '{other}'; admission disabled");
165            None
166        }
167    }
168}

170/// Attaches one weak observer per member to its bio-completion OQueue, each
171/// wrapped in a `Mutex` as the observer-based selection policies expect.
172///
173/// # Panics
174///
175/// Panics if a member is not a `VirtIoBlockDevice`.
176#[cfg(not(baseline_asterinas))]
177fn attach_weak_observers(
178    members: &[Arc<dyn aster_block::BlockDevice>],
179) -> Vec<Mutex<ostd::orpc::oqueue::WeakObserver<aster_block::bio::BlockDeviceCompletionStats>>> {
180    use aster_virtio::device::block::server_traits::BlockIOObservable as _;
181    use ostd::orpc::oqueue::{OQueueBase as _, ObservationQuery};
182
183    members
184        .iter()
185        .map(|member| {
186            let virtio = member
187                .downcast_ref::<VirtIoBlockDevice>()
188                .expect("RAID member must be a VirtIoBlockDevice");
189            Mutex::new(
190                virtio
191                    .bio_completion_oqueue()
192                    .attach_weak_observer(4, ObservationQuery::identity())
193                    .expect("Failed to attach weak observer to bio_completion_oqueue"),
194            )
195        })
196        .collect()
197}

199/// Exposes each member's `bio_completion` OQueue over OQFS, at
200/// `/oqueues/raid1/bio_completion/<index>/`.
201///
202/// Each OQueue is exposed via a projection onto [`BioCompletionStatsMessage`], a plain-field mirror
203/// of the whole [`aster_block::bio::BlockDeviceCompletionStats`] with a derived `Serialize` (a CBOR map keyed by
204/// field name).
205#[cfg(not(baseline_asterinas))]
206fn expose_bio_completion_oqueues(members: &[Arc<dyn aster_block::BlockDevice>]) {
207    use aster_block::bio::BlockDeviceCompletionStats;
208    use aster_virtio::device::block::server_traits::BlockIOObservable as _;
209    use ostd::orpc::oqueue::{OQueueBase as _, registry};
210
211    for (index, member) in members.iter().enumerate() {
212        let virtio = member
213            .downcast_ref::<VirtIoBlockDevice>()
214            .expect("RAID member must be a VirtIoBlockDevice");
215        registry::register_with_projection(
216            &path!(raid1.bio_completion[{ index }]),
217            &virtio.bio_completion_oqueue().as_any_oqueue(),
218            |stats: &BlockDeviceCompletionStats| BioCompletionStatsMessage::from(*stats),
219        );
220    }
221}

223/// A `Copy` projection of [`aster_block::bio::BlockDeviceCompletionStats`] exposing every field to userspace (see
224/// [`expose_bio_completion_oqueues`]).
225///
226/// Mirrored by `BioCompletionStats` in `test/initramfs/src/raid_policy_server/src/main.rs`.
227#[cfg(not(baseline_asterinas))]
228#[derive(Debug, Clone, Copy, serde::Serialize)]
229struct BioCompletionStatsMessage {
230    latency_us: u64,
231    outstanding_pages: u32,
232    queue_len: u32,
233    request_size_pages: u32,
234    device_id: u32,
235}
236
237#[cfg(not(baseline_asterinas))]
238impl From<aster_block::bio::BlockDeviceCompletionStats> for BioCompletionStatsMessage {
239    fn from(stats: aster_block::bio::BlockDeviceCompletionStats) -> Self {
240        Self {
241            latency_us: stats.latency.as_micros() as u64,
242            outstanding_pages: stats.outstanding_pages,
243            queue_len: stats.queue_len,
244            request_size_pages: stats.request_size_pages,
245            device_id: stats.device_id,
246        }
247    }
248}

250/// Creates the RAID-1 selection request OQueue (`/oqueues/raid1/selection_request/`, kernel -> user)
251/// and decision OQueue (`/oqueues/raid1/decision/`, user -> kernel), returning the request producer
252/// and decision consumer handed to [`selection_policies::UserspacePolicy`].
253#[cfg(not(baseline_asterinas))]
254fn setup_userspace_policy() -> (
255    ostd::orpc::oqueue::ValueProducer<selection_policies::SelectionRequestMessage>,
256    ostd::orpc::oqueue::Consumer<u32>,
257) {
258    use ostd::orpc::oqueue::{
259        ConsumableOQueue as _, ConsumableOQueueRef, OQueueBase as _, registry,
260    };
261
262    let request_path = path!(raid1.selection_request);
263    let request_oqueue = ConsumableOQueueRef::new(SELECTION_QUEUE_CAPACITY, request_path.clone());
264    // Register the request queue so user space policy server can observe requests.
265    registry::register(&request_path, &request_oqueue.as_any_oqueue());
266    let request_producer = request_oqueue
267        .attach_value_producer()
268        .expect("the RAID-1 selection request OQueue always allows a value producer");
269
270    let decision_path = path!(raid1.decision);
271    let decision_oqueue =
272        ConsumableOQueueRef::<u32>::new(SELECTION_QUEUE_CAPACITY, decision_path.clone());
273    registry::register_producible(&decision_path, &decision_oqueue);
274    let decision_consumer = decision_oqueue
275        .attach_consumer()
276        .expect("the RAID-1 decision OQueue always allows attaching its consumer");
277
278    (request_producer, decision_consumer)
279}

301/// Creates the Heimdall performance monitor over `members` and spawns its
302/// background inference thread. The returned handle is shared with the RAID
303/// device so the admission decisions stay in sync with the monitor.
304#[cfg(not(baseline_asterinas))]
305fn setup_heimdall(
306    members: &[Arc<dyn aster_block::BlockDevice>],
307) -> Arc<aster_raid::heimdall::Heimdall> {
308    use aster_virtio::device::block::server_traits::BlockIOObservable as _;
309    use ostd::orpc::oqueue::{OQueueBase as _, ObservationQuery};
310
311    let device_observers = members
312        .iter()
313        .map(|member| {
314            let virtio = member
315                .downcast_ref::<VirtIoBlockDevice>()
316                .expect("RAID member must be a VirtIoBlockDevice");
317            let observer = virtio
318                .bio_completion_oqueue()
319                .attach_strong_observer(ObservationQuery::identity())
320                .expect("Failed to attach strong observer for Heimdall");
321            (member.clone(), observer)
322        })
323        .collect();
324
325    // Heimdall's inference cadence is configurable via the kernel command line
326    // (`heimdall.batch_size`, `heimdall.inference_timeout_ms`); when unset, the
327    // storage statics keep their default initializers (see below).
328    let batch_size = HEIMDALL_BATCH_SIZE.load(Ordering::Relaxed) as usize;
329    let inference_timeout_ms = HEIMDALL_INFERENCE_TIMEOUT_MS.load(Ordering::Relaxed) as u64;
330
331    // The observers are owned by the monitor thread (not the shared handle), so
332    // `new` hands them back for `run` to take ownership of.
333    let (heimdall, observers) =
334        aster_raid::heimdall::Heimdall::new(device_observers, batch_size, inference_timeout_ms)
335            .expect("Failed to create Heimdall monitor");
336
337    let monitor = heimdall.clone();
338    ThreadOptions::new(move || {
339        info!("[heimdall] Heimdall monitor thread started");
340        monitor.run(observers);
341    })
342    .sched_policy(SchedPolicy::RealTime {
343        rt_prio: 50.try_into().unwrap(),
344        rt_policy: RealTimePolicy::RoundRobin {
345            base_slice_factor: None,
346        },
347    })
348    .spawn();
349
350    info!("[heimdall] Heimdall monitor initialized and thread spawned");
351    heimdall
352}

354/// Collects the RAID-1 member devices named on the kernel command line.
355///
356/// The member disk names are read from the `raid.members` parameter (see
357/// [`RAID_MEMBERS_PARAM`]) rather than hardcoded, so the disk layout can be
358/// reconfigured from the kernel command-line config (`kcmd_configs/*.toml`) and
359/// `tools/qemu_args.sh` without touching the kernel. Devices are looked up by name via [`aster_block::collect_all`]
360/// (the refactored registry keys devices by `DeviceId`, and virtio disks are
361/// auto-named `vda`, `vdb`, ... in probe order).
362fn collect_members() -> Result<Vec<Arc<dyn aster_block::BlockDevice>>> {
363    let Some(members_arg) = RAID_MEMBERS.get() else {
364        return_errno_with_message!(
365            Errno::EINVAL,
366            "the RAID-1 member disks are not configured (set 'raid.members=<name>,...')"
367        );
368    };
369
370    let member_names: Vec<&str> = members_arg
371        .split(',')
372        .map(str::trim)
373        .filter(|name| !name.is_empty())
374        .collect();
375
376    let all_devices = aster_block::collect_all();
377
378    let mut members = Vec::with_capacity(member_names.len());
379    for name in &member_names {
380        let Some(device) = all_devices.iter().find(|device| device.name() == *name) else {
381            return_errno_with_message!(Errno::ENOENT, "a RAID-1 member disk is missing");
382        };
383        members.push(device.clone());
384    }
385
386    Ok(members)
387}

389fn spawn_worker_thread(raid_id: DeviceId) {
390    let device = aster_block::lookup(raid_id).unwrap();
391    let task_fn = move || {
392        info!("spawn the RAID-1 device thread");
393        let raid = device.downcast_ref::<Raid1Device>().unwrap();
394        loop {
395            raid.handle_requests();
396        }
397    };
398
399    // Elevate to RealTime 50 so this I/O thread is not starved by other
400    // threads, which would otherwise cause long tail latencies.
401    ThreadOptions::new(task_fn)
402        .sched_policy(SchedPolicy::RealTime {
403            rt_prio: 50.try_into().unwrap(),
404            rt_policy: RealTimePolicy::RoundRobin {
405                base_slice_factor: None,
406            },
407        })
408        .spawn();
409}
