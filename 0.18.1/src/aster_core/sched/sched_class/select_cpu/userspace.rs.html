17/// Sends every task to the CPU that userspace wrote most recently.
18///
19/// Userspace writes a CPU number, CBOR-encoded, to `/oqueues/scheduler/select_cpu/produce`. A task
20/// whose affinity mask does not contain that CPU, or any task placed before the first write, is
21/// placed as [`LastCpuPolicy`] would place it.
22pub(crate) struct UserspacePolicy {
23    attachments: SpinLock<Attachments, LocalIrqDisabled>,
24    fallback: LastCpuPolicy,
25}
26
27struct Attachments {
28    /// Reads the most recent CPU written by userspace.
29    observer: WeakObserver<u32>,
30    /// Discards each written value. Without a consumer, the OQueue warns on every write.
31    _consumer: InlineConsumer<u32>,
32}
33
34impl UserspacePolicy {
35    /// Creates the policy and exports its OQueue at `/oqueues/scheduler/select_cpu`.
36    ///
37    /// # Panics
38    ///
39    /// Panics if that OQueue is already exported, i.e. if this is called twice.
40    pub(crate) fn new() -> Self {
41        let path = path!(scheduler.select_cpu);
42        let oqueue = ConsumableOQueueRef::<u32>::new(OQUEUE_CAPACITY, path.clone());
43        registry::register_producible(&path, &oqueue);
44        let observer = oqueue
45            .attach_weak_observer(1, ObservationQuery::identity())
46            .expect("a new OQueue accepts a weak observer");
47        let consumer = oqueue
48            .attach_consumer()
49            .and_then(|consumer| consumer.consume_inline(|_| {}))
50            .expect("a new OQueue accepts a consumer");
51        Self {
52            attachments: SpinLock::new(Attachments {
53                observer,
54                _consumer: consumer,
55            }),
56            fallback: LastCpuPolicy::new(),
57        }
58    }
59
60    /// Returns the CPU most recently written by userspace, if any.
61    fn requested_cpu(&self) -> Option<CpuId> {
62        let attachments = self.attachments.lock();
63        let observer = &attachments.observer;
64        let cpu = observer.weak_observe(observer.newest_cursor()).ok()??;
65        CpuId::try_from(cpu as usize).ok()
66    }
67}
68
69impl SelectCpuPolicy for UserspacePolicy {
70    fn select_cpu(&self, input: &SelectCpuInput) -> CpuId {
71        let affinity = input.thread.atomic_cpu_affinity().load(Ordering::Relaxed);
72        match self.requested_cpu() {
73            Some(cpu) if affinity.contains(cpu) => cpu,
74            _ => self.fallback.select_cpu(input),
75        }
76    }
77}
78
79/// The OQueue capacity. Producing never blocks, because the OQueue has no strong observer and its
80/// consumer runs inline, so a small ring is enough.
81const OQUEUE_CAPACITY: usize = 2;
