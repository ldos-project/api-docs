57/// The CPU-selection policy used when `sched.select_cpu` is not set.

63aster_cmdline::define_kv_param!("sched.select_cpu", SELECT_CPU_POLICY);

68pub(crate) fn init() {
69    let scheduler: &'static ClassScheduler = Box::leak(Box::new(ClassScheduler::new()));
70    CLASS_SCHEDULER.call_once(|| scheduler);
71
72    // Inject the scheduler into the ostd for actual scheduling work.
73    inject_scheduler(scheduler);
74
75    // Set the scheduler into the system for statistics.
76    // We set this after injecting the scheduler into ostd,
77    // so that the loadavg statistics are updated after the scheduler is used.
78    set_stats_from_scheduler(scheduler);
79}

81/// Installs the CPU-selection policy named by `sched.select_cpu`.
82///
83/// Must be called after `component::init_all(InitStage::Bootstrap)`, which parses the kernel
84/// command line, and before any task is spawned or woken.
85pub(crate) fn init_select_cpu_policy() {
86    let requested = SELECT_CPU_POLICY
87        .get()
88        .map(String::as_str)
89        .unwrap_or(DEFAULT_SELECT_CPU_POLICY);
90    let (name, policy): (&str, Arc<dyn SelectCpuPolicy>) = match requested {
91        "last_cpu" => ("last_cpu", Arc::new(LastCpuPolicy::new())),
92        "round_robin" => ("round_robin", Arc::new(RoundRobinPolicy::new())),
93        #[cfg(not(baseline_asterinas))]
94        "userspace" => ("userspace", Arc::new(UserspacePolicy::new())),
95        other => {
96            warn!("unknown sched.select_cpu '{other}'; keeping 'last_cpu'");
97            ("last_cpu", Arc::new(LastCpuPolicy::new()))
98        }
99    };
100    inject_select_cpu_policy(policy);
101    info!("sched: CPU-selection policy is '{name}'");
102}

106    CLASS_SCHEDULER.get().unwrap().inject_select_policy(policy);

113/// Represents the middle layer between scheduling classes and generic scheduler
114/// traits. It consists of all the sets of run queues for CPU cores. Other global
115/// information may also be stored here.
116pub(crate) struct ClassScheduler {
117    /// The per-CPU runqueues.
118    ///
119    /// We use the `LocalIrqDisabled` marker for this spinlock to ensure local IRQs are always disabled,
120    /// preventing potential deadlocks due to the fact that
121    /// the runqueues may be accessed in both the task and interrupt context (L1 and L2).
122    rqs: Box<[SpinLock<PerCpuClassRqSet, LocalIrqDisabled>]>,
123
124    /// The CPU-selection policy, set by [`Self::inject_select_policy`].
125    select_policy: PolicySlot<dyn SelectCpuPolicy>,
126}
127
128/// Represents the run queue for each CPU core. It stores a list of run queues for
129/// scheduling classes in its corresponding CPU core. The current task of this CPU
130/// core is also stored in this structure.
131struct PerCpuClassRqSet {
132    stop: stop::StopClassRq,
133    real_time: real_time::RealTimeClassRq,
134    fair: fair::FairClassRq,
135    idle: idle::IdleClassRq,
136    current: Option<(SchedEntity, CurrentRuntime)>,
137    /// The task just preempted on this CPU, while it waits to be queued again.
138    ///
139    /// A preempted task is still runnable, so the CPU-selection policy decides which run queue it
140    /// joins next. That queue may belong to another CPU, and its lock cannot be taken while this
141    /// one is held, so `try_pick_next` leaves the task here and
142    /// [`ClassScheduler::mut_local_rq_with`] queues it after releasing this lock.
143    preempted: Option<SchedEntity>,
144}
145
146/// Stores the runtime information of the current task.
147///
148/// This is used to calculate the time slice of the current task.
149///
150/// This struct is independent of the current `Arc<Task>` instead encapsulating the
151/// task, because the scheduling class implementations use `CurrentRuntime` and
152/// `SchedAttr` only.
153struct CurrentRuntime {
154    start: u64,
155    delta: u64,
156    period_delta: u64,
157}
158
159impl CurrentRuntime {
160    fn new() -> Self {
161        CurrentRuntime {
162            start: sched_clock(),
163            delta: 0,
164            period_delta: 0,
165        }
166    }
167
168    fn update(&mut self) {
169        let now = sched_clock();
170        self.delta = now - core::mem::replace(&mut self.start, now);
171        self.period_delta += self.delta;
172    }
173}

175/// The run queue for scheduling classes (the main trait). Scheduling classes
176/// should implement this trait to function as expected.
177trait SchedClassRq: Send + fmt::Debug {
178    /// Enqueues a task into the run queue.
179    fn enqueue(&mut self, task: Arc<Task>, flags: Option<EnqueueFlags>);
180
181    /// Returns the number of threads in the run queue.
182    fn len(&self) -> usize;
183
184    /// Checks if the run queue is empty.
185    fn is_empty(&self) -> bool {
186        self.len() == 0
187    }
188
189    /// Removes the task from the run queue, if it is queued. Returns true if it was present.
190    ///
191    /// The task must have the same priority it did when it was inserted, because the priority can
192    /// be used to find it in the queues.
193    ///
194    /// Not every run queue supports removing a specific task. Run queues that don't will always
195    /// return `false`.
196    fn remove(&mut self, task: &Arc<Task>) -> bool;
197
198    /// Picks the next task for running.
199    fn pick_next(&mut self) -> Option<Arc<Task>>;
200
201    /// Update the information of the current task.
202    ///
203    /// The return value of this method indicates whether there is another task
204    /// **in this run queue** to replace the current one.
205    fn update_current(&mut self, rt: &CurrentRuntime, attr: &SchedAttr, flags: UpdateFlags)
206    -> bool;
207}

209/// The scheduling attribute for a thread.
210///
211/// This is used to store the scheduling policy and runtime parameters for each
212/// scheduling class.
213#[derive(Debug)]
214pub(crate) struct SchedAttr {
215    policy: SchedPolicyState,
216    /// The CPU that the task is running on or was most recently scheduled on.
217    ///
218    /// This is distinct from [`ostd::task::scheduler::info::CommonSchedInfo::cpu()`] because it is
219    /// not cleared when the thread is descheduled.
220    last_cpu: AtomicCpuId,
221    real_time: real_time::RealTimeAttr,
222    fair: fair::FairAttr,
223    /// The policy before a forced schedule, if the thread's policy has been
224    /// temporarily changed by [`SchedAttr::set_forced_policy`].
225    prev_policy: SpinLock<Option<SchedPolicy>>,
226}
227
228impl SchedAttr {
229    /// Constructs a new `SchedAttr` with the given scheduling policy.
230    pub(crate) fn new(policy: SchedPolicy) -> Self {
231        Self {
232            policy: SchedPolicyState::new(policy),
233            last_cpu: AtomicCpuId::default(),
234            real_time: {
235                let (prio, policy) = match policy {
236                    SchedPolicy::RealTime { rt_prio, rt_policy } => (rt_prio.get(), rt_policy),
237                    _ => (RealTimePriority::MAX.get(), Default::default()),
238                };
239                real_time::RealTimeAttr::new(prio, policy)
240            },
241            fair: fair::FairAttr::new(match policy {
242                SchedPolicy::Fair(nice) => nice,
243                _ => Nice::default(),
244            }),
245            prev_policy: SpinLock::new(None),
246        }
247    }
248
249    /// Retrieves the current scheduling policy of the thread.
250    pub(crate) fn policy(&self) -> SchedPolicy {
251        self.policy.get()
252    }
253
254    /// Retrieves the scheduling policy that a thread cloned from this one starts with.
255    ///
256    /// This is the thread's own policy, not the one it may temporarily have for a forced
257    /// schedule.
258    pub(crate) fn inheritable_policy(&self) -> SchedPolicy {
259        // The scheduler tick may restore the policy on this CPU, so keep IRQs off while locked.
260        let prev_policy = self.prev_policy.disable_irq().lock();
261        prev_policy.unwrap_or_else(|| self.policy())
262    }
263
264    fn policy_kind(&self) -> SchedPolicyKind {
265        self.policy.kind()
266    }
267
268    /// Updates the scheduling policy of the thread.
269    ///
270    /// Specifically for real-time policies, if the new policy doesn't
271    /// specify a base slice factor for RR, the old one will be kept.
272    pub(crate) fn set_policy(&self, policy: SchedPolicy) {
273        self.policy.set(policy, |policy| match policy {
274            SchedPolicy::RealTime { rt_prio, rt_policy } => {
275                self.real_time.update(rt_prio.get(), rt_policy);
276            }
277            SchedPolicy::Fair(nice) => self.fair.update(nice),
278            _ => {}
279        });
280    }
281
282    /// Temporarily sets the highest real-time priority for a forced schedule.
283    ///
284    /// The current policy is saved, if not already saved, so that it can be
285    /// restored by [`SchedAttr::restore_forced_policy`].
286    fn set_forced_policy(&self) {
287        let mut prev = self.prev_policy.lock();
288        if prev.is_none() {
289            *prev = Some(self.policy());
290        }
291        self.set_policy(SchedPolicy::RealTime {
292            rt_prio: RealTimePriority::MIN,
293            rt_policy: RealTimePolicy::Fifo,
294        });
295    }
296
297    /// Restores the policy that was saved by [`SchedAttr::set_forced_policy`].
298    fn restore_forced_policy(&self) {
299        if let Some(policy) = self.prev_policy.lock().take() {
300            self.set_policy(policy);
301        }
302    }
303
304    pub(crate) fn update_policy<T>(&self, f: impl FnOnce(&mut SchedPolicy) -> T) -> T {
305        self.policy.update(|policy| {
306            let ret = f(policy);
307            match *policy {
308                SchedPolicy::RealTime { rt_prio, rt_policy } => {
309                    self.real_time.update(rt_prio.get(), rt_policy);
310                }
311                SchedPolicy::Fair(nice) => self.fair.update(nice),
312                _ => {}
313            }
314            ret
315        })
316    }
317
318    /// The CPU that the thread is running on or was most recently scheduled on.
319    pub(crate) fn last_cpu(&self) -> Option<CpuId> {
320        self.last_cpu.get()
321    }
322
323    /// Set the CPU this thread is being scheduled on.
324    fn set_last_cpu(&self, cpu_id: CpuId) {
325        self.last_cpu.set_anyway(cpu_id);
326    }
327}

329impl Scheduler for ClassScheduler {
330    fn enqueue(&self, task: Arc<Task>, flags: EnqueueFlags) -> Option<CpuId> {
331        let thread = task.as_thread()?.clone();
332
333        let (still_in_rq, cpu) = {
334            let selected_cpu_id = self.select_cpu(&thread, flags.into());
335
336            // If the task had already run on a CPU, use that.
337            if let Err(task_cpu_id) = task.cpu().set_if_is_none(selected_cpu_id) {
338                debug_assert!(flags != EnqueueFlags::Spawn);
339                (true, task_cpu_id)
340            } else {
341                (false, selected_cpu_id)
342            }
343        };
344
345        let mut rq = self.rqs[cpu.as_usize()].lock();
346
347        // Note: call set_if_is_none again to prevent a race condition.
348        let still_in_rq = still_in_rq && task.cpu().set_if_is_none(cpu).is_err();
349
350        if flags == EnqueueFlags::ForceSchedule {
351            // A forced task is temporarily assigned the highest real-time priority and is placed at
352            // the front of the run queue. Its previous policy is restored on the next
353            // `update_current` of the task. If the task is still in the run queue, it must be
354            // removed from its queue first; if that is not possible (the task is the current task,
355            // or it is in a run queue which doesn't support removal), the force fails entirely and
356            // falls back to normal scheduling.
357            if still_in_rq && !rq.remove_task(&task) {
358                warn!("Failed to force schedule task: {:?}", task);
359            } else {
360                thread.sched_attr().set_forced_policy();
361                rq.real_time.enqueue_front(task);
362                thread.sched_attr().set_last_cpu(cpu);
363                return Some(cpu);
364            }
365        }
366
367        if still_in_rq {
368            return None;
369        }
370
371        // Preempt if the new task has a higher priority.
372        let should_preempt = rq
373            .current
374            .as_ref()
375            .is_none_or(|((_, rq_current_thread), _)| {
376                thread.sched_attr().policy() < rq_current_thread.sched_attr().policy()
377            });
378
379        thread.sched_attr().set_last_cpu(cpu);
380        rq.enqueue_entity((task, thread), Some(flags));
381
382        should_preempt.then_some(cpu)
383    }
384
385    fn mut_local_rq_with(&self, f: &mut dyn FnMut(&mut dyn LocalRunQueue)) {
386        let guard = disable_local();
387        let preempted = {
388            let mut rq = self.rqs[guard.current_cpu().as_usize()].lock();
389            f(&mut *rq);
390            rq.preempted.take()
391        };
392        if let Some(entity) = preempted {
393            self.requeue_preempted(entity);
394        }
395    }
396
397    fn local_rq_with(&self, f: &mut dyn FnMut(&dyn LocalRunQueue)) {
398        let guard = disable_local();
399        f(&*self.rqs[guard.current_cpu().as_usize()].lock())
400    }
401
402    fn idle_time(&self) -> Duration {
403        let mut idle_time = Duration::from_nanos(0);
404        for class_rq in &self.rqs {
405            idle_time += class_rq.lock().idle.get_idle_time();
406        }
407        idle_time + Duration::from_secs(0)
408    }
409}

411impl ClassScheduler {
412    pub(crate) fn new() -> Self {
413        let class_rq = |cpu| {
414            SpinLock::new(PerCpuClassRqSet {
415                stop: stop::StopClassRq::new(),
416                real_time: real_time::RealTimeClassRq::new(cpu),
417                fair: fair::FairClassRq::new(cpu),
418                idle: idle::IdleClassRq::new(),
419                current: None,
420                preempted: None,
421            })
422        };
423        ClassScheduler {
424            rqs: all_cpus().map(class_rq).collect(),
425            select_policy: PolicySlot::empty(),
426        }
427    }
428
429    /// Queues a task that was preempted while still runnable on the CPU chosen by the policy.
430    fn requeue_preempted(&self, (task, thread): SchedEntity) {
431        let cpu = self.select_cpu(&thread, SelectCpuReason::Preempt);
432        let mut rq = self.rqs[cpu.as_usize()].lock();
433        // Move the run-queue marker straight to the new CPU. It must never pass through `None`,
434        // which `PosixThread::sleeping_state` reads as "asleep".
435        task.cpu().set_anyway(cpu);
436        thread.sched_attr().set_last_cpu(cpu);
437        // The task may still be running on its old CPU. If the new CPU picks it first, OSTD's
438        // context switch waits for the old CPU to let go of it.
439        rq.enqueue_entity((task, thread), None);
440    }
441
442    /// Replaces the CPU-selection policy. Tasks placed from now on use `policy`.
443    pub(crate) fn inject_select_policy(&self, policy: Arc<dyn SelectCpuPolicy>) {
444        self.select_policy.set(policy);
445    }
446
447    /// Asks the active CPU-selection policy where `thread` should be queued.
448    fn select_cpu(&self, thread: &Arc<Thread>, reason: SelectCpuReason) -> CpuId {
449        // `SelectCpuPolicy::select_cpu` promises policies that local IRQs are disabled.
450        let _guard = disable_local();
451        let queue_len = |cpu: CpuId| self.rqs[cpu.as_usize()].lock().load_stats().queue_len;
452        let input = SelectCpuInput {
453            reason,
454            thread,
455            queue_len: &queue_len,
456        };
457        self.select_policy
458            .with(|policy| policy.select_cpu(&input))
459            .expect("no CPU-selection policy is installed: a task was spawned or woken before `init_select_cpu_policy` ran")
460    }
461}

463impl PerCpuClassRqSet {
464    fn pick_next_entity(&mut self) -> Option<SchedEntity> {
465        (self.stop.pick_next())
466            .or_else(|| self.real_time.pick_next())
467            .or_else(|| self.fair.pick_next())
468            .or_else(|| self.idle.pick_next())
469            .and_then(|task| {
470                let thread = task.as_thread()?.clone();
471                Some((task, thread))
472            })
473    }
474
475    fn enqueue_entity(&mut self, (task, thread): SchedEntity, flags: Option<EnqueueFlags>) {
476        match thread.sched_attr().policy_kind() {
477            SchedPolicyKind::Stop => self.stop.enqueue(task, flags),
478            SchedPolicyKind::RealTime => self.real_time.enqueue(task, flags),
479            SchedPolicyKind::Fair => self.fair.enqueue(task, flags),
480            SchedPolicyKind::Idle => self.idle.enqueue(task, flags),
481        }
482    }
483
484    /// Removes the task from the run queues of this CPU, if it is queued in one of the classes that
485    /// supports that.
486    fn remove_task(&mut self, task: &Arc<Task>) -> bool {
487        let Some(t) = task.as_thread() else {
488            return false;
489        };
490        match t.sched_attr().policy_kind() {
491            SchedPolicyKind::Stop => self.stop.remove(task),
492            SchedPolicyKind::RealTime => self.real_time.remove(task),
493            SchedPolicyKind::Fair => self.fair.remove(task),
494            SchedPolicyKind::Idle => self.idle.remove(task),
495        }
496    }
497
498    fn load_stats(&self) -> PerCpuLoadStats {
499        let queue_len = (self.stop.len() + self.real_time.len() + self.fair.len()) as u32;
500        let is_idle = match &self.current {
501            Some(((_, thread), _)) => thread.sched_attr().policy_kind() == SchedPolicyKind::Idle,
502            None => true,
503        };
504        PerCpuLoadStats { queue_len, is_idle }
505    }
506}

508impl LocalRunQueue for PerCpuClassRqSet {
509    fn current(&self) -> Option<&Arc<Task>> {
510        self.current.as_ref().map(|((task, _), _)| task)
511    }
512
513    fn try_pick_next(&mut self) -> Option<&Arc<Task>> {
514        self.pick_next_entity().and_then(|next| {
515            // We guarantee that a task can appear at once in a `PerCpuClassRqSet`. So, the `next` cannot be the same
516            // as the current task here.
517            if let Some((old, _)) = self.current.replace((next, CurrentRuntime::new())) {
518                // `old` is still runnable. `ClassScheduler::mut_local_rq_with` places it after
519                // this lock is released, because the policy may send it to another CPU.
520                debug_assert!(self.preempted.is_none());
521                self.preempted = Some(old);
522            }
523            self.current.as_ref().map(|((task, _), _)| task)
524        })
525    }
526
527    fn update_current(&mut self, flags: UpdateFlags) -> bool {
528        let (should_preempt, mut lookahead) = if let Some(((_, cur), rt)) = &mut self.current {
529            rt.update();
530
531            // A forced schedule only lasts until the next `update_current` of
532            // the task: restore the original policy, if any.
533            cur.sched_attr().restore_forced_policy();
534
535            let attr = &cur.sched_attr();
536
537            match attr.policy_kind() {
538                SchedPolicyKind::Stop => (self.stop.update_current(rt, attr, flags), 0),
539                SchedPolicyKind::RealTime => (self.real_time.update_current(rt, attr, flags), 1),
540                SchedPolicyKind::Fair => (self.fair.update_current(rt, attr, flags), 2),
541                SchedPolicyKind::Idle => (self.idle.update_current(rt, attr, flags), 3),
542            }
543        } else {
544            (false, 4)
545        };
546
547        if matches!(flags, UpdateFlags::Wait | UpdateFlags::Exit) {
548            lookahead = 4;
549        }
550
551        should_preempt
552            || (lookahead >= 1 && !self.stop.is_empty())
553            || (lookahead >= 2 && !self.real_time.is_empty())
554            || (lookahead >= 3 && !self.fair.is_empty())
555            || (lookahead >= 4 && !self.idle.is_empty())
556    }
557
558    fn dequeue_current(&mut self) -> Option<Arc<Task>> {
559        self.current.take().map(|((cur_task, _), _)| {
560            cur_task.schedule_info().cpu.set_to_none();
561            cur_task
562        })
563    }
564}

566/// Holds per-CPU load information.
567struct PerCpuLoadStats {
568    /// The length of the run queue (excluding the idle task).
569    queue_len: u32,
570    /// If the CPU is currently idle.
571    ///
572    /// A CPU is said to be idle when it is running the idle task, or it is not
573    /// running any task at all. The latter case is very unlikely to happen
574    /// (almost a bug if it happens) as the idle task should always be runnable.
575    is_idle: bool,
576}
577
578impl SchedulerStats for ClassScheduler {
579    fn nr_queued_and_running(&self) -> (u32, u32) {
580        self.rqs.iter().fold((0, 0), |(queued, running), rq| {
581            let PerCpuLoadStats { queue_len, is_idle } = rq.lock().load_stats();
582            (queued + queue_len, running + u32::from(!is_idle))
583        })
584    }
585}
