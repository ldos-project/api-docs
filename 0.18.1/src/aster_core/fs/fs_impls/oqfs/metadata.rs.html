3//! The `metadata.yaml` file: a human-readable description of an OQueue.
4//!
5//! It is a regular file whose contents are regenerated on each read from the live export registry,
6//! reporting the queue's path (`name`/`oqueues`) and message `type_name`.
7

28/// The name of the metadata file inside each OQueue directory.
29pub(super) const FILE_NAME: &str = "metadata.yaml";
30
31/// The inode for a `metadata.yaml` file.
32pub(super) struct MetadataInode {
33    /// The OQueue path this file describes.
34    path: Path,
35    common: Common,
36}
37
38/// Creates a `metadata.yaml` inode for the OQueue at `path`.
39pub(super) fn new_inode(fs: Weak<OQueueFs>, path: Path) -> Arc<dyn Inode> {
40    let oqueue_fs = fs.upgrade().unwrap();
41    let ino = oqueue_fs.alloc_id();
42    // Root-only, like every other OQFS inode: `Metadata::new_file` always sets the owner to root,
43    // so owner-only bits mean only root can read this. The whole filesystem is restricted, not
44    // just the data files -- a world-readable `metadata.yaml` would let an unprivileged process
45    // enumerate the message type of every exported OQueue.
46    let metadata = Metadata::new_file(
47        ino,
48        mkmod!(u+r),
49        BLOCK_SIZE,
50        oqueue_fs.sb().container_dev_id,
51    );
52    let fs: Weak<dyn FileSystem> = fs;
53    Arc::new(MetadataInode {
54        path,
55        common: Common::new(metadata, fs),
56    })
57}
58
59impl MetadataInode {
60    /// Renders the YAML description, skipping the first `offset` bytes for the current read.
61    fn render(&self, offset: usize, writer: &mut VmWriter) -> Result<usize> {
62        let path = self.path.to_string();
63        let type_name = registry::lookup_export(&self.path)
64            .map(|export| export.type_name())
65            .unwrap_or("<unregistered>");
66
67        let mut printer = VmPrinter::new_skip(writer, offset);
68        writeln!(printer, "name: {}", path)?;
69        writeln!(printer, "oqueues:")?;
70        writeln!(printer, "  - {}", path)?;
71        writeln!(printer, "type_name: {}", type_name)?;
72        Ok(printer.bytes_written())
73    }
74}
75
76impl FileOps for MetadataInode {
77    fn read_at(
78        &self,
79        offset: usize,
80        writer: &mut VmWriter,
81        _status_flags: StatusFlags,
82    ) -> Result<usize> {
83        self.render(offset, writer)
84    }
85
86    fn write_at(
87        &self,
88        _offset: usize,
89        _reader: &mut VmReader,
90        _status_flags: StatusFlags,
91    ) -> Result<usize> {
92        Err(Error::new(Errno::EPERM))
93    }
94}
