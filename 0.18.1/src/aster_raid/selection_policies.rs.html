70/// hidden_layers, hidden_biases, output_layers, output_biases: machine learning model weights.
71/// There is one model per device. Each model contains three layers, an input layer,
72/// a hidden layer with 256 neurons, and an output layer with 2 neurons for the binary
73/// classification (fast/slow). Thus, there are two weight matrices and two bias vectors
74/// per device: a 31x256 matrix + 256 bias for the hidden layer and a 256x2 matrix + 2
75/// bias for the output layer.
76/// Each latency number is decomposed into 4 digits, and each number of outstanding
77/// request number is decomposed into 3 digits. Thus, the total number of input features
78/// is 3+4*(3+4) = 31. The number of history is R=4.
79///
80/// Number of Outstanding Requests: number of pending 4KB pages.
81/// List of observers attached to the list of member block devices.
82#[orpc_server]
83pub struct LinnOSPolicy {
84    read_cursor: AtomicUsize,
85    members: Vec<Arc<dyn BlockDevice>>,
86    observers: Vec<Mutex<ostd::orpc::oqueue::WeakObserver<BlockDeviceCompletionStats>>>,
87    hidden_layers: Vec<[[f32; 256]; 31]>,
88    hidden_biases: Vec<[f32; 256]>,
89    output_layers: Vec<[[f32; 2]; 256]>,
90    output_biases: Vec<[f32; 2]>,
91}
92
93impl core::fmt::Debug for LinnOSPolicy {
94    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
95        f.debug_struct("LinnOSPolicy")
96            .field("read_cursor", &self.read_cursor)
97            .field("members", &self.members)
98            .field(
99                "observers",
100                &format_args!("[{} observers]", self.observers.len()),
101            )
102            .finish()
103    }
104}

127            hidden_layers,
128            hidden_biases,
129            output_layers,
130            output_biases,
131            orpc_internal,
132        });
133
134        Ok(server)
135    }

138impl SelectionPolicy for LinnOSPolicy {
139    fn select_block_device(&self, selection: BioCandidates) -> Result<Arc<dyn BlockDevice>, Error> {
140        let candidates = selection.candidates;
141        let num_candidates = candidates.len();
142        let mut fail_cnt = 0;
143        let num_pages = selection.bio.num_pages();
144
145        loop {
146            let idx = self.read_cursor.fetch_add(1, Ordering::Relaxed);
147            let device_idx = candidates[idx % num_candidates];
148            let observer = self.observers[device_idx].lock();
149            let completion_trace = observer
150                .weak_observe_recent(4)
151                .expect("Failed to observe completion trace");
152
153            // Build the 31-element input feature vector:
154            //   [0..3]:  current outstanding pages (3 digits, from most recent trace)
155            //   For each history step i (0..4):
156            //     [3+i*7 .. 3+i*7+3]: outstanding pages (3 digits)
157            //     [3+i*7+3 .. 3+i*7+7]: latency in microseconds (4 digits)
158            let mut input = [0.0f32; 31];
159
160            // Current outstanding pages: use most recent trace entry, decompose into 3 digits
161            let current_outstanding =
162                num_pages as usize + self.members[device_idx].num_outstanding_pages() as usize;
163            input[0] = ((current_outstanding / 100) % 10) as f32;
164            input[1] = ((current_outstanding / 10) % 10) as f32;
165            input[2] = (current_outstanding % 10) as f32;
166
167            // Feature Engineering in LinnOS: Decompose numbers into digits.
168            // Historical features: 4 steps, each with 3 digits outstanding + 4 digits latency
169            let mut observed: [(usize, usize); 4] = [(0, 0); 4];
170            for (i, trace_entry) in completion_trace.iter().enumerate().take(4) {
171                let Some(trace_entry) = trace_entry else {
172                    continue;
173                };
174                let outstanding = trace_entry.outstanding_pages as usize;
175                let latency_us = trace_entry.latency.as_micros() as usize;
176                let base = 3 + i * 7;
177
178                observed[i] = (outstanding, latency_us);
179
180                // Outstanding pages -> 3 digits (hundreds, tens, ones)
181                input[base] = ((outstanding / 100) % 10) as f32;
182                input[base + 1] = ((outstanding / 10) % 10) as f32;
183                input[base + 2] = (outstanding % 10) as f32;
184
185                // Latency in microseconds -> 4 digits (thousands, hundreds, tens, ones)
186                input[base + 3] = ((latency_us / 1000) % 10) as f32;
187                input[base + 4] = ((latency_us / 100) % 10) as f32;
188                input[base + 5] = ((latency_us / 10) % 10) as f32;
189                input[base + 6] = (latency_us % 10) as f32;
190            }
191
192            // FPU math: disable preemption for the matrix multiplication.
193            // Kernel-space FPU state is not saved across a context switch, so a
194            // preempting user thread could clobber the floating-point
195            // accumulators mid-computation.
196            let predicted_fast = {
197                let _guard = ostd::task::disable_preempt();
198
199                // Hidden layer: input (31) x hidden_weights (31x256) + bias (256) -> hidden_out (256)
200                let hidden_weights = &self.hidden_layers[device_idx];
201                let hidden_bias = &self.hidden_biases[device_idx];
202                let mut hidden_out = [0.0f32; 256];
203                for j in 0..256 {
204                    let mut sum = hidden_bias[j];
205                    for i in 0..31 {
206                        sum += input[i] * hidden_weights[i][j];
207                    }
208                    // ReLU activation
209                    hidden_out[j] = if sum > 0.0 { sum } else { 0.0 };
210                }
211
212                // Output layer: hidden_out (256) x output_weights (256x2) + bias (2) -> output (2)
213                let output_weights = &self.output_layers[device_idx];
214                let output_bias = &self.output_biases[device_idx];
215                let mut output = [output_bias[0], output_bias[1]];
216                for k in 0..2 {
217                    for j in 0..256 {
218                        output[k] += hidden_out[j] * output_weights[j][k];
219                    }
220                }
221
222                // Argmax: output[0] < output[1] means fast, otherwise slow
223                output[0] < output[1]
224            };
225
226            if predicted_fast {
227                return Ok(self.members[device_idx].clone());
228            }
229
230            fail_cnt += 1;
231            // All candidates predicted slow -- fall back to round-robin among them
232            if fail_cnt >= num_candidates {
233                let fallback_idx =
234                    candidates[self.read_cursor.fetch_add(1, Ordering::Relaxed) % num_candidates];
235                // log::info!("Submitting to device {} as all devices are busy. output=[{:.4},{:.4}]", fallback_idx, output[0], output[1]);
236                return Ok(self.members[fallback_idx].clone());
237            }
238        }
239    }
240}

286impl SelectionPolicy for DecisionTreePolicy {
287    fn select_block_device(&self, selection: BioCandidates) -> Result<Arc<dyn BlockDevice>, Error> {
288        use crate::decision_tree_predictions::{predict_device0, predict_device1, predict_device2};
289
290        let candidates = selection.candidates;
291        let num_candidates = candidates.len();
292        let mut fail_cnt = 0;
293        let num_pages = selection.bio.num_pages();
294
295        loop {
296            let idx = self.read_cursor.fetch_add(1, Ordering::Relaxed);
297            let device_idx = candidates[idx % num_candidates];
298            let observer = self.observers[device_idx].lock();
299            let completion_trace = observer
300                .weak_observe_recent(4)
301                .expect("Failed to observe completion trace");
302
303            // Build the 31-element input feature vector as u8 digits (0–9)
304            let mut input = [0u8; 31];
305
306            let current_outstanding =
307                num_pages as usize + self.members[device_idx].num_outstanding_pages() as usize;
308            input[0] = ((current_outstanding / 100) % 10) as u8;
309            input[1] = ((current_outstanding / 10) % 10) as u8;
310            input[2] = (current_outstanding % 10) as u8;
311
312            for (i, trace_entry) in completion_trace.iter().enumerate().take(4) {
313                let Some(trace_entry) = trace_entry else {
314                    continue;
315                };
316                let outstanding = trace_entry.outstanding_pages as usize;
317                let latency_us = trace_entry.latency.as_micros() as usize;
318                let base = 3 + i * 7;
319
320                input[base] = ((outstanding / 100) % 10) as u8;
321                input[base + 1] = ((outstanding / 10) % 10) as u8;
322                input[base + 2] = (outstanding % 10) as u8;
323
324                input[base + 3] = ((latency_us / 1000) % 10) as u8;
325                input[base + 4] = ((latency_us / 100) % 10) as u8;
326                input[base + 5] = ((latency_us / 10) % 10) as u8;
327                input[base + 6] = (latency_us % 10) as u8;
328            }
329
330            let prediction = match device_idx {
331                0 => predict_device0(&input),
332                1 => predict_device1(&input),
333                2 => predict_device2(&input),
334                _ => 1, // unknown device: predict fast
335            };
336
337            if prediction == 1 {
338                return Ok(self.members[device_idx].clone());
339            }
340
341            fail_cnt += 1;
342            if fail_cnt >= num_candidates {
343                let fallback_idx =
344                    candidates[self.read_cursor.fetch_add(1, Ordering::Relaxed) % num_candidates];
345                return Ok(self.members[fallback_idx].clone());
346            }
347        }
348    }

524/// Max candidate indices carried in one [`SelectionRequestMessage`]; extras are dropped with a warning.
525const MAX_REQUEST_CANDIDATES: usize = 8;
526
527/// Request sent to the userspace policy server: the admitted candidate member indices for the read
528/// being routed. Fixed-size to stay `Copy` (required for OQueue observation).
529///
530/// Mirrored by `SelectionRequest` in `test/initramfs/src/raid_policy_server/src/main.rs`.
531#[derive(Debug, Clone, Copy, serde::Serialize)]
532pub struct SelectionRequestMessage {
533    candidate_count: u32,
534    candidates: [u32; MAX_REQUEST_CANDIDATES],
535}
536
537impl SelectionRequestMessage {
538    /// Pack an array of candidate indicis to a SelectionRequestMessage
539    fn from_candidates(candidates: &[usize]) -> Self {
540        if candidates.len() > MAX_REQUEST_CANDIDATES {
541            log::warn!(
542                "[raid] selection request has {} candidates, more than the {MAX_REQUEST_CANDIDATES} supported; truncating",
543                candidates.len()
544            );
545        }
546        let mut wire = [0u32; MAX_REQUEST_CANDIDATES];
547        let mut count = 0u32;
548        for (slot, &candidate) in wire.iter_mut().zip(candidates) {
549            *slot = candidate as u32;
550            count += 1;
551        }
552        Self {
553            candidate_count: count,
554            candidates: wire,
555        }
556    }
557}

559/// Max wait for the server's reply before falling back, so a hung server can't stall RAID I/O.
560const REPLY_TIMEOUT_MS: u64 = 200;
561
562/// The endpoints of one synchronous request/reply exchange, kept under one lock so at most one
563/// exchange is in flight at a time.
564struct SelectionChannel {
565    request_producer: ValueProducer<SelectionRequestMessage>,
566    reply_consumer: Consumer<u32>,
567    reply_timeout: Arc<TimeoutBlocker>,
568    block_on_many: BlockOnMany,
569}

571/// Selection policy driven by a userspace server over OQFS: each call sends the admitted candidates
572/// and blocks (bounded by `REPLY_TIMEOUT_MS`) for the server's reply.
573///
574/// Falls back to round-robin among the candidates when no server is attached (e.g. early boot or
575/// after it exits), on a reply timeout, or on a reply naming a no-longer-admitted candidate.
576#[orpc_server]
577pub struct UserspacePolicy {
578    read_cursor: AtomicUsize,
579    members: Vec<Arc<dyn BlockDevice>>,
580    channel: Mutex<SelectionChannel>,
581}

592impl UserspacePolicy {
593    pub fn new(
594        members: Vec<Arc<dyn BlockDevice>>,
595        request_producer: ValueProducer<SelectionRequestMessage>,
596        reply_consumer: Consumer<u32>,
597    ) -> Result<Arc<Self>, Error> {
598        let channel = SelectionChannel {
599            request_producer,
600            reply_consumer,
601            reply_timeout: TimeoutBlocker::new(),
602            block_on_many: BlockOnMany::new(),
603        };
604        let server = Self::new_with(|orpc_internal, _| Self {
605            read_cursor: AtomicUsize::new(0),
606            members,
607            channel: Mutex::new(channel),
608            orpc_internal,
609        });
610
611        Ok(server)
612    }
613
614    /// Produces `candidates` as a request and blocks for the server's reply, bounded by
615    /// [`REPLY_TIMEOUT_MS`]. Returns `None` (without producing anything) if no server is attached
616    /// to the request stream, or with a warning if the wait times out.
617    fn request_selection(&self, candidates: &[usize]) -> Option<usize> {
618        let mut channel = self.channel.lock();
619        let SelectionChannel {
620            request_producer,
621            reply_consumer,
622            reply_timeout,
623            block_on_many,
624        } = &mut *channel;
625
626        // If no Userspace policy server is attached.
627        if !request_producer.has_consumers() {
628            return None;
629        }
630        request_producer.produce(SelectionRequestMessage::from_candidates(candidates));
631
632        reply_timeout.arm_after(REPLY_TIMEOUT_MS * TIMER_FREQ / 1000);
633        let reply = loop {
634            if let Some(value) = reply_consumer.try_consume() {
635                break Some(value);
636            }
637            // if time out.
638            if reply_timeout.should_try() {
639                break None;
640            }
641            let blockers: [&dyn Blocker; 2] = [&*reply_consumer, &**reply_timeout];
642            block_on_many.block_on(blockers.into_iter());
643        };
644        reply_timeout.disarm();
645
646        match reply {
647            Some(value) => Some(value as usize),
648            None => {
649                log::warn!(
650                    "[raid] userspace policy server did not reply within {REPLY_TIMEOUT_MS}ms; falling back"
651                );
652                None
653            }
654        }
655    }
656}
657
658impl SelectionPolicy for UserspacePolicy {
659    fn select_block_device(&self, selection: BioCandidates) -> Result<Arc<dyn BlockDevice>, Error> {
660        let candidates = selection.candidates;
661
662        if let Some(device_idx) = self.request_selection(candidates) {
663            if candidates.contains(&device_idx) {
664                return Ok(self.members[device_idx].clone());
665            }
666            // The server named a device that is not currently admitted (or predates a change in
667            // membership); fall through to the round-robin fallback.
668            log::warn!(
669                "[raid] userspace policy server replied with device {device_idx}, not in the \
670                 current candidate set; falling back to round-robin"
671            );
672        }
673
674        let fallback_idx =
675            candidates[self.read_cursor.fetch_add(1, Ordering::Relaxed) % candidates.len()];
676        Ok(self.members[fallback_idx].clone())
677    }
678}
