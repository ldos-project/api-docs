11/// A combined trait for block devices that support I/O performance tracing.
12///
13/// This trait combines `BlockDevice` (basic block I/O operations) with
14/// `BlockIOObservable` (I/O performance trace queues), enabling policies
15/// like `LinnOSPolicy` to observe completion traces for intelligent device selection.
16#[cfg(not(baseline_asterinas))]
17pub trait ObservableBlockDevice: BlockDevice + BlockIOObservable + Debug {}
18
19/// Blanket implementation: any type implementing both traits automatically
20/// implements `ObservableBlockDevice`.
21#[cfg(not(baseline_asterinas))]
22impl<T: BlockDevice + BlockIOObservable + Debug> ObservableBlockDevice for T {}

24pub struct BioCandidates<'a> {
25    /// The request being routed.
26    pub bio: &'a mut SubmittedBio,
27    /// The admitted member indices to choose from (never empty).
28    pub candidates: &'a [usize],
29}
30
31#[orpc_trait]
32pub trait SelectionPolicy: Debug {
33    /// Chooses the member device to read from among the admitted `candidates`.
34    ///
35    /// The policy must return one of the devices whose index appears in
36    /// `selection.candidates`. If the policy cannot decide, for whatever reason,
37    /// this should return an error; the caller will use some fallback.
38    fn select_block_device(&self, selection: BioCandidates) -> Result<Arc<dyn BlockDevice>, Error>;
39}
