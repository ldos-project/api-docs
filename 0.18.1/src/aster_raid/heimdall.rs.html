19/// Heimdall: an asynchronous device performance monitor for RAID-1 arrays.
20///
21/// Heimdall maintains one ML model per member device. A dedicated background
22/// thread ([`Heimdall::run`]) owns the per-device strong observers and
23/// continuously drains completion stats from each device's OQueue. Every
24/// `batch_size` completions, it runs an ML inference to update that device's
25/// fast/slow indicator. Or the inference is triggered if `inference_timeout_ms`
26/// elapses since the last inference for that device. Whichever condition is met
27/// first.
28///
29/// Model architecture (per device):
30///   Linear(INPUT_DIM, 16) -> ReLU -> Linear(16, 1) -> Sigmoid
31///
32/// The observers are deliberately *not* stored in this shared handle: they are
33/// owned solely by the monitor thread, so this struct stays `Sync` and can be
34/// shared with the RAID device via `Arc` without wrapping the (`!Sync`)
35/// observers in a lock.
36///
37/// Selection policies can query `is_device_fast(idx)` to incorporate Heimdall's
38/// classification into their scheduling decisions.
39pub struct Heimdall {
40    members: Vec<Arc<dyn BlockDevice>>,
41    /// Per-device fast/slow indicator. `true` means the device is predicted fast.
42    fast_indicators: Vec<AtomicBool>,
43    /// Per-device fc1 weights: [INPUT_DIM][HIDDEN_SIZE]
44    fc1_weights: Vec<[[f32; HIDDEN_SIZE]; INPUT_DIM]>,
45    /// Per-device fc1 biases: [HIDDEN_SIZE]
46    fc1_biases: Vec<[f32; HIDDEN_SIZE]>,
47    /// Per-device fc3 weights: [HIDDEN_SIZE]
48    fc3_weights: Vec<[f32; HIDDEN_SIZE]>,
49    /// Per-device fc3 biases: scalar
50    fc3_biases: Vec<f32>,
51    /// Number of completion records to drain before running an inference.
52    batch_size: usize,
53    /// Inference timeout in milliseconds.
54    inference_timeout_ms: u64,
55}
56

76impl Heimdall {
77    /// Creates a new Heimdall monitor.
78    ///
79    /// `device_observers` — one `(member device, strong observer)` pair per
80    /// RAID-1 member; the observer is attached to the member's bio completion
81    /// OQueue.
82    /// `batch_size` — completion records to drain before running an inference.
83    /// `inference_timeout_ms` — inference timeout in milliseconds.
84    ///
85    /// Returns the shared handle together with the per-member observers. The
86    /// caller must hand the observers to [`Self::run`] on the monitor thread.
87    pub fn new(
88        device_observers: Vec<(
89            Arc<dyn BlockDevice>,
90            StrongObserver<BlockDeviceCompletionStats>,
91        )>,
92        batch_size: usize,
93        inference_timeout_ms: u64,
94    ) -> Result<(Arc<Self>, Vec<StrongObserver<BlockDeviceCompletionStats>>), Error> {
95        use crate::heimdall_weights::{FC1_BIASES, FC1_WEIGHTS, FC3_BIASES, FC3_WEIGHTS};
96
97        let num_devices = device_observers.len();
98
99        let (members, observers): (Vec<_>, Vec<_>) = device_observers.into_iter().unzip();
100
101        let fast_indicators: Vec<AtomicBool> = (0..num_devices)
102            .map(|_| AtomicBool::new(true)) // optimistic: assume fast initially
103            .collect();
104
105        let fc1_weights: Vec<_> = (0..num_devices).map(|i| *FC1_WEIGHTS[i]).collect();
106        let fc1_biases: Vec<_> = (0..num_devices).map(|i| *FC1_BIASES[i]).collect();
107        let fc3_weights: Vec<_> = (0..num_devices).map(|i| *FC3_WEIGHTS[i]).collect();
108        let fc3_biases: Vec<_> = (0..num_devices).map(|i| FC3_BIASES[i]).collect();
109
110        log::info!("Heimdall created with {} devices", fast_indicators.len());
111
112        let heimdall = Arc::new(Self {
113            members,
114            fast_indicators,
115            fc1_weights,
116            fc1_biases,
117            fc3_weights,
118            fc3_biases,
119            batch_size,
120            inference_timeout_ms,
121        });
122
123        Ok((heimdall, observers))
124    }

126    /// Returns whether device `idx` is currently classified as fast.
127    pub fn is_device_fast(&self, idx: usize) -> bool {
128        self.fast_indicators[idx].load(Ordering::Relaxed)
129    }

131    /// Returns the fast indicator for device `idx`.
132    ///
133    /// `true` means the device is currently predicted fast; `false` means slow.
134    pub fn check_device(&self, idx: usize) -> bool {
135        self.fast_indicators[idx].load(Ordering::Relaxed)
136    }

138    /// The number of member devices being monitored.
139    pub fn num_devices(&self) -> usize {
140        self.members.len()
141    }
142
143    /// Main monitoring loop. This should be spawned on a dedicated thread, which
144    /// takes ownership of `observers` (one per member, in member order).
145    ///
146    /// For each device, drains completion records from its strong observer.
147    /// Inference is triggered for a device when either condition is met first:
148    ///   1. `batch_size` records have been drained, or
149    ///   2. `inference_timeout_ms` have elapsed since the last inference.
150    pub fn run(&self, observers: Vec<StrongObserver<BlockDeviceCompletionStats>>) {
151        let num_devices = self.members.len();
152        // TIMER_FREQ is in Hz, so this converts the millisecond timeout to jiffies.
153        let timeout_jiffies = self.inference_timeout_ms * ostd::timer::TIMER_FREQ / 1000;
154
155        // Per-device batch buffers for accumulating stats between inferences.
156        let mut batch_buffers: Vec<Vec<BlockDeviceCompletionStats>> = (0..num_devices)
157            .map(|_| Vec::with_capacity(self.batch_size))
158            .collect();
159
160        // Per-device jiffies timestamp of the last inference (or loop start).
161        let now = Jiffies::elapsed().as_u64();
162        let mut last_inference_jiffies = alloc::vec![now; num_devices];
163
164        let timeout = TimeoutBlocker::new();
165        let mut block_on_many = BlockOnMany::new();
166
167        loop {
168            let mut earliest_deadline: Option<u64> = None;
169            for device_idx in 0..num_devices {
170                let observer = &observers[device_idx];
171
172                // Drain all available records (non-blocking).
173                loop {
174                    match observer.try_strong_observe() {
175                        Ok(Some(stats)) => {
176                            batch_buffers[device_idx].push(stats);
177
178                            // Condition 1: batch is full.
179                            // Do device inference, then break to give other devices a turn.
180                            if batch_buffers[device_idx].len() >= self.batch_size {
181                                self.run_inference(device_idx, &mut batch_buffers[device_idx]);
182                                last_inference_jiffies[device_idx] = Jiffies::elapsed().as_u64();
183                                break;
184                            }
185                        }
186                        Ok(None) => break, // queue is empty, move on to timeout check
187                        Err(OQueueError::Revoked { .. }) => {
188                            log::warn!("Heimdall: observer for device {} detached", device_idx);
189                            break;
190                        }
191                        Err(e) => {
192                            log::warn!("Heimdall: error observing device {}: {:?}", device_idx, e);
193                            break;
194                        }
195                    }
196                }
197
198                // Condition 2: timeout elapsed and there is at least some data.
199                let elapsed = Jiffies::elapsed()
200                    .as_u64()
201                    .wrapping_sub(last_inference_jiffies[device_idx]);
202                if elapsed >= timeout_jiffies && !batch_buffers[device_idx].is_empty() {
203                    self.run_inference(device_idx, &mut batch_buffers[device_idx]);
204                    last_inference_jiffies[device_idx] = Jiffies::elapsed().as_u64();
205                }
206
207                if !batch_buffers[device_idx].is_empty() {
208                    let deadline =
209                        last_inference_jiffies[device_idx].saturating_add(timeout_jiffies);
210                    earliest_deadline =
211                        Some(earliest_deadline.map_or(deadline, |d| d.min(deadline)));
212                }
213            }
214
215            // Arm the timeout only if at least one device has a pending partial batch;
216            match earliest_deadline {
217                Some(deadline) => timeout.arm_at(deadline),
218                None => timeout.disarm(),
219            }
220
221            let blockers = observers
222                .iter()
223                .map(|observer| observer as &dyn Blocker)
224                .chain(core::iter::once(&*timeout as &dyn Blocker));
225            block_on_many.block_on(blockers);
226        }
227    }

229    /// Run inference for a single device and update its fast indicator.
230    fn run_inference(&self, device_idx: usize, batch: &mut Vec<BlockDeviceCompletionStats>) {
231        // Model output: 1 → slow (reject IO), 0 → fast (accept IO).
232        let is_slow = self.infer_device_speed(device_idx, batch);
233        self.fast_indicators[device_idx].store(!is_slow, Ordering::Relaxed);
234        batch.clear();
235    }
236
237    /// Build the 11-element input feature vector from a batch of completion stats.
238    ///
239    /// Features (matching HeimdallNet training input):
240    ///   [0]  queue_len_now      — queue_len of the most recent record
241    ///   [1]  size_now           — request_size_pages of the most recent record
242    ///   [2]  hist_que_len_t-1   — queue_len of the 2nd-most-recent record
243    ///   [3]  hist_que_len_t-2   — queue_len of the 3rd-most-recent record
244    ///   [4]  hist_que_len_t-3   — queue_len of the 4th-most-recent record
245    ///   [5]  hist_lat_t-1       — latency_us of the 2nd-most-recent record
246    ///   [6]  hist_lat_t-2       — latency_us of the 3rd-most-recent record
247    ///   [7]  hist_lat_t-3       — latency_us of the 4th-most-recent record
248    ///   [8]  hist_thpt_t-1      — request_size_pages / latency_us (2nd-most-recent)
249    ///   [9]  hist_thpt_t-2      — request_size_pages / latency_us (3rd-most-recent)
250    ///   [10] hist_thpt_t-3      — request_size_pages / latency_us (4th-most-recent)
251    fn build_features(&self, batch: &[BlockDeviceCompletionStats]) -> [f32; INPUT_DIM] {
252        let mut input = [0.0f32; INPUT_DIM];
253        let n = batch.len();
254        if n == 0 {
255            return input;
256        }
257
258        // Most recent record: "now" features.
259        let now = &batch[n - 1];
260        input[0] = now.queue_len as f32;
261        input[1] = now.request_size_pages as f32;
262
263        // Historical records: t-1 = batch[n-2], t-2 = batch[n-3], t-3 = batch[n-4].
264        // Missing history (batch too small) stays 0.0.
265        for hist in 0..3usize {
266            let idx = n.wrapping_sub(hist + 2);
267            if idx < n {
268                let rec = &batch[idx];
269                input[2 + hist] = rec.queue_len as f32;
270                let latency_us = rec.latency.as_micros() as u64;
271                input[5 + hist] = latency_us as f32;
272                input[8 + hist] = if latency_us > 0 {
273                    rec.request_size_pages as f32 / latency_us as f32
274                } else {
275                    0.0
276                };
277            }
278        }
279
280        input
281    }

283    /// Run the HeimdallNet forward pass for `device_idx` on the given batch.
284    ///
285    /// Each device has its own model weights.
286    /// Architecture:
287    ///   Linear(INPUT_DIM, 16) -> ReLU -> Linear(16, 1) -> Sigmoid
288    ///
289    /// Returns `true` if the model output >= 0.5 (sigmoid(logit) >= 0.5 ⟺ logit >= 0).
290    fn infer_device_speed(&self, device_idx: usize, batch: &[BlockDeviceCompletionStats]) -> bool {
291        let _guard = disable_preempt();
292        let input = self.build_features(batch);
293
294        // fc1: input (INPUT_DIM) x fc1_weights (INPUT_DIM x HIDDEN_SIZE) + bias -> ReLU
295        let w1 = &self.fc1_weights[device_idx];
296        let b1 = &self.fc1_biases[device_idx];
297        let mut h1 = [0.0f32; HIDDEN_SIZE];
298        for j in 0..HIDDEN_SIZE {
299            let mut sum = b1[j];
300            for i in 0..INPUT_DIM {
301                sum += input[i] * w1[i][j];
302            }
303            h1[j] = if sum > 0.0 { sum } else { 0.0 }; // ReLU
304        }
305
306        // fc3: h1 (HIDDEN_SIZE) x fc3_weights (HIDDEN_SIZE) + bias -> Sigmoid
307        let w3 = &self.fc3_weights[device_idx];
308        let b3 = self.fc3_biases[device_idx];
309        let mut logit = b3;
310        for j in 0..HIDDEN_SIZE {
311            logit += h1[j] * w3[j];
312        }
313
314        // Sigmoid: 1 / (1 + exp(-x)).  Equivalent to: logit >= 0.
315        // We skip the actual sigmoid computation since we only need the
316        // threshold comparison at 0.5.
317        logit >= 0.0
318    }
