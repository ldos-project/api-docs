3//! RAID-1 virtual block device
4//!
5//! This module implements a simple RAID-1 adapter that exposes a single logical
6//! `BlockDevice` backed by multiple member devices (mirrors). Requests are
7//! admitted through a software queue, then translated into per-member BIOs:
8//!
9//! - Reads: each BIO is dispatched to one selected member (round-robin) and
10//!   submitted asynchronously so different BIOs can overlap across devices.
11//! - Writes: the same data is fanned out to all members; completion is reported
12//!   only after all replicas complete.
13//!
14//! Capacity and limits are the minimum across members (e.g., `nr_sectors`,
15//! `max_nr_segments_per_bio`). Error handling is conservative: any failed child
16//! write makes the parent fail.
17
18#![no_std] // BlockDevice crate also not using rust std and not using unsafe code.
19#![deny(unsafe_code)]

62/// A RAID-1 block device that mirrors I/O to multiple member devices.
63#[derive(Debug)]
64#[orpc_server]
65pub struct Raid1Device {
66    /// Member block devices that store identical data (mirrors).
67    members: Vec<Arc<dyn BlockDevice>>,
68    queue: BioRequestSingleQueue,
69
70    /// Basic capacity limits for the logical device (min across members).
71    metadata: BlockDeviceMeta,
72
73    /// A cursor for selecting the next member for read operations (round-robin).
74    #[cfg(baseline_asterinas)]
75    read_cursor: AtomicUsize,
76
77    /// The policy to select the read member.
78    #[cfg(not(baseline_asterinas))]
79    selection_policy: Arc<dyn SelectionPolicy>,
80
81    /// The optional admission policy that pre-filters which members a read may
82    /// target. When present, only the admitted members are offered to the
83    /// selection policy.
84    #[cfg(not(baseline_asterinas))]
85    admission: Option<Arc<heimdall::Heimdall>>,
86
87    /// Cached list of all member indices (`0..members.len()`), used as the
88    /// candidate set when no admission policy is active. Avoids a per-read
89    /// allocation on the common path.
90    #[cfg(not(baseline_asterinas))]
91    all_indices: Vec<usize>,
92
93    name: String,
94    id: DeviceId,
95}
96
97#[derive(Debug, Snafu)]
98pub enum Raid1DeviceError {
99    #[snafu(display("RAID-1 device requires at least two members"))]
100    NotEnoughMembers,
101    #[snafu(display("failed to register the RAID-1 device"))]
102    Block { source: aster_block::Error },
103}
104
105/// The major ID of the RAID device class.
106static RAID_MAJOR_ID: Once<MajorIdOwner> = Once::new();
107
108/// A counter that hands out a unique minor ID to each instantiated RAID device.
109static NR_RAID_DEVICE: AtomicU32 = AtomicU32::new(0);
110
111/// Allocates a device ID for a new RAID device.
112/// The RAID class major ID is allocated lazily on first use;
113/// the minor ID is a simple counter.
114fn allocate_device_id() -> Result<DeviceId, Raid1DeviceError> {
115    if RAID_MAJOR_ID.get().is_none() {
116        let major = aster_block::allocate_major().context(BlockSnafu)?;
117        RAID_MAJOR_ID.call_once(|| major);
118    }
119    let major = RAID_MAJOR_ID.get().unwrap().get();
120    let minor = NR_RAID_DEVICE.fetch_add(1, Ordering::Relaxed);
121    log::info!(
122        "[raid] allocating device ID for RAID-1 device: major={}, minor={}",
123        major.get(),
124        minor
125    );
126    Ok(DeviceId::new(major, MinorId::new(minor)))
127}

129impl Raid1Device {
130    /// Creates a new RAID-1 device backed by `members`.
131    ///
132    /// # Panics
133    ///
134    /// Panics if fewer than two members are provided.
135    pub fn init(
136        name: &str,
137        members: Vec<Arc<dyn BlockDevice>>,
138        #[cfg(not(baseline_asterinas))] selection_policy: Arc<dyn SelectionPolicy>,
139        #[cfg(not(baseline_asterinas))] admission: Option<Arc<heimdall::Heimdall>>,
140    ) -> Result<DeviceId, Raid1DeviceError> {
141        ensure!(members.len() >= 2, NotEnoughMembersSnafu);
142
143        let id = allocate_device_id()?;
144
145        // Compute the minimal metadata across all members.
146        let metadata = Self::min_metadata(&members);
147        // Initialize the admission queue using the strictest segment limit.
148        let queue =
149            BioRequestSingleQueue::with_max_nr_segments_per_bio(metadata.max_nr_segments_per_bio);
150
151        #[cfg(not(baseline_asterinas))]
152        let all_indices = (0..members.len()).collect();
153
154        #[cfg(not(baseline_asterinas))]
155        let device = Self::new_with(|orpc_internal, _weak_self| Raid1Device {
156            members,
157            queue,
158            metadata,
159            selection_policy,
160            admission,
161            all_indices,
162            name: name.to_owned(),
163            id,
164            orpc_internal,
165        });
166        #[cfg(baseline_asterinas)]
167        let device = Arc::new_cyclic(|_weak_self| Raid1Device {
168            members,
169            queue,
170            metadata,
171            read_cursor: AtomicUsize::new(0),
172            name: name.to_owned(),
173            id,
174        });
175
176        aster_block::register(device.clone()).context(BlockSnafu)?;
177
178        Ok(id)
179    }

181    /// Dequeues and processes the next request from the staging queue.
182    pub fn handle_requests(&self) {
183        let request = self.queue.dequeue();
184        self.process_request(request);
185    }
186
187    /// Dispatches a request by type. The RAID-1 device accepts the same BIOs as
188    /// any `BlockDevice` and applies RAID semantics underneath.
189    fn process_request(&self, request: BioRequest) {
190        match request.type_() {
191            BioType::Read => self.process_read_async(request),
192            BioType::Write => self.process_write_async(request),
193            BioType::Flush => self.process_flush(request),
194        }
195    }
196
197    /// Chooses the member device to serve a read.
198    ///
199    /// The admission policy (if any) first pre-filters the members down to those
200    /// it admits; the selection policy then chooses among the admitted members.
201    /// If the admission policy rejects every member, all members are offered so
202    /// that I/O is never stalled.
203    #[cfg(not(baseline_asterinas))]
204    fn select_member(&self, bio: &mut SubmittedBio) -> Arc<dyn BlockDevice> {
205        // Fast path: no admission policy — offer every member without allocating.
206        let Some(heimdall) = &self.admission else {
207            let selection = BioCandidates {
208                bio,
209                candidates: &self.all_indices,
210            };
211            return self
212                .selection_policy
213                .select_block_device(selection)
214                .unwrap();
215        };
216
217        // Admission active: offer only the members Heimdall currently admits,
218        // falling back to all members if it admits none (so I/O is never stalled).
219        let admitted: Vec<usize> = self
220            .all_indices
221            .iter()
222            .copied()
223            .filter(|&index| heimdall.is_device_fast(index))
224            .collect();
225        let candidates: &[usize] = if admitted.is_empty() {
226            &self.all_indices
227        } else {
228            &admitted
229        };
230        let selection = BioCandidates { bio, candidates };
231        self.selection_policy
232            .select_block_device(selection)
233            .unwrap()
234    }
235
236    /// Processes read requests asynchronously.
237    ///
238    /// Each `SubmittedBio` in the merged `BioRequest` is assigned to a read
239    /// member by the selection policy (device 0 if asterinas baseline) and submitted with `Bio::submit` to overlap device
240    /// I/O. Completion of the parent is reported after the child finishes.
241    fn process_read_async(&self, request: BioRequest) {
242        for parent in request.into_bios() {
243            // `select_member` needs `&mut parent` (to compute `num_pages`); the
244            // baseline path always uses device 0 and never mutates the parent.
245            #[cfg(not(baseline_asterinas))]
246            let mut parent = parent;
247
248            #[cfg(not(baseline_asterinas))]
249            let member = self.select_member(&mut parent);
250
251            #[cfg(baseline_asterinas)]
252            let member = self.members[0].clone();
253
254            let start_sid = parent.sid_range().start;
255            let segments = parent.segments().to_vec();
256            let guard = ParentGuard::new(parent);
257            let child = Bio::new(
258                BioType::Read,
259                start_sid,
260                segments,
261                Some(Box::new(move |status| {
262                    guard.complete(status);
263                })),
264            );
265            // We can use a throw-away IoBatch because all handling is in the callback.
266            let _ = child.submit(&*member, &mut IoBatch::new());
267        }
268    }
269
270    /// Processes write requests asynchronously by fanning out to all mirrors.
271    ///
272    /// Each child BIO carries a callback that atomically decrements a shared
273    /// counter. The last callback to fire (or the dispatch thread on submission
274    /// failure) completes the parent. Any failed member marks the write as
275    /// `IoError`; all members must succeed for `Complete` to be reported.
276    fn process_write_async(&self, request: BioRequest) {
277        use core::sync::atomic::{AtomicBool, AtomicUsize};
278
279        use ostd::sync::{LocalIrqDisabled, SpinLock};
280
281        for parent in request.into_bios() {
282            let n = self.members.len();
283            let remaining = Arc::new(AtomicUsize::new(n));
284            let had_error = Arc::new(AtomicBool::new(false));
285
286            // Extract before moving parent into the guard.
287            let start_sid = parent.sid_range().start;
288            let segments = parent.segments().to_vec();
289            let guard = Arc::new(SpinLock::<_, LocalIrqDisabled>::new(Some(
290                ParentGuard::new(parent),
291            )));
292
293            for member in &self.members {
294                let member = member.clone();
295
296                let child = Bio::new(BioType::Write, start_sid, segments.clone(), {
297                    let remaining = remaining.clone();
298                    let had_error = had_error.clone();
299                    let guard = guard.clone();
300                    Some(Box::new(move |status| {
301                        if status != BioStatus::Complete {
302                            had_error.store(true, Ordering::Release);
303                        }
304                        // If that's the last device completing the Bio
305                        if remaining.fetch_sub(1, Ordering::AcqRel) == 1 {
306                            // If any device every set an error
307                            let status = if had_error.load(Ordering::Acquire) {
308                                BioStatus::IoError
309                            } else {
310                                BioStatus::Complete
311                            };
312                            if let Some(g) = guard.lock().take() {
313                                g.complete(status);
314                            }
315                        }
316                    }))
317                });
318
319                // On submission failure the callback never fires (see `Bio::submit`),
320                // so the dispatch thread accounts for this member here using the
321                // loop-level `remaining`/`had_error`/`guard`.
322                if member.submit(child, &mut IoBatch::new()).is_err() {
323                    had_error.store(true, Ordering::Release);
324                    if remaining.fetch_sub(1, Ordering::AcqRel) == 1
325                        && let Some(g) = guard.lock().take()
326                    {
327                        g.complete(BioStatus::IoError);
328                    }
329                }
330            }
331        }
332    }
333
334    /// Propagates a flush to all members and completes after they finish.
335    fn process_flush(&self, request: BioRequest) {
336        for parent in request.into_bios() {
337            let status = self.fanout_to_members(&parent, BioType::Flush, Vec::new);
338            parent.complete(status);
339        }
340    }
341
342    /// Submits the same BIO to all members and aggregates completion.
343    ///
344    /// Returns `Complete` only if every child completes; otherwise
345    /// returns a conservative `IoError`.
346    fn fanout_to_members<F>(
347        &self,
348        parent: &SubmittedBio,
349        bio_type: BioType,
350        mut segments_builder: F,
351    ) -> BioStatus
352    where
353        F: FnMut() -> Vec<BioSegment>,
354    {
355        let mut io_batch = IoBatch::new();
356        let mut submission_failed = false;
357
358        for member in &self.members {
359            // Build a child BIO for this member.
360            let child = Bio::new(bio_type, parent.sid_range().start, segments_builder(), None);
361            if member.submit(child, &mut io_batch).is_err() {
362                submission_failed = true
363            }
364        }
365
366        let mut aggregated_status = if submission_failed {
367            // set error if any of the child requests fails to submit.
368            Some(BioStatus::IoError)
369        } else {
370            None
371        };
372
373        if io_batch.wait_all().is_err() {
374            aggregated_status = Some(BioStatus::IoError);
375        }
376
377        // Default to success if no errors were observed.
378        aggregated_status.unwrap_or(BioStatus::Complete)
379    }

381    /// Computes minimal metadata across members (capacity and segment limit).
382    fn min_metadata(members: &[Arc<dyn BlockDevice>]) -> BlockDeviceMeta {
383        let mut iter = members.iter();
384        let first = iter
385            .next()
386            .expect("Raid1Device requires at least one member");
387        let mut meta = first.metadata();
388
389        for member in iter {
390            let member_meta = member.metadata();
391            // Total number of addressable sectors is bounded by the smallest member.
392            meta.nr_sectors = cmp::min(meta.nr_sectors, member_meta.nr_sectors);
393            // Segment limit per BIO is also bounded by the smallest member.
394            meta.max_nr_segments_per_bio = cmp::min(
395                meta.max_nr_segments_per_bio,
396                member_meta.max_nr_segments_per_bio,
397            );
398        }
399
400        meta
401    }

403    /// Clones segments from a parent BIO for use by a child BIO.
404    #[expect(dead_code)]
405    fn clone_segments(parent: &SubmittedBio) -> Vec<BioSegment> {
406        parent.segments().to_vec()
407    }

409    /// Updates an aggregated status with a candidate error using priority.
410    #[expect(dead_code)]
411    fn update_status(current: &mut Option<BioStatus>, candidate: BioStatus) {
412        if candidate == BioStatus::Complete {
413            return;
414        }
415
416        match current {
417            Some(existing) => {
418                if candidate.priority() > (*existing).priority() {
419                    *current = Some(candidate);
420                }
421            }
422            None => *current = Some(candidate),
423        }
424    }
425
426    /// Checks whether a sector range fits within the logical device capacity.
427    fn range_within_capacity(&self, range: &Range<Sid>) -> bool {
428        range.end.to_raw() <= self.metadata.nr_sectors as u64
429    }

432impl BlockDevice for Raid1Device {
433    /// Enqueues a BIO to the RAID device’s admission queue.
434    fn enqueue(&self, bio: SubmittedBio) -> Result<(), BioEnqueueError> {
435        // Reject BIOs that exceed the logical capacity.
436        if !self.range_within_capacity(bio.sid_range()) {
437            return Err(BioEnqueueError::Refused);
438        }
439        // Otherwise, enqueue for processing.
440        self.queue.enqueue(bio)
441    }
