31    pub(super) fn drm_set_client_cap(&self, cmd: ioctl_defs::SetClientCap) -> Result<i32> {
32        /// DRM client capabilities accepted by `DRM_IOCTL_SET_CLIENT_CAP`.
33        ///
34        /// Reference: <https://elixir.bootlin.com/linux/v6.17/source/include/uapi/drm/drm.h#L791>.
35        #[repr(u64)]
36        #[derive(Debug, TryFromInt)]
37        enum DrmSetCapability {
38            Stereo3D = 0x1,
39            UniversalPlane = 0x2,
40            Atomic = 0x3,
41            AspectRatio = 0x4,
42            WritebackConnectors = 0x5,
43            CursorPlaneHotspot = 0x6,
44        }
45
46        let args: DrmSetClientCap = cmd.read()?;
47
48        let Ok(cap) = DrmSetCapability::try_from(args.capability) else {
49            return_errno_with_message!(Errno::EINVAL, "the DRM client capability is unknown");
50        };
51
52        match cap {
53            DrmSetCapability::Stereo3D => self.set_client_caps(
54                DrmClientCaps::STEREO_3D,
55                parse_boolean_capability(args.value)?,
56            ),
57            DrmSetCapability::UniversalPlane => self.set_client_caps(
58                DrmClientCaps::UNIVERSAL_PLANES,
59                parse_boolean_capability(args.value)?,
60            ),
61            DrmSetCapability::Atomic => {
62                // TODO: Support atomic modesetting.
63                return_errno_with_message!(
64                    Errno::EOPNOTSUPP,
65                    "the DRM device lacks atomic modesetting"
66                );
67            }
68            DrmSetCapability::AspectRatio => self.set_client_caps(
69                DrmClientCaps::ASPECT_RATIO,
70                parse_boolean_capability(args.value)?,
71            ),
72            DrmSetCapability::WritebackConnectors => {
73                if !self.has_client_caps(DrmClientCaps::ATOMIC) {
74                    return_errno_with_message!(
75                        Errno::EINVAL,
76                        "the atomic DRM client capability must be enabled before writeback connectors"
77                    );
78                }
79
80                self.set_client_caps(
81                    DrmClientCaps::WRITEBACK_CONNECTORS,
82                    parse_boolean_capability(args.value)?,
83                );
84            }
85            DrmSetCapability::CursorPlaneHotspot => {
86                if !self.device().has_features(DrmFeatures::CURSOR_HOTSPOT) {
87                    return_errno_with_message!(
88                        Errno::EOPNOTSUPP,
89                        "the DRM device lacks cursor hotspot support"
90                    );
91                }
92
93                if !self.has_client_caps(DrmClientCaps::ATOMIC) {
94                    return_errno_with_message!(
95                        Errno::EINVAL,
96                        "the atomic DRM client capability must be enabled before cursor hotspots"
97                    );
98                }
99
100                self.set_client_caps(
101                    DrmClientCaps::CURSOR_PLANE_HOTSPOT,
102                    parse_boolean_capability(args.value)?,
103                );
104            }
105        }
106        Ok(0)
107    }

109    pub(super) fn drm_mode_get_resources(&self, cmd: ioctl_defs::ModeGetResources) -> Result<i32> {
110        cmd.with_data_ptr(|args_ptr| {
111            let mut args: DrmModeGetResources = args_ptr.read()?;
112            let mode_config = self.device().as_kms_ops().unwrap().mode_config();
113
114            let object_store = mode_config.object_store().lock();
115
116            let crtc_ids = object_store.collect_sorted_crtc_ids();
117            let encoder_ids = object_store.collect_sorted_encoder_ids();
118            let connector_ids = object_store.collect_sorted_connector_ids();
119
120            // All store data needed below has been collected into local values.
121            // Release the mutex before copying to userspace, which may fault and
122            // does not require access to the object store.
123            drop(object_store);
124
125            let framebuffer_ids = self.framebuffer_ids();
126
127            copy_array_to_user(args_ptr.vm(), args.crtc_id_ptr, args.count_crtcs, &crtc_ids)?;
128            copy_array_to_user(
129                args_ptr.vm(),
130                args.encoder_id_ptr,
131                args.count_encoders,
132                &encoder_ids,
133            )?;
134            copy_array_to_user(
135                args_ptr.vm(),
136                args.connector_id_ptr,
137                args.count_connectors,
138                &connector_ids,
139            )?;
140            copy_array_to_user(
141                args_ptr.vm(),
142                args.fb_id_ptr,
143                args.count_fbs,
144                &framebuffer_ids,
145            )?;
146
147            args.count_crtcs = crtc_ids.len() as u32;
148            args.count_encoders = encoder_ids.len() as u32;
149            args.count_connectors = connector_ids.len() as u32;
150            args.count_fbs = framebuffer_ids.len() as u32;
151
152            let min_size = mode_config.min_fb_size();
153            let max_size = mode_config.max_fb_size();
154            args.min_width = min_size.width();
155            args.max_width = max_size.width();
156            args.min_height = min_size.height();
157            args.max_height = max_size.height();
158
159            args_ptr.write(&args)?;
160
161            Ok(())
162        })?;
163
164        Ok(0)
165    }

167    pub(super) fn drm_mode_get_crtc(&self, cmd: ioctl_defs::ModeGetCrtc) -> Result<i32> {
168        let mut args: DrmModeCrtc = cmd.read()?;
169        let mode_config = self.device().as_kms_ops().unwrap().mode_config();
170        let object_store = mode_config.object_store().lock();
171
172        let crtc = object_store
173            .lookup_crtc_by_id(args.crtc_id)
174            .ok_or(Errno::ENOENT)?;
175
176        args.gamma_size = crtc.gamma_size();
177        let mode = {
178            let crtc_state = crtc.state().lock();
179            crtc_state
180                .is_enabled()
181                .then(|| crtc_state.display_mode())
182                .flatten()
183        };
184
185        let primary_plane = crtc.primary_plane().upgrade().ok_or(Errno::ENOENT)?;
186        let plane_state = primary_plane.state().lock();
187        // TODO: Convert source coordinates from 16.16 fixed-point once supported.
188        args.x = plane_state.source_rect().x();
189        args.y = plane_state.source_rect().y();
190        args.fb_id = plane_state
191            .framebuffer()
192            .upgrade()
193            .map_or(0, |framebuffer| framebuffer.id());
194        drop(plane_state);
195
196        if let Some(mode) = mode {
197            args.mode = DrmModeInfo::try_from(mode)?;
198            args.mode_valid = 1;
199        } else {
200            args.mode_valid = 0;
201        }
202
203        drop(object_store);
204
205        cmd.write(&args)?;
206        Ok(0)
207    }

229    pub(super) fn drm_mode_get_connector(&self, cmd: ioctl_defs::ModeGetConnector) -> Result<i32> {
230        cmd.with_data_ptr(|args_ptr| {
231            let mut args: DrmModeGetConnector = args_ptr.read()?;
232
233            // Linux treats `GETCONNECTOR` with `count_modes == 0` as a forced probe request.
234            // Only the DRM master is allowed to refresh the connector state in this path;
235            // non-master callers fall back to a read-only query.
236            if args.count_modes == 0 && self.is_master() {
237                let kms_ops = self.device().as_kms_ops().unwrap();
238                // Keep the connector alive without holding the object-store mutex
239                // across the driver callback, which may perform I/O or acquire other locks.
240                let connector = {
241                    let object_store = kms_ops.mode_config().object_store().lock();
242                    object_store
243                        .lookup_connector_by_id(args.connector_id)
244                        .cloned()
245                        .ok_or(Errno::ENOENT)?
246                };
247                let probe_state = kms_ops.probe_connector(&connector)?;
248                connector.update_probe_state(probe_state);
249            }
250
251            let mode_config = self.device().as_kms_ops().unwrap().mode_config();
252            let object_store = mode_config.object_store().lock();
253
254            let connector = object_store
255                .lookup_connector_by_id(args.connector_id)
256                .ok_or(Errno::ENOENT)?;
257            args.connector_type = connector.type_() as u32;
258            args.connector_type_id = connector.type_index();
259            let encoder_id = {
260                let state = connector.state().lock();
261                state.encoder().upgrade().map_or(0, |encoder| encoder.id())
262            };
263            args.encoder_id = encoder_id;
264
265            let possible_encoders = connector.possible_encoders();
266            let mut encoder_ids = Vec::with_capacity(possible_encoders.count_ones());
267            for index in possible_encoders.iter_ones() {
268                let encoder_index = KmsObjectIndex::new(index);
269                let encoder_id = object_store
270                    .lookup_encoder_by_index(encoder_index)
271                    .map(|encoder| encoder.id())
272                    .ok_or(Errno::ENOENT)?;
273
274                encoder_ids.push(encoder_id);
275            }
276
277            let (prop_ids, prop_values) = self.visible_property_values(connector.properties());
278
279            let mode_info = {
280                let probe_state = connector.probe_state().lock();
281                let display_info = probe_state.display_info();
282                args.connection = probe_state.status() as u32;
283                args.mm_width = display_info.mm_width();
284                args.mm_height = display_info.mm_height();
285                args.subpixel = display_info.subpixel_order();
286
287                probe_state
288                    .display_modes()
289                    .iter()
290                    .copied()
291                    .map(DrmModeInfo::try_from)
292                    .collect::<Result<Vec<_>>>()?
293            };
294
295            drop(object_store);
296
297            copy_array_to_user(args_ptr.vm(), args.modes_ptr, args.count_modes, &mode_info)?;
298            copy_array_to_user(args_ptr.vm(), args.props_ptr, args.count_props, &prop_ids)?;
299            copy_array_to_user(
300                args_ptr.vm(),
301                args.prop_values_ptr,
302                args.count_props,
303                &prop_values,
304            )?;
305            copy_array_to_user(
306                args_ptr.vm(),
307                args.encoders_ptr,
308                args.count_encoders,
309                &encoder_ids,
310            )?;
311
312            args.count_encoders = encoder_ids.len() as u32;
313            args.count_modes = mode_info.len() as u32;
314            args.count_props = prop_ids.len() as u32;
315
316            args_ptr.write(&args)?;
317            Ok(())
318        })?;
319
320        Ok(0)
321    }

323    pub(super) fn drm_mode_get_property(&self, cmd: ioctl_defs::ModeGetProperty) -> Result<i32> {
324        let mut args: DrmModeGetProperty = cmd.read()?;
325
326        cmd.with_data_ptr(|args_ptr| {
327            let mode_config = self.device().as_kms_ops().unwrap().mode_config();
328            let object_store = mode_config.object_store().lock();
329
330            let property = object_store
331                .lookup_property(args.prop_id)
332                .ok_or(Errno::ENOENT)?;
333
334            let values = match property.value_type() {
335                DrmPropertyValueType::Range { min, max } => vec![*min, *max],
336                DrmPropertyValueType::SignedRange { min, max } => {
337                    vec![*min as u64, *max as u64]
338                }
339                DrmPropertyValueType::Enum(entries) | DrmPropertyValueType::Bitmask(entries) => {
340                    entries.iter().map(|entry| entry.value()).collect()
341                }
342                DrmPropertyValueType::Object(object_type) => vec![*object_type as u64],
343                DrmPropertyValueType::Plain | DrmPropertyValueType::Blob => vec![],
344            };
345            let enum_entries = match property.value_type() {
346                DrmPropertyValueType::Enum(entries) | DrmPropertyValueType::Bitmask(entries) => {
347                    entries.to_vec()
348                }
349                _ => Vec::new(),
350            };
351
352            args.name = *property.name();
353            args.flags = property.flags().bits();
354            drop(object_store);
355
356            copy_array_to_user(args_ptr.vm(), args.values_ptr, args.count_values, &values)?;
357            copy_array_to_user(
358                args_ptr.vm(),
359                args.enum_blob_ptr,
360                args.count_enum_blobs,
361                &enum_entries,
362            )?;
363
364            args.count_values = values.len() as u32;
365            args.count_enum_blobs = enum_entries.len() as u32;
366
367            args_ptr.write(&args)?;
368            Ok(())
369        })?;
370
371        Ok(0)
372    }

374    pub(super) fn drm_mode_get_blob(&self, cmd: ioctl_defs::ModeGetPropBlob) -> Result<i32> {
375        let mut args: DrmModeGetBlob = cmd.read()?;
376
377        cmd.with_data_ptr(|args_ptr| {
378            let blob = {
379                let mode_config = self.device().as_kms_ops().unwrap().mode_config();
380                let object_store = mode_config.object_store().lock();
381                object_store
382                    .lookup_property_blob(args.blob_id)
383                    .ok_or(Errno::ENOENT)?
384                    .clone()
385            };
386
387            let data = blob.data();
388            if args.length != 0 && !data.is_empty() {
389                if args.data == 0 {
390                    return_errno_with_message!(
391                        Errno::EFAULT,
392                        "the DRM blob output pointer is null"
393                    );
394                }
395
396                let write_len = core::cmp::min(args.length as usize, data.len());
397                args_ptr
398                    .vm()
399                    .write_bytes(args.data as usize, &data[..write_len])?;
400            }
401
402            args.length = blob.len() as u32;
403            args_ptr.write(&args)?;
404            Ok(())
405        })?;
406
407        Ok(0)
408    }

410    pub(super) fn drm_mode_add_fb(&self, cmd: ioctl_defs::ModeAddFb) -> Result<i32> {
411        let mut args: DrmModeFbCmd = cmd.read()?;
412
413        let pixel_format = DrmPixelFormat::try_from((args.bpp, args.depth))?;
414
415        let mode_config = self.device().as_kms_ops().unwrap().mode_config();
416        let min_size = mode_config.min_fb_size();
417        let max_size = mode_config.max_fb_size();
418        let size = DrmSize::new(args.width, args.height);
419
420        if !size.is_within(
421            min_size.width()..=max_size.width(),
422            min_size.height()..=max_size.height(),
423        ) {
424            return_errno_with_message!(
425                Errno::EINVAL,
426                "the DRM framebuffer dimensions are outside the supported range"
427            );
428        }
429
430        let supports_pixel_format = {
431            let object_store = mode_config.object_store().lock();
432            object_store
433                .planes()
434                .any(|plane| plane.pixel_formats().contains(&pixel_format))
435        };
436        if !supports_pixel_format {
437            return_errno_with_message!(
438                Errno::EINVAL,
439                "no DRM plane supports the framebuffer pixel format"
440            );
441        }
442
443        let gem_object = self.lookup_gem_object(args.handle)?;
444        let framebuffer = {
445            let mut object_store = mode_config.object_store().lock();
446            let fb_id = object_store.alloc_object_id()?;
447            let framebuffer = DrmFramebuffer::new(
448                fb_id,
449                DrmSize::new(args.width, args.height),
450                pixel_format,
451                DrmFramebufferFlags::empty(),
452                args.pitch,
453                0,
454                0,
455                gem_object,
456            );
457
458            match framebuffer {
459                Ok(framebuffer) => Arc::new(framebuffer),
460                Err(error) => {
461                    object_store.free_object_id(fb_id);
462                    return Err(error);
463                }
464            }
465        };
466        let fb_id = framebuffer.id();
467        args.fb_id = fb_id;
468
469        if let Err(error) = cmd.write(&args) {
470            mode_config.object_store().lock().free_object_id(fb_id);
471            return Err(error);
472        }
473        self.add_framebuffer(framebuffer);
474
475        Ok(0)
476    }

478    pub(super) fn drm_mode_rm_fb(&self, cmd: ioctl_defs::ModeRmFb) -> Result<i32> {
479        let fb_id: u32 = cmd.read()?;
480
481        let framebuffer = self.lookup_framebuffer(fb_id).ok_or(Errno::ENOENT)?;
482        let mode_config = self.device().as_kms_ops().unwrap().mode_config();
483        let mut object_store = mode_config.object_store().lock();
484
485        for plane in object_store.planes() {
486            if plane
487                .state()
488                .lock()
489                .framebuffer()
490                .ptr_eq(&Arc::downgrade(&framebuffer))
491            {
492                return_errno_with_message!(
493                    Errno::EBUSY,
494                    "the DRM framebuffer is still in use by a plane"
495                );
496            }
497        }
498
499        self.remove_framebuffer(fb_id).ok_or(Errno::ENOENT)?;
500        object_store.free_object_id(fb_id);
501
502        Ok(0)
503    }

525            };
526
527            copy_array_to_user(
528                args_ptr.vm(),
529                args.plane_id_ptr,
530                args.count_planes,
531                &plane_ids,
532            )?;
533
534            args.count_planes = plane_ids.len() as u32;
535
536            args_ptr.write(&args)?;
537            Ok(())
538        })?;
539
540        Ok(0)

588    pub(super) fn drm_mode_object_get_props(
589        &self,
590        cmd: ioctl_defs::ModeObjectGetProps,
591    ) -> Result<i32> {
592        let mut args: DrmModeObjectGetProps = cmd.read()?;
593
594        cmd.with_data_ptr(|args_ptr| {
595            let mode_config = self.device().as_kms_ops().unwrap().mode_config();
596            let object_store = mode_config.object_store().lock();
597
598            let object_type =
599                DrmKmsObjectType::try_from(args.obj_type).map_err(|_| Errno::ENOENT)?;
600
601            let properties = match object_type {
602                DrmKmsObjectType::Plane => object_store
603                    .lookup_plane_by_id(args.obj_id)
604                    .map(|plane| plane.properties()),
605                DrmKmsObjectType::Crtc => object_store
606                    .lookup_crtc_by_id(args.obj_id)
607                    .map(|crtc| crtc.properties()),
608                DrmKmsObjectType::Connector => object_store
609                    .lookup_connector_by_id(args.obj_id)
610                    .map(|connector| connector.properties()),
611                DrmKmsObjectType::Encoder => {
612                    if object_store.lookup_encoder_by_id(args.obj_id).is_some() {
613                        return_errno_with_message!(
614                            Errno::EINVAL,
615                            "the DRM encoder has no property table"
616                        );
617                    }
618                    None
619                }
620                DrmKmsObjectType::Any => object_store.lookup_any_object_properties(args.obj_id)?,
621                _ => None,
622            }
623            .ok_or_else(|| {
624                Error::with_message(
625                    Errno::ENOENT,
626                    "the DRM object type does not match the object ID",
627                )
628            })?;
629
630            let (prop_ids, prop_values) = self.visible_property_values(properties);
631            drop(object_store);
632
633            copy_array_to_user(args_ptr.vm(), args.props_ptr, args.count_props, &prop_ids)?;
634            copy_array_to_user(
635                args_ptr.vm(),
636                args.prop_values_ptr,
637                args.count_props,
638                &prop_values,
639            )?;
640
641            args.count_props = prop_ids.len() as u32;
642            args_ptr.write(&args)?;
643            Ok(())
644        })?;
645
646        Ok(0)
647    }

649    fn visible_property_values(&self, properties: &DrmPropertyAttachments) -> (Vec<u32>, Vec<u64>) {
650        let mut prop_ids = Vec::new();
651        let mut prop_values = Vec::new();
652
653        for attachment in properties.attachments() {
654            let property = attachment.property();
655            let id = property.id();
656            let value = attachment.value();
657
658            if property.flags().contains(DrmPropertyFlags::ATOMIC)
659                && !self.has_client_caps(DrmClientCaps::ATOMIC)
660            {
661                continue;
662            }
663
664            prop_ids.push(id);
665            prop_values.push(value);
666        }
667
668        (prop_ids, prop_values)
669    }
670}
671
672fn parse_boolean_capability(value: u64) -> Result<bool> {
673    match value {
674        0 => Ok(false),
675        1 => Ok(true),
676        _ => return_errno_with_message!(
677            Errno::EINVAL,
678            "a boolean DRM client capability must be zero or one"
679        ),
680    }
681}
682
683fn copy_array_to_user<T: Pod>(
684    userspace: &impl VmIo,
685    user_ptr: u64,
686    user_capacity: u32,
687    values: &[T],
688) -> Result<()> {
689    let copy_count = core::cmp::min(user_capacity as usize, values.len());
690    if copy_count == 0 {
691        return Ok(());
692    }
693    if user_ptr == 0 {
694        return_errno_with_message!(Errno::EFAULT, "the DRM userspace array pointer is null");
695    }
696
697    userspace.write_slice(user_ptr as usize, &values[..copy_count])?;
698    Ok(())
699}
