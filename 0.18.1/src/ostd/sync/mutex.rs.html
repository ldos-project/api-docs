13/// A mutex with waitqueue.
14pub struct Mutex<T: ?Sized> {
15    lock: AtomicBool,
16    queue: WaitQueue,
17    /// The captured context information from acquire. This lock should only be held to copy the
18    /// bytes in an out.
19    #[cfg(feature = "track_mutex")]
20    acquire_info: Option<super::SpinLock<StackInfo>>,
21    val: UnsafeCell<T>,
22}
23

28impl<T> Mutex<T> {
29    /// Creates a new mutex.
30    pub const fn new(val: T) -> Self {
31        Self {
32            lock: AtomicBool::new(false),
33            queue: WaitQueue::new(),
34            #[cfg(feature = "track_mutex")]
35            acquire_info: None,
36            val: UnsafeCell::new(val),
37        }
38    }
39
40    /// Set mutex to capture acquire information.
41    ///
42    /// This captures the call context (stack, thread, etc) when the lock is acquired and prints it
43    /// when a thread blocks on the lock.
44    ///
45    /// This is a no-op if the library is built without the `track_mutex` feature.
46    ///
47    /// ## Usage
48    ///
49    /// When this is enabled, any time a thread fails to acquire the lock it will log (at the `info` level):
50    ///
51    /// ```text
52    /// INFO: Failed to acquire lock:
53    /// Held by: [information about holder]
54    /// Failed at: [information about failing acquire]
55    /// ```
56    /// (The information is provided by [`StackInfo`].)
57    ///
58    /// The result can be processed with `cargo osdk enhance-log`.
59    pub const fn with_acquire_info(self, capture_acquire_info: bool) -> Self {
60        self.maybe_with_acquire_info(capture_acquire_info)
61    }
62
63    // The private method is used to simplify the `cfg`s.
64
65    #[cfg(feature = "track_mutex")]
66    const fn maybe_with_acquire_info(mut self, capture_acquire_info: bool) -> Self {
67        if capture_acquire_info {
68            self.acquire_info = Some(super::SpinLock::new(StackInfo::empty()));
69        } else {
70            self.acquire_info = None;
71        }
72        self
73    }
74
75    #[cfg(not(feature = "track_mutex"))]
76    const fn maybe_with_acquire_info(self, _capture_acquire_info: bool) -> Self {
77        self
78    }
79}
80
81impl<T: Default> Default for Mutex<T> {
82    fn default() -> Self {
83        Self::new(Default::default())
84    }
85}

87impl<T: ?Sized> Mutex<T> {
88    #[cfg(feature = "track_mutex")]
89    #[track_caller]
90    fn report_acquire_failure(&self) {
91        use crate::early_println;
92
93        if let Some(acquire_info) = &self.acquire_info {
94            let info = *acquire_info.lock();
95            early_println!(
96                "Blocking on Mutex:\nHeld by: {}\nFailed to acquire at: {}",
97                info,
98                StackInfo::new(2)
99            );
100        }
101    }
102
103    /// Acquires the mutex.
104    ///
105    /// This method runs in a block way until the mutex can be acquired.
106    #[track_caller]
107    pub fn lock(&self) -> MutexGuard<'_, T> {
108        #[cfg(feature = "track_mutex")]
109        if let Some(r) = self.try_lock() {
110            return r;
111        } else {
112            self.report_acquire_failure();
113        }
114        self.queue.wait_until(|| self.try_lock())
115    }
116
117    /// Tries to acquire the mutex immediately.
118    pub fn try_lock(&self) -> Option<MutexGuard<'_, T>> {
119        self.acquire_lock()
120            // We cannot use `then_some`. Otherwise, the possible dropping of the temporary
121            // guard will cause an unexpected unlock.
122            // SAFETY: The lock is successfully acquired when creating the guard.
123            .then(|| unsafe { MutexGuard::new(self) })
124    }
125
126    /// Returns a mutable reference to the underlying data.
127    ///
128    /// This method is zero-cost: By holding a mutable reference to the lock, the compiler has
129    /// already statically guaranteed that access to the data is exclusive.
130    pub fn get_mut(&mut self) -> &mut T {
131        self.val.get_mut()
132    }
133
134    /// Releases the mutex and wake up one thread which is blocked on this mutex.
135    ///
136    /// # Safety
137    ///
138    /// The caller must ensure that it has acquired the lock.
139    unsafe fn unlock(&self) {
140        // SAFETY: The safety is upheld by the caller.
141        unsafe { self.release_lock() };
142        self.queue.wake_one();
143    }
144
145    #[track_caller]
146    fn acquire_lock(&self) -> bool {
147        let succeeded = self
148            .lock
149            .compare_exchange(false, true, Ordering::Acquire, Ordering::Relaxed)
150            .is_ok();
151        #[cfg(feature = "track_mutex")]
152        if succeeded && let Some(acquire_info) = &self.acquire_info {
153            let info = StackInfo::new(2);
154            *acquire_info.lock() = info;
155        }
156        succeeded
157    }
158
159    /// # Safety
160    ///
161    /// The caller must ensure that it has acquired the lock.
162    unsafe fn release_lock(&self) {
163        self.lock.store(false, Ordering::Release);
164    }
165
166    /// Get information about the last point in the program this mutex was acquired.
167    pub fn last_acquire_info(&self) -> Option<StackInfo> {
168        #[cfg(feature = "track_mutex")]
169        return self.acquire_info.as_ref().map(|i| *i.lock());
170        #[cfg(not(feature = "track_mutex"))]
171        return None;
172    }
173}

185/// A guard that provides exclusive access to the data protected by a [`Mutex`].
186#[clippy::has_significant_drop]
187#[must_use]
188pub struct MutexGuard<'a, T: ?Sized> {
189    mutex: &'a Mutex<T>,
190}
191
192impl<'a, T: ?Sized> MutexGuard<'a, T> {
193    /// # Safety
194    ///
195    /// The caller must ensure that the given reference of [`Mutex`] lock has been successfully
196    /// acquired in the current context. When the created [`MutexGuard`] is dropped, it will unlock
197    /// the [`Mutex`].
198    unsafe fn new(mutex: &'a Mutex<T>) -> Self {
199        MutexGuard { mutex }
200    }
201
202    /// Returns the [`Mutex`] associated with this guard.
203    pub fn get_lock(guard: &Self) -> &'a Mutex<T> {
204        guard.mutex
205    }
206}
207
208impl<T: ?Sized> Deref for MutexGuard<'_, T> {
209    type Target = T;
210
211    fn deref(&self) -> &Self::Target {
212        // SAFETY: The mutex is locked, which provides exclusive access to the data.
213        unsafe { &*self.mutex.val.get() }
214    }
215}
216
217impl<T: ?Sized> DerefMut for MutexGuard<'_, T> {
218    fn deref_mut(&mut self) -> &mut Self::Target {
219        // SAFETY: The mutex is locked, which provides exclusive access to the data.
220        unsafe { &mut *self.mutex.val.get() }
221    }
222}
223
224impl<T: ?Sized> Drop for MutexGuard<'_, T> {
225    fn drop(&mut self) {
226        // SAFETY: The mutex is locked.
227        unsafe { self.mutex.unlock() };
228    }
229}

231impl<T: ?Sized + fmt::Debug> fmt::Debug for MutexGuard<'_, T> {
232    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
233        fmt::Debug::fmt(&**self, f)
234    }
235}

253    // A regression test for a bug fixed in [#1279](https://github.com/asterinas/asterinas/pull/1279).
254    #[ktest]
255    fn try_lock_does_not_unlock() {
256        let lock = Mutex::new(0);
257        assert!(!lock.lock.load(Ordering::Relaxed));
258
259        // A successful lock
260        let guard1 = lock.lock();
261        assert!(lock.lock.load(Ordering::Relaxed));
262
263        // A failed `try_lock` won't drop the lock
264        assert!(lock.try_lock().is_none());
265        assert!(lock.lock.load(Ordering::Relaxed));
266
267        // Ensure the lock is held until here
268        drop(guard1);
269    }

271    #[ktest]
272    fn mutex_tracking_threaded() {
273        let mutex = Arc::new(Mutex::new(0).with_acquire_info(true));
274
275        let guard = mutex.lock();
276
277        let _handle = TaskOptions::new({
278            let mutex = mutex.clone();
279            move || {
280                let mut v = mutex.lock();
281                *v = 1;
282            }
283        })
284        .spawn()
285        .unwrap();
286
287        sleep(Duration::from_millis(1000));
288
289        drop(guard);
290
291        sleep_with_predicate(Duration::from_millis(1000), || *mutex.lock() == 1);
292
293        info!(
294            "The test should have printed:
295Failed to acquire lock:
296Held by: [stack info from main thread]
297Failed at: [stack info from spawned task]"
298        );
299    }
