4//!
5//! The [`VmSpace`] struct is provided to manage the virtual memory space of a
6//! user. Cursors are used to traverse and modify over the virtual memory space
7//! concurrently. The VM space cursor [`self::Cursor`] is just a wrapper over
8//! the page table cursor, providing efficient, powerful concurrent accesses
9//! to the page table.

42/// VmMappingPolicy controls the level of pages that are mapped in response to a page fault. If the
43/// value returned by [`VmMappingPolicy::get_page_level`] is 1, a base page is mapped, and for
44/// larger level the corresponding level page is mapped.
45#[orpc_trait]
46pub trait VmMappingPolicy: Sync + Send {
47    /// Get the level of page that should be mapped in request to a fault at
48    /// `req.page_aligned_addr`.
49    fn get_page_level(&self, req: &VmMappingRequest) -> Result<PagingLevel, RPCError>;
50}
51
52/// Default [`VmMappingPolicy`] implementation that always maps base pages.
53#[orpc_server(VmMappingPolicy)]
54struct VmMappingPolicyBasePagesOnly {}
55
56#[orpc_impl]
57impl VmMappingPolicy for VmMappingPolicyBasePagesOnly {
58    fn get_page_level(&self, _req: &VmMappingRequest) -> Result<PagingLevel, RPCError> {
59        Ok(1)
60    }
61}

63/// A virtual address space for user-mode tasks, enabling safe manipulation of user-space memory.
64///
65/// The `VmSpace` type provides memory isolation guarantees between user-space and
66/// kernel-space. For example, given an arbitrary user-space pointer, one can read and
67/// write the memory location referred to by the user-space pointer without the risk of
68/// breaking the memory safety of the kernel space.
69///
70/// # Task Association Semantics
71///
72/// As far as OSTD is concerned, a `VmSpace` is not necessarily associated with a task. Once a
73/// `VmSpace` is activated (see [`VmSpace::activate`]), it remains activated until another
74/// `VmSpace` is activated **possibly by another task running on the same CPU**.
75///
76/// This means that it's up to the kernel to ensure that a task's `VmSpace` is always activated
77/// while the task is running. This can be done by using the injected post schedule handler
78/// (see [`inject_post_schedule_handler`]) to always activate the correct `VmSpace` after each
79/// context switch.
80///
81/// If the kernel otherwise decides not to ensure that the running task's `VmSpace` is always
82/// activated, the kernel must deal with race conditions when calling methods that require the
83/// `VmSpace` to be activated, e.g., [`UserMode::execute`], [`VmSpace::reader`],
84/// [`VmSpace::writer`]. Otherwise, the behavior is unspecified, though it's guaranteed _not_ to
85/// compromise the kernel's memory safety.
86///
87/// # Memory Backing
88///
89/// A newly-created `VmSpace` is not backed by any physical memory pages. To
90/// provide memory pages for a `VmSpace`, one can allocate and map physical
91/// memory ([`UFrame`]s) to the `VmSpace` using the cursor.
92///
93/// A `VmSpace` can also attach a page fault handler, which will be invoked to
94/// handle page faults generated from user space.
95///
96/// [`inject_post_schedule_handler`]: crate::task::inject_post_schedule_handler
97/// [`UserMode::execute`]: crate::user::UserMode::execute
98pub struct VmSpace {
99    pt: PageTable<UserPtConfig>,
100    cpus: AtomicCpuSet,
101    iomems: SpinLock<Vec<IoMem>>,
102    vm_mapping_policy: Arc<dyn VmMappingPolicy>,
103}

209    /// Activates the page table on the current CPU.
210    pub fn activate(self: &Arc<Self>) {
211        let preempt_guard = disable_preempt();
212        let cpu = preempt_guard.current_cpu();
213
214        let last_ptr = ACTIVATED_VM_SPACE.load();
215
216        if last_ptr == Arc::as_ptr(self) {
217            return;
218        }
219
220        // Record ourselves in the CPU set and the activated VM space pointer.
221        // `Acquire` to ensure the modification to the PT is visible by this CPU.
222        self.cpus.add(cpu, Ordering::Acquire);
223
224        let self_ptr = Arc::into_raw(Arc::clone(self)) as *mut VmSpace;
225        ACTIVATED_VM_SPACE.store(self_ptr);
226
227        if !last_ptr.is_null() {
228            // SAFETY: The pointer is cast from an `Arc` when it's activated
229            // the last time, so it can be restored and only restored once.
230            let last = unsafe { Arc::from_raw(last_ptr) };
231            last.cpus.remove(cpu, Ordering::Relaxed);
232        }
233
234        self.pt.activate();
235    }

237    /// Creates a reader to read data from the user space of the current task.
238    ///
239    /// Returns `Err` if this `VmSpace` doesn't belong to the user space of the current task
240    /// or the `vaddr` and `len` do not represent a user space memory range.
241    ///
242    /// Users must ensure that no other page table is activated in the current task during the
243    /// lifetime of the created `VmReader`. This guarantees that the `VmReader` can operate correctly.
244    pub fn reader(&self, vaddr: Vaddr, len: usize) -> Result<VmReader<'_, Fallible>> {
245        if current_page_table_paddr() != self.pt.root_paddr()
246            || !super::is_in_user_space(vaddr, len)
247        {
248            return AccessDeniedSnafu.fail();
249        }
250
251        // SAFETY: The memory range is in user space, as checked above.
252        Ok(unsafe { VmReader::<Fallible>::from_user_space(vaddr as *const u8, len) })
253    }
254
255    /// Creates a writer to write data into the user space.
256    ///
257    /// Returns `Err` if this `VmSpace` doesn't belong to the user space of the current task
258    /// or the `vaddr` and `len` do not represent a user space memory range.
259    ///
260    /// Users must ensure that no other page table is activated in the current task during the
261    /// lifetime of the created `VmWriter`. This guarantees that the `VmWriter` can operate correctly.
262    pub fn writer(&self, vaddr: Vaddr, len: usize) -> Result<VmWriter<'_, Fallible>> {
263        if current_page_table_paddr() != self.pt.root_paddr()
264            || !super::is_in_user_space(vaddr, len)
265        {
266            return AccessDeniedSnafu.fail();
267        }
268
269        // `VmWriter` is neither `Sync` nor `Send`, so it will not live longer than the current
270        // task. This ensures that the correct page table is activated during the usage period of
271        // the `VmWriter`.
272        //
273        // SAFETY: The memory range is in user space, as checked above.
274        Ok(unsafe { VmWriter::<Fallible>::from_user_space(vaddr as *mut u8, len) })
275    }

282    /// Creates a reader/writer pair to read data from and write data into the user space.
283    ///
284    /// Returns `Err` if this `VmSpace` doesn't belong to the user space of the current task
285    /// or the `vaddr` and `len` do not represent a user space memory range.
286    ///
287    /// Users must ensure that no other page table is activated in the current task during the
288    /// lifetime of the created `VmReader` and `VmWriter`. This guarantees that the `VmReader`
289    /// and the `VmWriter` can operate correctly.
290    ///
291    /// This method is semantically equivalent to calling [`Self::reader`] and [`Self::writer`]
292    /// separately, but it avoids double checking the validity of the memory region.
293    pub fn reader_writer(
294        &self,
295        vaddr: Vaddr,
296        len: usize,
297    ) -> Result<(VmReader<'_, Fallible>, VmWriter<'_, Fallible>)> {
298        if current_page_table_paddr() != self.pt.root_paddr()
299            || !super::is_in_user_space(vaddr, len)
300        {
301            return AccessDeniedSnafu.fail();
302        }
303
304        // SAFETY: The memory range is in user space, as checked above.
305        let reader = unsafe { VmReader::<Fallible>::from_user_space(vaddr as *const u8, len) };
306
307        // `VmWriter` is neither `Sync` nor `Send`, so it will not live longer than the current
308        // task. This ensures that the correct page table is activated during the usage period of
309        // the `VmWriter`.
310        //
311        // SAFETY: The memory range is in user space, as checked above.
312        let writer = unsafe { VmWriter::<Fallible>::from_user_space(vaddr as *mut u8, len) };
313
314        Ok((reader, writer))
315    }

325    /// Finds the [`IoMem`] that contains the given physical address.
326    ///
327    /// It is a private method for internal use only. Please refer to
328    /// [`CursorMut::find_iomem_by_paddr`] for more details.
329    fn find_iomem_by_paddr(&self, paddr: Paddr) -> Option<(IoMem, usize)> {
330        let iomems = self.iomems.lock();
331        for iomem in iomems.iter() {
332            let start = iomem.paddr();
333            let end = start + iomem.size();
334            if paddr >= start && paddr < end {
335                let offset = paddr - start;
336                return Some((iomem.clone(), offset));
337            }
338        }
339        None
340    }
341}

343/// The cursor for querying over the VM space without modifying it.
344///
345/// It exclusively owns a sub-tree of the page table, preventing others from
346/// reading or modifying the same sub-tree. Two read-only cursors can not be
347/// created from the same virtual address range either.
348pub struct Cursor<'a>(page_table::Cursor<'a, UserPtConfig>);
349
350impl Cursor<'_> {
351    /// Queries the mapping at the current virtual address.
352    ///
353    /// If the cursor is pointing to a valid virtual address that is locked,
354    /// it will return the virtual address range and the mapped item.
355    pub fn query(&mut self) -> Result<(Range<Vaddr>, Option<VmQueriedItem<'_>>)> {
356        let (range, item) = self.0.query()?;
357        Ok((range, item.map(VmQueriedItem::from)))
358    }
359
360    /// Moves the cursor forward to the next mapped virtual address.
361    ///
362    /// If there is mapped virtual address following the current address within
363    /// next `len` bytes, it will return that mapped address. In this case,
364    /// the cursor will stop at the mapped address.
365    ///
366    /// Otherwise, it will return `None`. And the cursor may stop at any
367    /// address after `len` bytes.
368    ///
369    /// # Panics
370    ///
371    /// Panics if:
372    ///  - the length is longer than the remaining range of the cursor;
373    ///  - the length is not page-aligned.
374    pub fn find_next(&mut self, len: usize) -> Option<Vaddr> {
375        self.0.find_next(len)
376    }
377
378    /// Jumps to the virtual address.
379    ///
380    /// If the target address is out of the range, this method will return `Err`.
381    ///
382    /// # Panics
383    ///
384    /// This method panics if the address has bad alignment.
385    pub fn jump(&mut self, va: Vaddr) -> Result<()> {
386        self.0.jump(va)?;
387        Ok(())
388    }
389
390    /// Gets the virtual address of the current slot.
391    pub fn virt_addr(&self) -> Vaddr {
392        self.0.virt_addr()
393    }

455    /// Maps a frame into the current slot.
456    ///
457    /// This method will bring the cursor to the next slot after the modification.
458    ///
459    /// # Panics
460    ///
461    /// Panics if:
462    ///  - the current virtual address is already mapped;
463    ///  - the current virtual address is outside the cursor's range.
464    pub fn map(&mut self, frame: UFrame, prop: PageProperty) {
465        let item = VmItem::new_tracked(frame, prop);
466
467        // SAFETY: It is safe to map untyped memory into the userspace.
468        unsafe { self.pt_cursor.map(item) };
469    }

471    /// Maps a range of [`IoMem`] into the current slot.
472    ///
473    /// The memory region to be mapped is the [`IoMem`] range starting at
474    /// `offset` and extending to `offset + len`, or to the end of [`IoMem`],
475    /// whichever comes first. This method will bring the cursor to the next
476    /// slot after the modification.
477    ///
478    /// # Limitations
479    ///
480    /// Once an instance of `IoMem` is mapped to a `VmSpace`,
481    /// then the `IoMem` instance will only be dropped when the `VmSpace` is
482    /// dropped, not when all the mappings backed by the `IoMem` are destroyed
483    /// with the `unmap` method.
484    ///
485    /// # Panics
486    ///
487    /// Panics if
488    ///  - `len`, `offset`, or the address range of the `IoMem` instance is
489    ///    not aligned to the page size;
490    ///  - the current virtual address is already mapped.
491    pub fn map_iomem(&mut self, io_mem: IoMem, prop: PageProperty, len: usize, offset: usize) {
492        assert_eq!(len % PAGE_SIZE, 0);
493        assert_eq!(offset % PAGE_SIZE, 0);
494
495        let io_mem_paddr = io_mem.paddr();
496        let io_mem_size = io_mem.size();
497        assert_eq!(io_mem_paddr % PAGE_SIZE, 0);
498        assert_eq!(io_mem_size % PAGE_SIZE, 0);
499
500        if offset >= io_mem_size {
501            return;
502        }
503
504        let paddr_begin = io_mem_paddr + offset;
505        let paddr_end = if io_mem_size - offset < len {
506            io_mem_paddr + io_mem_size
507        } else {
508            io_mem_paddr + len + offset
509        };
510
511        for current_paddr in (paddr_begin..paddr_end).step_by(PAGE_SIZE) {
512            // SAFETY: It is safe to map I/O memory into the userspace.
513            unsafe {
514                self.pt_cursor
515                    .map(VmItem::new_untracked_io(current_paddr, prop))
516            };
517        }
518
519        fn io_mem_contains(parent: &IoMem, child: &IoMem) -> bool {
520            parent.paddr() <= child.paddr() && child.end_paddr() <= parent.end_paddr()
521        }
522
523        // If the `iomems` list in `VmSpace` does not contain the current I/O
524        // memory, push it to maintain the correct reference count.
525        let mut iomems = self.vmspace.iomems.lock();
526        if !iomems.iter().any(|iomem| io_mem_contains(iomem, &io_mem)) {
527            iomems.retain(|iomem| !io_mem_contains(&io_mem, iomem));
528            iomems.push(io_mem);
529        }
530    }

536    /// [`Self::query`]. If the query returns a [`VmQueriedItem::MappedIoMem`], this method is
537    /// guaranteed to succeed with the specific physical address. However, if the corresponding
538    /// mapping is subsequently unmapped, it is unspecified whether this method will still succeed
539    /// or not.
540    ///
541    /// On success, this method returns the `IoMem` and the offset from the `IoMem` start to the
542    /// given physical address. Otherwise, this method returns `None`.
543    pub fn find_iomem_by_paddr(&self, paddr: Paddr) -> Option<(IoMem, usize)> {
544        self.vmspace.find_iomem_by_paddr(paddr)
545    }

547    /// Clears the mapping starting from the current slot,
548    /// and returns the number of unmapped pages.
549    ///
550    /// This method will bring the cursor forward by at least `len` bytes
551    /// in the virtual address space, but not past the end of the cursor's range.
552    ///
553    /// Already-absent mappings encountered by the cursor will be skipped. It
554    /// is valid to unmap a range that is not mapped.
555    ///
556    /// It must issue and dispatch a TLB flush after the operation. Otherwise,
557    /// the memory safety will be compromised. Please call this function less
558    /// to avoid the overhead of TLB flush. Using a large `len` is wiser than
559    /// splitting the operation into multiple small ones.
560    ///
561    /// # Panics
562    ///
563    /// Panics if:
564    ///  - the length is longer than the remaining range of the cursor;
565    ///  - the length is not page-aligned.
566    pub fn unmap(&mut self, len: usize) -> usize {
567        let end_va = self.virt_addr() + len;
568        let mut num_unmapped: usize = 0;
569        loop {
570            // SAFETY:
571            // 1. It is safe to unmap memory in the userspace.
572            // 2. We drop the unmapped items only after flushing TLB entries, which is safe.
573            let Some(frag) = (unsafe { self.pt_cursor.take_next(end_va - self.virt_addr()) })
574            else {
575                break; // No more mappings in the range.
576            };
577
578            match frag {
579                PageTableFrag::Mapped { va, item, .. } => {
580                    // SAFETY: If the item is not a scalar (e.g., a frame
581                    // pointer), we will drop it after the RCU grace period
582                    // (see `issue_tlb_flush_with`).
583                    let (item, panic_guard) = unsafe { RcuDrop::into_inner(item) };
584
585                    match item {
586                        VmItem {
587                            mapped_item: MappedItem::TrackedFrame(old_frame),
588                            ..
589                        } => {
590                            num_unmapped += 1;
591
592                            let rcu_frame = RcuDrop::new(old_frame);
593                            panic_guard.forget();
594                            let rcu_frame = Frame::rcu_from_unsized(rcu_frame);
595                            self.flusher
596                                .issue_tlb_flush_with(TlbFlushOp::for_single(va), rcu_frame);
597                        }
598                        VmItem {
599                            mapped_item: MappedItem::UntrackedIoMem { .. },
600                            ..
601                        } => {
602                            panic_guard.forget();
603
604                            // Flush the TLB entry for the current address, but
605                            // in the current design, we cannot drop the
606                            // corresponding `IoMem`. This is because we manage
607                            // the range of I/O as a whole, but the frames
608                            // handled here might be one segment of it.
609                            self.flusher.issue_tlb_flush(TlbFlushOp::for_single(va));
610                        }
611                    }
612                }
613                PageTableFrag::StrayPageTable {
614                    pt,
615                    va,
616                    len,
617                    num_frames,
618                } => {
619                    num_unmapped += num_frames;
620
621                    self.flusher.issue_tlb_flush_with(
622                        TlbFlushOp::for_range(va..va + len),
623                        Frame::rcu_from_unsized(pt),
624                    );
625                }
626            }
627        }
628
629        self.flusher.dispatch_tlb_flush();
630
631        num_unmapped
632    }
633
634    /// Applies the operation to the next slot of mapping within the range.
635    ///
636    /// The range to be found in is the current virtual address with the
637    /// provided length.
638    ///
639    /// The function stops and yields the actually protected range if it has
640    /// actually protected a page, no matter if the following pages are also
641    /// required to be protected.
642    ///
643    /// It also makes the cursor moves forward to the next page after the
644    /// protected one. If no mapped pages exist in the following range, the
645    /// cursor will stop at the end of the range and return [`None`].
646    ///
647    /// Note that it will **NOT** flush the TLB after the operation. Please
648    /// make the decision yourself on when and how to flush the TLB using
649    /// [`Self::flusher`].
650    ///
651    /// # Panics
652    ///
653    /// Panics if:
654    ///  - the length is longer than the remaining range of the cursor;
655    ///  - the length is not page-aligned.
656    pub fn protect_next(
657        &mut self,
658        len: usize,
659        mut op: impl FnMut(&mut PageFlags, &mut CachePolicy),
660    ) -> Option<Range<Vaddr>> {
661        // SAFETY: It is safe to set `PageFlags` and `CachePolicy` of memory
662        // in the userspace.
663        unsafe {
664            self.pt_cursor.protect_next(len, &mut |prop| {
665                op(&mut prop.flags, &mut prop.cache);
666            })
667        }
668    }
669
670    /// Run `f` on every mapped page in the range `start` to `end`. This will only run on non-I/O
671    /// regions that have TLB entries.
672    pub fn do_for_each_submapping<F>(&mut self, start: usize, end: usize, mut f: F) -> Result<()>
673    where
674        F: FnMut(&Range<Vaddr>, &Frame<dyn AnyUFrameMeta>, &PageProperty) -> Result<()>,
675    {
676        self.jump(start)?;
677
678        while self.virt_addr() < end {
679            // Query under the current cursor. If the virtual address is unmapped then we
680            // fail and cannot map a huge page.
681            let (sub_range, sub_mapping) = self.query()?;
682
683            // The mapping might not exist if the page hasn't been faulted in yet. We can
684            // skip over such mappings.
685            if let Some(VmQueriedItem::MappedRam {
686                frame: ref sub_frame,
687                prop: sub_props,
688            }) = sub_mapping
689            {
690                f(&sub_range, sub_frame, &sub_props)?;
691            }
692
693            if sub_range.end >= end {
694                break;
695            }
696            // Advance the cursor to the end of this mapping
697            self.jump(sub_range.end)?;
698        }
699
700        Ok(())
701    }
702}

719/// The result of a query over the VM space.
720pub enum VmQueriedItem<'a> {
721    /// The current slot is mapped, the frame within is allocated from the
722    /// physical memory.
723    MappedRam {
724        /// The mapped frame.
725        frame: FrameRef<'a, dyn AnyUFrameMeta>,
726        /// The property of the slot.
727        prop: PageProperty,
728    },
729    /// The current slot is mapped, the frame within is allocated from the
730    /// MMIO memory.
731    MappedIoMem {
732        /// The physical address of the corresponding I/O memory.
733        paddr: Paddr,
734        /// The property of the slot.
735        prop: PageProperty,
736    },
737}
738
739impl VmQueriedItem<'_> {
740    /// Returns the page property of the mapped item.
741    pub fn prop(&self) -> &PageProperty {
742        match self {
743            Self::MappedRam { prop, .. } => prop,
744            Self::MappedIoMem { prop, .. } => prop,
745        }
746    }
747}
748
749/// Internal representation of a VM item.
750///
751/// This is kept private to ensure memory safety. The public interface
752/// should use `VmQueriedItem` for querying mapping information.
753#[derive(Clone, Debug, PartialEq)]
754pub(crate) struct VmItem {
755    prop: PageProperty,
756    mapped_item: MappedItem,
757}
758
759/// A reference to a VM item.
760#[derive(Debug)]
761pub(crate) struct VmItemRef<'a> {
762    prop: PageProperty,
763    mapped_item: MappedItemRef<'a>,
764}
765
766#[derive(Clone, Debug, PartialEq)]
767enum MappedItem {
768    TrackedFrame(UFrame),
769    UntrackedIoMem { paddr: Paddr, level: PagingLevel },
770}
771
772#[derive(Debug)]
773enum MappedItemRef<'a> {
774    TrackedFrame(FrameRef<'a, dyn AnyUFrameMeta>),
775    UntrackedIoMem { paddr: Paddr, level: PagingLevel },
776}
777
778impl VmItem {
779    /// Creates a new `VmItem` that maps a tracked frame.
780    pub(super) fn new_tracked(frame: UFrame, prop: PageProperty) -> Self {
781        Self {
782            prop,
783            mapped_item: MappedItem::TrackedFrame(frame),
784        }
785    }
786
787    /// Creates a new `VmItem` that maps an untracked I/O memory.
788    fn new_untracked_io(paddr: Paddr, prop: PageProperty) -> Self {
789        Self {
790            prop,
791            mapped_item: MappedItem::UntrackedIoMem { paddr, level: 1 },
792        }
793    }
794}
795
796impl<'a> From<VmItemRef<'a>> for VmQueriedItem<'a> {
797    fn from(item: VmItemRef<'a>) -> Self {
798        match item.mapped_item {
799            MappedItemRef::TrackedFrame(frame) => VmQueriedItem::MappedRam {
800                frame,
801                prop: item.prop,
802            },
803            MappedItemRef::UntrackedIoMem { paddr, level } => {
804                debug_assert_eq!(level, 1);
805                VmQueriedItem::MappedIoMem {
806                    paddr,
807                    prop: item.prop,
808                }
809            }
810        }
811    }
812}

815pub(crate) struct UserPtConfig {}

817// SAFETY: `item_raw_info`, `item_into_raw`, `item_from_raw`, and
818// `item_ref_from_raw` are correctly implemented with respect to the `Item` and
819// `ItemRef` types.
820unsafe impl PageTableConfig for UserPtConfig {
821    const TOP_LEVEL_INDEX_RANGE: Range<usize> = 0..256;
822
823    type E = PageTableEntry;
824    type C = PagingConsts;
825
826    type Item = VmItem;
827    type ItemRef<'a> = VmItemRef<'a>;
828
829    fn item_raw_info(item: &Self::Item) -> (Paddr, PagingLevel, PageProperty) {
830        match &item.mapped_item {
831            MappedItem::TrackedFrame(frame) => {
832                let mut prop = item.prop;
833                prop.priv_flags -= PrivilegedPageFlags::AVAIL1; // Clear AVAIL1 for tracked frames
834                let level = frame.map_level();
835                let paddr = frame.paddr();
836                (paddr, level, prop)
837            }
838            MappedItem::UntrackedIoMem { paddr, level } => {
839                let mut prop = item.prop;
840                prop.priv_flags |= PrivilegedPageFlags::AVAIL1; // Set AVAIL1 for I/O memory
841                (*paddr, *level, prop)
842            }
843        }
844    }
845
846    unsafe fn item_from_raw(paddr: Paddr, level: PagingLevel, prop: PageProperty) -> Self::Item {
847        if prop.priv_flags.contains(PrivilegedPageFlags::AVAIL1) {
848            // `AVAIL1` is set, this is I/O memory.
849            VmItem::new_untracked_io(paddr, prop)
850        } else {
851            // `AVAIL1` is clear, this is tracked memory.
852            // SAFETY: The caller ensures safety.
853            let frame = unsafe { Frame::<dyn AnyUFrameMeta>::from_raw(paddr) };
854            // TODO(aneesh): this should really only be done during split_if_mapped_huge_all
855            frame.set_map_level(level);
856            debug_assert_eq!(frame.map_level(), level);
857            VmItem::new_tracked(frame, prop)
858        }
859    }
860
861    unsafe fn item_ref_from_raw<'a>(
862        paddr: Paddr,
863        level: PagingLevel,
864        prop: PageProperty,
865    ) -> Self::ItemRef<'a> {
866        if prop.priv_flags.contains(PrivilegedPageFlags::AVAIL1) {
867            // `AVAIL1` is set, this is I/O memory.
868            VmItemRef {
869                prop,
870                mapped_item: MappedItemRef::UntrackedIoMem { paddr, level },
871            }
872        } else {
873            // `AVAIL1` is clear, this is tracked memory.
874            // SAFETY: The caller ensures that the frame outlives `'a` and that
875            // the type matches the frame.
876            let frame_ref = unsafe { FrameRef::<dyn AnyUFrameMeta>::borrow_paddr(paddr) };
877            VmItemRef {
878                prop,
879                mapped_item: MappedItemRef::TrackedFrame(frame_ref),
880            }
881        }
882    }
883
884    fn split_item(item: Self::Item) -> Self::Item {
885        let VmItem {
886            prop: _,
887            mapped_item,
888        } = &item;
889        if let MappedItem::TrackedFrame(frame) = mapped_item {
890            frame.set_map_level(frame.map_level() - 1);
891        }
892        item
893    }
894
895    fn init_split_item_subpage(item: Self::Item, level: PagingLevel) {
896        // The subpages of a hugepage start in an uninitialized state, so the reference to frame we
897        // get below has an invalid reference count and should be treated as an unused frame. It
898        // might have contents though, so while we can treat the metadata as unused, we should not
899        // touch the page content.
900        let VmItem {
901            prop: _,
902            mapped_item,
903        } = item;
904        let MappedItem::TrackedFrame(frame) = mapped_item else {
905            panic!("init_split_item_subpage called on I/O item")
906        };
907
908        // TODO(aneesh): what should the metadata be here? We assume it's untyped user memory, in
909        // which case () is correct, but more generally it might be better to pass in a ref to the
910        // frame this was split from copy the metadata.
911        let _ = ManuallyDrop::new(Frame::from_unused(frame.paddr(), (), level));
912
913        // We call forget here because we don't want to affect the reference count we initialized.
914        core::mem::forget(frame);
915    }
