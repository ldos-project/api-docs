29/// Tasks can block waiting for a [`Blocker`] to notify them to retry an action.
30///
31/// User code should generally only call [`Blocker::block_until`] or [`CurrentTask::block_on`].
32pub trait Blocker {
33    /// Return true if performing the action may succeed and should be attempted. This must be *very* fast and cannot
34    /// block for any condition itself. This is because it will be called inside the scheduler with locks held.
35    ///
36    /// This should be an approximation of the success of a `try_` function such as
37    /// [`Consumer::try_consume`](`crate::orpc::oqueue::Consumer::try_consume`). This *must* return
38    /// true if `try`ing would succeed, but may also return true spuriously even if it will fail.
39    ///
40    /// This must have Acquire ordering.
41    fn should_try(&self) -> bool;
42
43    /// Add a task to the wait queue of `self`. After this call, the task must be awoken if [`Self::should_try`] may
44    /// return `true` again.
45    ///
46    /// This must have Release ordering.
47    ///
48    /// This returns an ID which can be passed to [`Self::remove`] (on the same instance) to improve the
49    /// performance of removal.
50    fn enqueue(&self, waker: &Arc<Waker>) -> WakerKey;
51
52    /// Remove a task from the wait queue of self. After this call, the task will not be awoken when
53    /// when `should_try` will return true. This undoes the effect of [`Self::enqueue`].
54    fn remove(&self, key: WakerKey);
55
56    /// Block on self repeately until `cond` returns Some. This assumes that this blocker will be woken if `cond()`
57    /// would change.
58    ///
59    /// To block on multiple `Blockers`, use [`CurrentTask::block_on`].
60    fn block_until<T>(&self, cond: impl Fn() -> Option<T>) -> T
61    where
62        Self: Sized,
63    {
64        loop {
65            let ret = cond();
66            match ret {
67                Some(returned) => {
68                    return returned;
69                }
70                None => {
71                    if let Some(t) = Task::current() {
72                        t.block_on(&[self]);
73                    }
74                }
75            };
76        }
77    }
78}

80// Optional Blockers are allowed. If the blocker is `None`, then it will never wake up and performs
81// no other handling.
82impl<T: Blocker> Blocker for Option<T> {
83    fn should_try(&self) -> bool {
84        self.as_ref().is_some_and(T::should_try)
85    }
86
87    fn enqueue(&self, waker: &Arc<Waker>) -> WakerKey {
88        if let Some(blocker) = self {
89            blocker.enqueue(waker)
90        } else {
91            WakerKey::default()
92        }
93    }
94
95    fn remove(&self, key: WakerKey) {
96        if let Some(blocker) = self {
97            blocker.remove(key)
98        }
99    }
100}

102impl CurrentTask {
103    /// Wait for multiple blockers, waking if any wake. This is equivalent to
104    /// [`Blocker::block_until`] if there is only one blocker.
105    pub fn block_on<const N: usize>(&self, blockers: &[&dyn Blocker; N]) {
106        CurrentServer::abort_point();
107        let (waiter, waker) = Waiter::new_pair();
108
109        // 1. Register for all blockers.
110        let keys = blockers.map(|blocker| blocker.enqueue(&waker));
111
112        // 2. Check if any of the blockers are actually ready.
113        if !blockers.iter().any(|b| b.should_try()) {
114            // 3. Block if no blockers are ready. This will immediately wake if any blocker woke
115            //    between `prepare_to_wait` and here. This prevents this thread from dropping a wake
116            //    between `should_try` and `wait_cancellable`.
117            waiter.wait_cancellable(|| {
118                if CurrentServer::is_aborted() {
119                    Err(())
120                } else {
121                    Ok(())
122                }
123            });
124        }
125
126        // 4. Unregister with all the blockers. This avoids the blocker queues growing without
127        //    bound.
128        for (blocker, key) in blockers.iter().zip(keys) {
129            blocker.remove(key);
130        }
131
132        CurrentServer::abort_point();
133    }
134}

136/// The struct implementing blocking on an unknown number of blockers at the same time.
137///
138/// This is a struct instead of a method to allow caching an internal allocation between calls.
139/// Without this, each call would cause an allocation and free. The optimization assumes that each
140/// call to [`Self::block_on`] will be called with an iterator of roughly the same length.
141pub struct BlockOnMany {
142    cached_keys: Vec<WakerKey>,
143}
144
145impl BlockOnMany {
146    /// Create a new BlockOnMany
147    pub fn new() -> Self {
148        BlockOnMany {
149            cached_keys: Vec::new(),
150        }
151    }
152
153    /// Wait for multiple blockers, waking if any wake. This clones `blockers` 3 times, so it is
154    /// important that `clone()` be fast.
155    pub fn block_on<'a>(&mut self, blockers: impl Iterator<Item = &'a dyn Blocker> + Clone) {
156        CurrentServer::abort_point();
157        let (waiter, waker) = Waiter::new_pair();
158
159        // 1. Register for all blockers.
160        self.cached_keys
161            .extend(blockers.clone().map(|blocker| blocker.enqueue(&waker)));
162
163        // 2. Check if any of the blockers are actually ready.
164        if !blockers.clone().any(|b| b.should_try()) {
165            // 3. Block if no blockers are ready. This will immediately wake if any blocker woke
166            //    between `enqueue` and here. This prevents this thread from dropping a wake between
167            //    `should_try` and `wait_cancellable`.
168            waiter.wait_cancellable(|| {
169                if CurrentServer::is_aborted() {
170                    Err(())
171                } else {
172                    Ok(())
173                }
174            });
175        }
176
177        // 4. Unregister with all the blockers. This avoids the blocker queues growing without
178        //    bound.
179        for (blocker, key) in blockers.zip(self.cached_keys.drain(..)) {
180            blocker.remove(key);
181        }
182
183        CurrentServer::abort_point();
184    }
185}
186
187impl Default for BlockOnMany {
188    fn default() -> Self {
189        Self::new()
190    }
