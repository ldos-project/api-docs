3//! A [`Blocker`] that becomes ready after a timeout.
4
5use alloc::sync::Arc;
6use core::sync::atomic::{AtomicU64, Ordering};
7
8use super::Blocker;
9use crate::{
10    sync::{WaitQueue, Waker, WakerKey},
11    timer::{self, Jiffies},
12};
13
14/// A [`Blocker`] that becomes ready once an armed jiffies deadline elapses.
15///
16/// Add it to a wait set (e.g. [`BlockOnMany::block_on`]), potentially alongside other
17/// blockers, so a thread that is waiting for events can also be woken
18/// by a timeout. It could be used to build loops that want to block until
19/// *either* a message arrives *or* a deadline passes.
20///
21/// It starts disarmed and can be armed and disarmed repeatedly:
22/// - [`arm_at`](Self::arm_at) / [`arm_after`](Self::arm_after) set the deadline.
23/// - [`disarm`](Self::disarm) makes it never fire.
24///
25/// # Implementation and caveats
26///
27/// [`new`](Self::new) registers a per-tick timer callback (via
28/// [`timer::register_callback_on_cpu`]) that wakes the blocker's waiters once the
29/// deadline passes. That callback lives as long as the process runs and **cannot
30/// be unregistered**, so this type is intended for long-lived blockers rather
31/// than short-lived, frequently-created ones. Once the `TimeoutBlocker` is
32/// dropped, the callback becomes an inexpensive no-op (it holds only a weak
33/// reference and does one atomic load per tick).
34///
35/// [`BlockOnMany::block_on`]: super::BlockOnMany::block_on
36pub struct TimeoutBlocker {
37    /// Absolute jiffies deadline; `u64::MAX` means disarmed.
38    deadline: AtomicU64,
39    /// Waiters to wake once the deadline elapses.
40    wait_queue: WaitQueue,
41}
42
43impl TimeoutBlocker {
44    /// Creates a new, disarmed `TimeoutBlocker`.
45    ///
46    /// The value is returned inside an [`Arc`] because the internal timer
47    /// callback holds a weak reference to it.
48    pub fn new() -> Arc<Self> {
49        let this = Arc::new(Self {
50            deadline: AtomicU64::new(u64::MAX),
51            wait_queue: WaitQueue::new(),
52        });
53
54        // Wake the registered waiters once the deadline elapses. The callback
55        // holds only a `Weak` reference, so it becomes a no-op if the blocker is
56        // dropped.
57        let weak = Arc::downgrade(&this);
58        timer::register_callback_on_cpu(move || {
59            let Some(this) = weak.upgrade() else {
60                return;
61            };
62            if Jiffies::elapsed().as_u64() >= this.deadline.load(Ordering::Relaxed) {
63                this.wait_queue.wake_all();
64            }
65        });
66
67        this
68    }
69
70    /// Arms the timeout to fire at the absolute jiffies value `deadline`.
71    pub fn arm_at(&self, deadline: u64) {
72        self.deadline.store(deadline, Ordering::Relaxed);
73    }
74
75    /// Arms the timeout to fire `jiffies` jiffies from now.
76    pub fn arm_after(&self, jiffies: u64) {
77        let deadline = Jiffies::elapsed().as_u64().saturating_add(jiffies);
78        self.deadline.store(deadline, Ordering::Relaxed);
79    }
80
81    /// Disarms the timeout so it never fires.
82    pub fn disarm(&self) {
83        self.deadline.store(u64::MAX, Ordering::Relaxed);
84    }
85}
86
87impl Blocker for TimeoutBlocker {
88    fn should_try(&self) -> bool {
89        Jiffies::elapsed().as_u64() >= self.deadline.load(Ordering::Relaxed)
90    }
91
92    fn enqueue(&self, waker: &Arc<Waker>) -> WakerKey {
93        self.wait_queue.enqueue(waker.clone())
94    }
95
96    fn remove(&self, key: WakerKey) {
97        self.wait_queue.remove(key)
98    }
99}
