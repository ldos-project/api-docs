3//! ORPC monitors are objects which hold state and have a set of methods which operate on that
4//! state. They are **not** servers, and are orthogonal to them. As such, monitors must be inside a
5//! server. The standard usage is to have a single monitor holding all the state of the server and
6//! forwarding all server methods into that monitor.
7//!
8//! Monitors are created using the [`orpc_monitor`] macro.
9
10pub use orpc_macros::orpc_monitor;
11
12use crate::orpc::oqueue::{ValueProducer, new_reply_pair};
13
14/// **INTERNAL FOR MACRO USE ONLY**
15///
16/// Make a synchronous request over an OQueue. `make_request` is called with the reply producer
17/// attachment to create the request message. The request is sent over `producer`. The consumer of
18/// that message should guarantee exactly one reply will be published.
19#[doc(hidden)]
20pub fn synchronous_request<C: Send + 'static, R: Send + 'static>(
21    producer: &ValueProducer<C>,
22    make_request: impl FnOnce(ValueProducer<R>) -> C,
23) -> R {
24    let (reply_producer, reply_consumer) = new_reply_pair::<R>();
25    producer.produce(make_request(reply_producer));
26    reply_consumer.consume()
27}

46    #[orpc_server()]

51    pub(super) struct TestState {
52        x: i32,
53    }
54
55    #[orpc_monitor(pub)]
56    impl TestState {
57        /// A method which can be attached to an OQueue as an observer so it is called everytime
58        /// something is produced. This attachment is done with a generated method called
59        /// `attach_update`.
60        #[strong_observer]
61        pub(super) fn update(&mut self, x: i32) -> Result<(), RPCError> {
62            self.x = (self.x + x * 3) / 4;
63            Ok(())
64        }
65
66        /// A method which can be attached as a consumer. This is analogous to the observer
67        /// capabilities of `update`.
68        #[consumer]
69        pub(super) fn next(&mut self, _: ()) -> Result<(), RPCError> {
70            self.x += 1;
71            Ok(())
72        }
73
74        pub(super) fn get(&mut self) -> Result<i32, RPCError> {
75            Ok(self.x)
76        }
77    }

79    fn spawn_server() -> Arc<TestServer> {
80        let server = TestServer::new_with(|orpc_internal, _| TestServer {
81            monitor: TestStateMonitor::new(Path::test()),
82            orpc_internal,
83        });
84        server.monitor.start(server.clone(), TestState { x: 0 });
85        server
86    }

88    #[ktest]
89    fn monitor_updates_from_strong_observer() {
90        let values = GenericOQueueRef::new(2, Path::test());
91        let server = spawn_server();
92        server
93            .monitor
94            .attach_update(
95                values
96                    .attach_strong_observer(ObservationQuery::identity())
97                    .unwrap(),
98            )
99            .unwrap();
100
101        let producer = values.attach_ref_producer().unwrap();
102
103        producer.produce_ref(&0);
104        producer.produce_ref(&100);
105        assert_eq_eventually!(server.monitor.get().unwrap(), 75);
106
107        producer.produce_ref(&100);
108        assert_eq_eventually!(server.monitor.get().unwrap(), 93);
109    }
