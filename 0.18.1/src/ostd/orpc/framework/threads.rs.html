3//! The framework for starting threads associated with a server.
4//!
5//! You should consider if using a thread is correct, or if you should use the message handler
6//! framework. The message handler framework is more flexible and declarative.
7
8use log::error;
9use spin::Once;
10
11use crate::{
12    orpc::{
13        errors,
14        framework::{CurrentServer, Server},
15    },
16    prelude::{Arc, Box},
17    task::TaskOptions,
18};
19
20/// The body of a ORPC thread as a closure.
21pub type ThreadMain = Box<dyn FnOnce() -> Result<(), Box<dyn core::error::Error>> + Send + 'static>;
22
23/// The type of the function used to implement the `spawn_thread` function.
24pub(crate) type SpawnThreadFn = fn(Arc<dyn Server + Send + Sync + 'static>, ThreadMain);
25
26/// Injected function for spawning new threads. See [`inject_spawn_thread`].
27pub(crate) static SPAWN_THREAD_FN: Once<SpawnThreadFn> = Once::new();
28
29/// Start a new server thread. This should only be called while spawning a server.
30pub fn spawn_thread(
31    server: Arc<dyn Server + Send + Sync + 'static>,
32    body: impl (FnOnce() -> Result<(), Box<dyn core::error::Error>>) + Send + 'static,
33) {
34    if let Some(spawn_fn) = SPAWN_THREAD_FN.get() {
35        spawn_fn(server, Box::new(body));
36        return;
37    }
38
39    TaskOptions::new(wrap_server_thread_body(server, Box::new(body)))
40        .spawn()
41        .unwrap();
42}
43
44/// Return a closure wrapping the body of a server thread with the machinery to setup the execution
45/// context and catch and handle errors.
46///
47/// This should only be used by spawn_thread implementations which are injected using
48/// [`inject_spawn_thread`].
49pub fn wrap_server_thread_body(
50    server: Arc<dyn Server + Send + Sync + 'static>,
51    body: ThreadMain,
52) -> impl FnOnce() {
53    move || {
54        if let Err(payload) = crate::panic::catch_unwind({
55            let server = server.clone();
56            move || {
57                Server::orpc_server_base(server.as_ref()).attach_task();
58                let _server_context =
59                    CurrentServer::enter_server_context(server.orpc_server_base());
60                if let Err(e) = body() {
61                    Server::orpc_server_base(server.as_ref()).abort(&e);
62                }
63            }
64        }) {
65            let err = errors::RPCError::from_panic(payload);
66            error!("Server thread panicked: {}", err);
67            Server::orpc_server_base(server.as_ref()).abort(&err);
68        }
69    }
70}
71
72/// Set a custom function for spawning threads. This allows overriding the default thread spawning
73/// behavior. This is required in kernels, like Asterinas, that do not run raw OSTD
74/// [`Task`](`ostd::task::Task`)s correctly.
75pub fn inject_spawn_thread(func: SpawnThreadFn) {
76    SPAWN_THREAD_FN.call_once(|| func);
77}
