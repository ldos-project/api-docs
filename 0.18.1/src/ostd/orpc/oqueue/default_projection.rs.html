13/// Supplies the value exported for an OQueue message when using [`register_with`](super::registry::register_with).
14///
15/// The `DefaultProjection` derive recursively projects fields that implement this trait. POD
16/// values project to themselves; mark fields that should be opaque identifiers with
17/// `#[projection(opaque)]`, which projects them through [`OpaqueId`].
18pub trait DefaultProjection {
19    /// The projected value type written to the export stream.
20    type Projection: Copy + Send + Serialize + 'static;
21
22    /// Returns the default projection of this message.
23    fn default_projection(&self) -> Self::Projection;
24}
25
26/// Tag to mark types as trivially projectable
27pub trait Projectable {}
28
29impl Projectable for bool {}
30impl Projectable for i32 {}
31impl Projectable for i64 {}
32impl Projectable for isize {}
33impl Projectable for u32 {}
34impl Projectable for u64 {}
35impl Projectable for usize {}
36impl Projectable for f32 {}
37impl Projectable for f64 {}
38
39impl Projectable for core::net::Ipv4Addr {}
40impl Projectable for core::net::Ipv6Addr {}
41
42impl<T> DefaultProjection for T
43where
44    T: Copy + Send + Serialize + Projectable + 'static,
45{
46    type Projection = Self;
47
48    fn default_projection(&self) -> Self::Projection {
49        *self
50    }
51}

53/// An ID of an object. This include an opaque ID (often a pointer value), and the type name. The
54/// type name is stored as a reference to a static string, so this struct is small and `Copy`.
55#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
56pub struct OID {
57    /// An opaque ID of the object. This value has no meaning, but is unique for the lifetime of the
58    /// object.
59    ///
60    /// NOTE: Ideally, this would be unique over the lifetime of the system. However, that isn't
61    /// true yet.
62    pub id: usize,
63
64    /// The name of the type of the object. This comes from [`core::any::type_name`] and, like that,
65    /// has an undefined format, so it cannot safely be parsed.
66    pub type_name: &'static str,
67}
68
69impl OID {
70    /// Create the OID for a value. `T` should be provided explicitly because doing so forces the
71    /// compiler to generate the ID from something of type `T`. That prevents accidentally
72    /// generating the ID from a pointer to a reference or a box-like type. The compiler auto-deref
73    /// rules will generally unwrap the object as needed to get the correct reference.
74    pub fn of_ref<T: ?Sized>(v: &T) -> OID {
75        OID {
76            id: core::ptr::from_ref(v) as *const () as usize,
77            type_name: type_name::<T>(),
78        }
79    }
80}
81
82/// Supplies an opaque identifier for a value projected with `#[projection(opaque)]`.
83///
84/// By default, the identifier is the address of the value itself along with the name of the type.
85/// Smart pointers and references identify the value they point to, so that copies of the same smart
86/// pointer share an identifier.
87pub trait OpaqueId {
88    /// Returns the opaque identifier of this value.
89    fn opaque_id(&self) -> OID;
90}
91
92impl<T: ?Sized> OpaqueId for &T {
93    fn opaque_id(&self) -> OID {
94        OID::of_ref::<T>(self)
95    }
96}
97
98impl<T: ?Sized> OpaqueId for Arc<T> {
99    fn opaque_id(&self) -> OID {
100        OID::of_ref::<T>(self)
101    }
102}
103
104impl<T: ?Sized> OpaqueId for Box<T> {
105    fn opaque_id(&self) -> OID {
106        OID::of_ref::<T>(self)
107    }
108}

110#[cfg(ktest)]
111mod test {
112    use serde::Serialize;
113
114    use super::DefaultProjection;
115    use crate::{orpc::oqueue::default_projection::OID, prelude::*};
116
117    // If bool is DefaultProjection this should compile.
118    // This is worth testing as bool is treated slightly special and only implements IntoBytes and
119    // not FromBytes, which caused problems when we had a blanked impl on the trait Pod.
120    #[expect(dead_code)]
121    #[derive(DefaultProjection)]
122    struct TestProjectBool(bool);
123
124    #[derive(DefaultProjection)]
125    struct ProjectedLeaf {
126        value: u32,
127    }
128
129    #[derive(Clone, Copy, serde::Serialize)]
130    struct SpecialProjectionOutput(u32);
131
132    struct SpecialProjection(u32);
133
134    impl DefaultProjection for SpecialProjection {
135        type Projection = SpecialProjectionOutput;
136
137        fn default_projection(&self) -> Self::Projection {
138            SpecialProjectionOutput(self.0 + 1)
139        }
140    }
141
142    #[derive(DefaultProjection)]
143    struct ProjectedMessage {
144        leaf: ProjectedLeaf,
145        #[projection(opaque)]
146        ref_type: &'static str,
147        special: SpecialProjection,
148    }
149
150    #[ktest]
151    fn default_projection_serializes_nested_custom_and_opaque_fields() {
152        let message = ProjectedMessage {
153            leaf: ProjectedLeaf { value: 0x1234_5678 },
154            ref_type: "opaque",
155            special: SpecialProjection(41),
156        };
157        let projected = message.default_projection();
158
159        let mut actual = Vec::new();
160        Serialize::serialize(
161            &projected,
162            &mut minicbor_serde::Serializer::new(&mut actual),
163        )
164        .unwrap();
165
166        let mut decoder = minicbor::Decoder::new(&actual);
167        assert_eq!(decoder.map().unwrap(), Some(3));
168        assert_eq!(decoder.str().unwrap(), "leaf");
169        assert_eq!(decoder.map().unwrap(), Some(1));
170        assert_eq!(decoder.str().unwrap(), "value");
171        assert_eq!(decoder.u32().unwrap(), message.leaf.value);
172        assert_eq!(decoder.str().unwrap(), "ref_type");
173        assert_eq!(decoder.map().unwrap(), Some(2));
174        assert_eq!(decoder.str().unwrap(), "id");
175        decoder.u64().unwrap();
176        assert_eq!(decoder.str().unwrap(), "type_name");
177        assert!(decoder.str().unwrap().contains("str"));
178        assert_eq!(decoder.str().unwrap(), "special");
179        assert_eq!(decoder.u32().unwrap(), 42);
180        assert_eq!(decoder.position(), actual.len());
181    }
182
183    #[derive(DefaultProjection)]
184    struct ProjectedPointers<'a> {
185        #[projection(opaque)]
186        arc: Arc<u32>,
187        #[projection(opaque)]
188        boxed: Box<u32>,
189        #[projection(opaque)]
190        reference: &'a u32,
191    }
192
193    #[ktest]
194    fn default_projection_get_inner_value_pointer_from_smart_ptr() {
195        let data: u32 = 3;
196
197        let arc = Arc::new(1);
198        let message = ProjectedPointers {
199            arc: arc.clone(),
200            boxed: Box::new(2),
201            reference: &data,
202        };
203        let projected = message.default_projection();
204
205        assert_eq!(projected.arc, OID::of_ref(arc.as_ref()));
206        assert_eq!(projected.boxed, OID::of_ref(message.boxed.as_ref()));
207        assert_eq!(projected.reference, OID::of_ref(&data));
208    }
