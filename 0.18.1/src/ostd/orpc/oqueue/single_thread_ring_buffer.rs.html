2//! A simple implementation of OQueues used for [`super::OQueue`] using locks. This is a baseline
3//! implementation that supports all features, but it may have performance issues in highly
4//! contended cases.
5
6use alloc::alloc::AllocError;
7use core::{
8    any::{TypeId, type_name},
9    mem::MaybeUninit,
10};
11
12use smallvec::SmallVec;

14use crate::{orpc::oqueue::Cursor, util::untyped_box::SliceAllocation};

16/// A non-thread-safe ring-buffer which supports multiple strong readers (consumers and strong
17/// observers) and weak observers. This is used for the internal ring-buffers of the locking OQueue
18/// implementation ([`LockingOQueueInner`]). It will management multiple instances to store
19/// extracted data for observers.
20///
21/// This does not need to distinguish consumers from strong observers because it is not thread-safe,
22/// so multiple readers can use the same head. When there are consumers they will all use head `0`.
23/// Consumers are incompatible with strong or weak observers. This is represented by the fact that
24/// `try_consume` is unsafe, because it will cause UB in later `try_strong_observe` and
25/// `try_weak_observe` calls.
26///
27/// The values in the ring buffer and opaque sequences of bytes to erase their type. Some accessors
28/// have type parameters to specify the type that is in the buffer (which is also dynamically
29/// checked). Specifically, operations which "take" a non-`Copy` value out of the buffer. This is
30/// needed to make sure that a potential `Drop` on that type is called.
31///
32/// Many operation require `Send` on the type they are manipulating, since the `RingBuffer` needs to
33/// be `Send` so it can be placed inside a lock.
34///
35/// ## Terminology
36///
37/// * a *strong reader* is either a consumer or a strong observers. Both have the same behavior in
38///   the context of this type.
39/// * a *slot* is a space in the buffer to hold an element. Often identified by an index into the
40///   array. The term "slot" distinguishes a value form an index.
41/// * an *index* is a position into the abstract infinite length sequence. This RingBuffer stores a
42///   tail of that sequence.
43///
44/// ## Safety Note
45///
46/// This type is `Send` and that is only safe if the erased type held in this is also `Send`. This
47/// is guaranteed by the type bound on [`RingBuffer::new`]. The implementation is internally checked
48/// by the bound on [`RingBuffer::slot_mut`].
49///
50/// This will attempt to drop any values still in the buffer when this is dropped when they are
51/// overwritten. However, if there is no consumer, this will not happen. Because of this, any
52/// RingBuffer with a type which is not `Copy` should have a consumer.
53pub(super) struct RingBuffer {
54    /// The type of the elements.
55    element_type: TypeId,
56
57    /// The name of the type of this ring buffer for debug messages. The errors are rare enough that
58    /// this will likely be removed in the future.
59    #[cfg(debug_assertions)]
60    element_type_name: &'static str,
61
62    /// The storage space for ringbuffer elements. This is a raw buffer which will be indexed based
63    /// on the actual type of elements.
64    buffer: SliceAllocation,
65
66    /// The index of the next element to write in the buffer. Used by producers. This must be stored
67    /// for each buffer because different ring may move at different speed if there are filters in
68    /// queries.
69    tail_index: usize,
70
71    /// The heads used by consumers and strong observers.
72    ///
73    /// Using [`SmallVec`] here allows up to 2 heads to exist inline to the RingBuffer struct. This
74    /// will be enough for many cases and avoids needing a pointer dereference to get the head in
75    /// the consumer case.
76    strong_reader_heads: SmallVec<[usize; 2]>,
77
78    /// Function to drop all values still in the queue. This is safe, because it will simply discard
79    /// values until the queue has no droppable values in it.
80    drop_fn: fn(&mut RingBuffer),
81}
82
83impl Drop for RingBuffer {
84    fn drop(&mut self) {
85        (self.drop_fn)(self);
86    }
87}
88
89impl RingBuffer {
90    /// Create a new ring buffer for a specific type.
91    pub(super) fn new<T: Send + 'static>(n_buffer_elements: usize) -> Result<Self, AllocError> {
92        // TODO: PERFORMANCE: We could potentially eliminate the padding and not align the values in
93        // the buffer to save space. However, safety of this will require more thinking and may well
94        // not help, so we leave it as is for now.
95        let buffer = SliceAllocation::new::<T>(n_buffer_elements)?;
96
97        Ok(RingBuffer {
98            element_type: TypeId::of::<T>(),
99            #[cfg(debug_assertions)]
100            element_type_name: type_name::<T>(),
101            buffer,
102            tail_index: 0,
103            strong_reader_heads: SmallVec::default(),
104            drop_fn: |this| {
105                if this.strong_reader_heads.is_empty() {
106                    // There is no reader so there is no way to know what part of the buffer can be
107                    // safely dropped.
108                    return;
109                }
110                let head_id = 0;
111                while let Some(v) = this.try_get_for_head::<T>(head_id) {
112                    // SAFETY: The return value of `try_get_for_head` is always initialized. If head
113                    // ID 0 is the consumer and `T` has a `drop`, then we are dropping the values as
114                    // if they were consumed. If head ID 0 is a strong observer, draining is not
115                    // needed because `T: Copy`. Regardless, there are no active consumer or
116                    // observers because `this` is being dropped.
117                    unsafe { v.assume_init_read() };
118                    this.strong_reader_heads[head_id] += 1;
119                }
120            },
121        })
122    }
123
124    /// Panic if the type doesn't match this ringbuffer.
125    ///
126    /// TODO: PERFORMANCE: This is called a lot, and while it will probably be elided most of the
127    /// time. Making the callers to this unsafe and having their callers perform the check might
128    /// reduce the number of checks.
129    #[inline(always)]
130    #[track_caller]
131    pub(super) fn assert_type<T: Send + 'static>(&self) {
132        #[cfg(debug_assertions)]
133        let element_type_name = self.element_type_name;
134        #[cfg(not(debug_assertions))]
135        let element_type_name = "[not captured]";
136        assert_eq!(
137            TypeId::of::<T>(),
138            self.element_type,
139            "requested {} constructed with {}",
140            type_name::<T>(),
141            element_type_name
142        );
143    }
144
145    fn mod_len(&self, i: usize) -> usize {
146        i % self.buffer.len()
147    }
148
149    /// Get a reference to the slot which is used for the slot index `i`. This does no checking as
150    /// to the data that is actually held in that slot. It could be uninitialized.
151    fn slot_mut<T: Send + 'static>(&mut self, i: usize) -> &mut MaybeUninit<T> {
152        self.assert_type::<T>();
153
154        // SAFETY: buffer was constructed with type `T` as checked above and the type is `Send`, so
155        // reading the value out of `self` (which is `Send`) is allowed.
156        unsafe { self.buffer.at_mut_unchecked(i) }
157    }
158
159    /// True if it is safe to produce by writing a value into the buffer and increasing the
160    /// tail_index.
161    pub(super) fn can_produce(&self) -> bool {
162        let next_tail_slot = self.mod_len(self.tail_index + 1);
163
164        !self
165            .strong_reader_heads
166            .iter()
167            .any(|h| next_tail_slot == self.mod_len(*h))
168    }
169
170    /// Try to write the value into the buffer. This will return `Some(v)` if `v` could not be moved
171    /// into the buffer.
172    pub(super) fn try_produce<T: Send + 'static>(&mut self, v: T) -> Option<T> {
173        self.assert_type::<T>();
174
175        // TODO: PERFORMANCE: The caller will often already have checked `can_produce`, so maybe we
176        // should use a `produce_unchecked` instead.
177        if !self.can_produce() {
178            return Some(v);
179        };
180
181        let uninit_ref = self.slot_mut(self.mod_len(self.tail_index));
182
183        // This can destroy an instance without dropping when there is no consumer. This is still
184        // safe. See the note in the safety section of the RingBuffer doc.
185        uninit_ref.write(v);
186
187        self.tail_index += 1;
188
189        None
190    }
191
192    pub(super) fn can_get_for_head(&self, head_id: usize) -> bool {
193        self.strong_reader_heads[head_id] != self.tail_index
194    }
195
196    /// Get a reference to a value at the head stored in `strong_reader_heads[head_id]`. The
197    /// returned value is always initialized when returned. It is returned as `MaybeUninit<T>`, so
198    /// allow mutating it leaving it in an uninitialized state. This does not update that head.
199    fn try_get_for_head<T: Send + 'static>(
200        &mut self,
201        head_id: usize,
202    ) -> Option<&mut MaybeUninit<T>> {
203        self.assert_type::<T>();
204
205        if self.can_get_for_head(head_id) {
206            Some(self.slot_mut(self.mod_len(self.strong_reader_heads[head_id])))
207        } else {
208            None
209        }
210    }
211
212    /// Consume the value at head 0 (`strong_reader_heads[0]`).
213    ///
214    /// ## Safety
215    ///
216    /// There must be only one head, i.e., `strong_reader_heads.len() == 1`, and weak observers may
217    /// not be in use.
218    pub(super) unsafe fn try_consume<T: Send + 'static>(&mut self) -> Option<T> {
219        self.assert_type::<T>();
220
221        debug_assert_eq!(
222            self.strong_reader_heads.len(),
223            1,
224            "consuming from ring buffer with more than one head"
225        );
226        let res = self.try_get_for_head(0)?;
227        // SAFETY: The return value of `try_get_for_head` is always initialized. The only head is
228        // incremented below, meaning the slot will be treated as uninit until overwritten again
229        // preventing a double drop.
230        let res = unsafe { res.assume_init_read() };
231        self.strong_reader_heads[0] += 1;
232        Some(res)
233    }
234
235    pub(super) fn try_strong_observe<T>(&mut self, head_id: usize) -> Option<T>
236    where
237        T: Copy + Send + 'static,
238    {
239        self.assert_type::<T>();
240
241        let res = self.try_get_for_head(head_id)?;
242        // SAFETY: The return value of `try_get_for_head` is always initialized. This leaves the
243        // value initialized (as it is `Copy`). `T` is `!Drop` (due to being `Copy`), so it's OK to
244        // leave the value initialized until it is overwritten.
245        let res = unsafe { res.assume_init_read() };
246        self.strong_reader_heads[head_id] += 1;
247        Some(res)
248    }
249
250    pub(super) fn try_weak_observe<T>(&mut self, index: Cursor) -> Option<T>
251    where
252        T: Copy + Send + 'static,
253    {
254        self.assert_type::<T>();
255
256        let Cursor(index) = index;
257        if index < self.tail_index.saturating_sub(self.buffer.len()) || index >= self.tail_index {
258            return None;
259        }
260        let slot = self.slot_mut(self.mod_len(index));
261        // SAFETY: `index < self.tail_index` so the slot must have been initialized at some point
262        // and since there are no consumers the value will not have been invalidated. Safety does
263        // not depend on the slot containing the current value.
264        Some(unsafe { slot.assume_init_read() })
265    }
266
267    /// Allocate a new strong reader head ID. To allow consumers, this should only be called once.
268    pub(super) fn new_strong_reader(&mut self) -> usize {
269        let id = self.strong_reader_heads.len();
270        self.strong_reader_heads.push(self.tail_index);
271        id
272    }
273
274    /// Get the cursor for the most recent value in `self`.
275    pub(super) fn newest_cursor(&self) -> Cursor {
276        Cursor(self.tail_index.saturating_sub(1))
277    }
278
279    /// Get the cursor for the oldest value still available in `self`.
280    pub(super) fn oldest_cursor(&self) -> Cursor {
281        let Cursor(i) = self.newest_cursor();
282        // Return the (most recent() - (the buffer size) or zero if the buffer isn't full yet.
283        if i < self.buffer.len() {
284            Cursor(0)
285        } else {
286            Cursor(i - (self.buffer.len() - 1))
287        }
288    }
