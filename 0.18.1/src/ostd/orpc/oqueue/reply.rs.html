3//! A potentially specialized OQueue implementation used only for replies to requests. These OQueues
4//! are expected to be transient and can be optimized for that case.
5//!
6//! (NOTE: Currently, this is just a normal OQueue.)
7
8use crate::orpc::oqueue::{
9    ConsumableOQueue, ConsumableOQueueRef, Consumer, OQueueError, ValueProducer,
10};
11
12/// The OQueue implementation to use for ephemeral reply queues.
13///
14/// (NOTE: This exists as an alias so that it can later be replaced with an optimized implementation
15/// if needed.)
16pub type ReplyQueue<T> = ConsumableOQueueRef<T>;
17
18type ReplyHandlePair<T> = (ValueProducer<T>, Consumer<T>);
19
20impl<T: Send + 'static> ReplyQueue<T> {
21    /// Construct a producer/consumer pair for handling an async message reply.
22    pub fn new_pair() -> Result<ReplyHandlePair<T>, OQueueError> {
23        let oqueue = ConsumableOQueueRef::new_anonymous(2);
24        Ok((oqueue.attach_value_producer()?, oqueue.attach_consumer()?))
25    }
26}
