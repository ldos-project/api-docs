60/// The OQueue event a [`WakeHook`] waits on.
61#[derive(Clone, Copy, PartialEq, Eq, Debug)]
62pub(super) enum WakeEvent {
63    /// A value was produced, so observers and consumers may have something to read.
64    Produced,
65    /// The last full buffer that blocked producers gained space, so a produce would not block.
66    SpaceFreed,
67}
68
69/// A hook registered with `on_ready` on [`ValueProducer`](super::ValueProducer),
70/// [`Consumer`](super::Consumer), or [`StrongObserver`](super::StrongObserver). The hook stays
71/// registered until this is dropped.
72pub struct WakeHook {
73    oqueue: Weak<dyn UntypedOQueueImplementation>,
74    event: WakeEvent,
75    key: WakeHookKey,
76}
77
78impl WakeHook {
79    /// Registers `hook` to run on each `event` of `oqueue`.
80    ///
81    /// The hook runs in atomic context, so it must not sleep or add or remove wake hooks on this
82    /// OQueue.
83    pub(super) fn register(
84        oqueue: Arc<dyn UntypedOQueueImplementation>,
85        event: WakeEvent,
86        hook: Box<dyn Fn() + Send>,
87    ) -> Self {
88        let key = oqueue.add_wake_hook(event, hook);
89        Self {
90            oqueue: Arc::downgrade(&oqueue),
91            event,
92            key,
93        }
94    }
95}
96
97impl Drop for WakeHook {
98    fn drop(&mut self) {
99        // A dropped OQueue took its hooks with it, so there is nothing to unregister.
100        if let Some(oqueue) = self.oqueue.upgrade() {
101            oqueue.remove_wake_hook(self.event, self.key);
102        }
103    }
104}
105
106/// The registered [`WakeHook`]s of an OQueue, one list per [`WakeEvent`] so that a wake-up only
107/// visits the hooks waiting on that event.
108#[derive(Default)]
109struct WakeHooks {
110    produced: SlotMap<WakeHookKey, Box<dyn Fn() + Send>>,
111    space_freed: SlotMap<WakeHookKey, Box<dyn Fn() + Send>>,
112}
113
114impl WakeHooks {
115    /// Returns the hooks waiting on `event`.
116    fn for_event(&mut self, event: WakeEvent) -> &mut SlotMap<WakeHookKey, Box<dyn Fn() + Send>> {
117        match event {
118            WakeEvent::Produced => &mut self.produced,
119            WakeEvent::SpaceFreed => &mut self.space_freed,
120        }
121    }
122}

124/// The underlying implementation of OQueues. This is used within an `Arc` to reference the OQueue
125/// internally. Externally, code will use [`super::OQueueRef`] and similar.
126///
127/// ## Adding implementations
128///
129/// This does not need to be a single unified implementation. This could be replaced by an enum of
130/// different implementations and then have operations dispatched appropriately. This may be needed
131/// to support locking and lock-free implementation. It could even provide a way to use a
132/// hypothetical `dyn OQueueDynImplementation<T>` backend; however, `OQueueDynImplementation` would
133/// require quite a few type-erased unsafe operations.
134pub(crate) struct OQueueImplementation<D: ElementDescriptor> {
135    // TODO(arthurp): A number of methods perform allocation while this lock is held. Do we actually
136    // want to disable IRQs?
137    inner: SpinLock<OQueueInner<D>, LocalIrqDisabled>,
138    /// The size to use for the consumer and strong-observer ring-buffers.
139    len: usize,
140    supports_consume: bool,
141    path: Option<Path>,
142    /// Wait queue for observers. Awoken when there are elements to observe.
143    observe_wait_queue: WaitQueue,
144    /// Wait queue for consumers. Awoken when there are elements to consume. This is
145    /// [switched-to](`WaitQueue::switch_to_one`), so it is separate from `observe_wait_queue`.
146    pub(super) consume_wait_queue: WaitQueue,
147    /// The last task to attach to this OQueue as a consumer. This is weak so as not to keep the
148    /// [`Task`] alive if no other consumer appears for a long period of time.
149    last_consumer: SpinLock<Option<Weak<Task>>, LocalIrqDisabled>,
150    /// Wait queue for producers. Awoken when there is space for more elements.
151    produce_wait_queue: WaitQueue,
152    /// Hooks run alongside the waiter wake-ups, one list per event.
153    wake_hooks: SpinLock<WakeHooks, LocalIrqDisabled>,
154    /// The number of full buffers that block producers: the consumer ring and non-revocable
155    /// strong-observer rings. A produce does not block iff this is 0.
156    num_full_buffers: AtomicUsize,
157}

159impl<D: ElementDescriptor + 'static> OQueueImplementation<D> {
160    /// Create a new OQueue.
161    ///
162    /// * `len` is the ring buffer length used for consumers and strong-observers.
163    /// * `supports_consume` specifies the attachment it allows later.
164    /// * `paths` are the paths associated with this OQueue; pass an empty `Vec` for anonymous
165    ///   queues.
166    pub(crate) fn new(mut len: usize, supports_consume: bool, path: Option<Path>) -> Self {
167        if len < 2 {
168            warn!(
169                "Creating an OQueue with length {len} is automatically increased to 2. Ring buffers smaller than 2 are not supported."
170            );
171            len = 2;
172        }
173        Self {
174            inner: SpinLock::new(OQueueInner {
175                consumer_ring_buffer: Default::default(),
176                n_consumers: 0,
177                observer_ring_buffers: Default::default(),
178                inline_strong_observers: Default::default(),
179                inline_consumer: None,
180            }),
181            len,
182            supports_consume,
183            path,
184            observe_wait_queue: WaitQueue::new(),
185            consume_wait_queue: WaitQueue::new(),
186            last_consumer: SpinLock::new(None),
187            produce_wait_queue: WaitQueue::new(),
188            wake_hooks: SpinLock::new(WakeHooks::default()),
189            num_full_buffers: AtomicUsize::new(0),
190        }
191    }
192
193    /// Return the path associated with this OQueue, or `None` for anonymous queues.
194    pub(super) fn path(&self) -> Option<&Path> {
195        self.path.as_ref()
196    }

198    /// Detach a consumer. This will free the consumer ring buffer if there are no consumers left.
199    pub(super) fn detach_consumer(self: &Arc<Self>) {
200        let mut inner = self.inner.lock();
201        inner.n_consumers -= 1;
202        let mut freed = false;
203        if inner.n_consumers == 0 {
204            let ring_buffer = inner.consumer_ring_buffer.take();
205            freed = self.unfull(ring_buffer.is_some_and(|rb| !rb.can_produce()));
206        }
207        drop(inner);
208        if freed {
209            self.wake_space_freed();
210        }
211    }
212
213    pub(super) fn detach_inline_consumer(&self) {
214        let mut inner = self.inner.lock();
215        inner.inline_consumer = None;
216    }

218    /// Create a new ring buffer for values of type `U` and attach it to this `self` as an
219    /// observation ring buffer.
220    fn new_observation_ring_buffer<U>(
221        self: &Arc<Self>,
222        query: ObservationQuery<D, U>,
223        len: usize,
224        is_strong: bool,
225        revocable: bool,
226    ) -> Result<ObserverKey, OQueueError>
227    where
228        U: Copy + Send + 'static,
229    {
230        let mut ring_buffer = ObservationRingBuffer::new::<U>(query, len)
231            .map_err(|_| ResourceUnavailableSnafu.build())?;
232        ring_buffer.revocable = revocable;
233
234        if is_strong {
235            let id = ring_buffer
236                .ring_buffer
237                .as_mut()
238                .expect("newly created observation ring buffer is present")
239                .new_strong_reader();
240            // We don't reuse ring buffers, so for now this is always 0.
241            assert_eq!(id, 0);
242        }
243
244        let mut inner = self.inner.lock();
245        let observer_key = inner.observer_ring_buffers.insert(ring_buffer);
246        Ok(observer_key)
247    }

290    /// Attach a strong observer to the OQueue by calling a strong observer function with each
291    /// value.
292    ///
293    /// This cannot be combined with the below query based one because the "identity query" cannot
294    /// exist for `T: !Copy`.
295    pub(super) fn attach_inline_strong_observer(
296        self: &Arc<Self>,
297        f: impl for<'a> Fn(&'a D::Element<'a>) + Send + 'static,
298    ) -> Result<InlineStrongObserver, OQueueError> {
299        let mut inner = self.inner.lock();
300        let key = inner.inline_strong_observers.insert(Box::new(f));
301        Ok(InlineStrongObserver {
302            oqueue: self.clone(),
303            inline_observer_id: key,
304            _phantom: PhantomData,
305        })

308    /// Attach a strong observer to the OQueue by calling a strong observer function with each
309    /// value. The value is passed through a query. This is used for converting a strong-observer
310    /// attachment into an inline.
311    ///
312    /// This removes the ring buffer associated with `observer_id` so it is no longer valid (and may
313    /// be reallocated). This should *only* be called from something which take ownership of
314    /// observer_id and converts it into the new form.
315    ///
316    /// This returns a new ID associated with the registered inline attachment. This will not be the
317    /// same as the argument.
318    pub(super) fn convert_strong_observer_to_inline<U: Send + Copy + 'static>(
319        this: &dyn UntypedOQueueImplementation,
320        observer_id: ObserverKey,
321        f: Box<dyn Fn(&U) + Send + 'static>,
322    ) -> Result<InlineObserverKey, OQueueError> {
323        let this: &Self = (this as &dyn Any).downcast_ref().unwrap();
324        let mut inner = this.inner.lock();
325
326        // Move the ring buffer out
327        let mut observation_ring_buffer = inner
328            .observer_ring_buffers
329            .remove(observer_id)
330            .expect("tried to attach inline strong observer for observer_id which did not exist");
331        let freed = this.unfull(observation_ring_buffer.blocks_producers());
332        let query = (observation_ring_buffer.query as Box<dyn Any>)
333            .downcast::<ObservationQuery<D, U>>()
334            .expect("tried to attach inline strong observer with a different type from original attachment");
335
336        // Drain the ring buffer into the function. A revoked observer has no buffer, so there is
337        // nothing to drain.
338        // TODO: Use a real head_id when we share ring buffers.
339        if let Some(ring_buffer) = &mut observation_ring_buffer.ring_buffer {
340            let head_id = 0;
341            while let Some(v) = ring_buffer.try_strong_observe::<U>(head_id) {
342                f(&v)
343            }
344        }
345
346        // Register the actual handler. The type ascription (rather than letting `insert` infer
347        // it) is required for rustc to unify the closure's argument lifetime with `MD::Message`'s
348        // own GAT lifetime, as `for<'a> Fn(&'a MD::Element<'a>)` requires.
349        #[expect(clippy::type_complexity)]
350        let handler: Box<dyn for<'a> Fn(&'a D::Element<'a>) + Send> = Box::new(move |v| {
351            if let Some(v) = query.call(v) {
352                f(&v)
353            }
354        });
355        let key = inner.inline_strong_observers.insert(handler);
356        drop(inner);
357        if freed {
358            this.wake_space_freed();
359        }
360
361        Ok(key)
362    }

392impl<D: ElementDescriptor + 'static> OQueueImplementation<D> {
393    /// True if try_produce into self is *expected* to succeed. It may still fail if there is a
394    /// concurrent producer.
395    pub(super) fn can_produce(&self) -> bool {
396        self.num_full_buffers.load(Ordering::Acquire) == 0
397    }

399    /// Produce a value by reference, blocking until it completes.
400    pub(super) fn produce_ref<'a>(&self, v: &'a D::Element<'a>) {
401        self.produce_wait_queue.wait_until(|| {
402            if self.try_produce_ref(v) {
403                Some(())
404            } else {
405                None
406            }
407        })
408    }
409
410    /// Attempt to produce a value by reference. This executes all the queries on `v` and places the
411    /// query results into the appropriate observation ring buffers.
412    pub(super) fn try_produce_ref<'a>(&self, v: &'a D::Element<'a>) -> bool {
413        let mut inner = self.inner.lock();
414        assert!(inner.consumer_ring_buffer.is_none());
415        // A slow observer must never block a producer, so drop any whose ring is full. As this is
416        // an observation OQueue (no consumers), this also means production always succeeds.
417        inner.revoke_full_observers();
418        debug_assert_eq!(
419            inner.can_produce(),
420            self.can_produce(),
421            "num_full_buffers is out of sync with the ring buffers"
422        );
423        if inner.can_produce() {
424            // Every counted ring has room here, so a counted ring that is full afterwards has just
425            // become full.
426            let mut newly_full = 0;
427            for ObservationRingBuffer {
428                query,
429                ring_buffer,
430                revocable,
431                ..
432            } in inner.observer_ring_buffers.values_mut()
433            {
434                if let Some(ring_buffer) = ring_buffer {
435                    query.call_into(v, ring_buffer);
436                    if !*revocable && !ring_buffer.can_produce() {
437                        newly_full += 1;
438                    }
439                }
440            }
441            for f in inner.inline_strong_observers.values() {
442                f(v)
443            }
444            self.num_full_buffers
445                .fetch_add(newly_full, Ordering::Relaxed);
446            drop(inner);
447
448            self.wake_after_produce();
449
450            true
451        } else {
452            false
453        }
454    }
455
456    /// Wake observers and consumers after a successful produce.
457    fn wake_after_produce(&self) {
458        self.observe_wait_queue.wake_all();
459        self.run_wake_hooks(WakeEvent::Produced);
460
461        // Try to wake a consumer which is blocked on this OQueue. If no OQueue was awoken and there
462        // was some task which consumed on this OQueue before, wake that.
463        if !self.consume_wait_queue.switch_to_one() {
464            let mut last_consumer = self.last_consumer.lock();
465            if let Some(task) = last_consumer.as_ref().and_then(Weak::upgrade) {
466                drop(last_consumer);
467
468                switch_to_target(task);
469            } else {
470                *last_consumer = None;
471            }
472        }
473    }
474
475    /// Updates `num_full_buffers` after a counted buffer changed or was removed, where `was_full`
476    /// says whether that buffer had been full and now no longer is. Call this with `inner` held.
477    ///
478    /// Returns whether no full buffer is left, in which case the caller must call
479    /// [`Self::wake_space_freed`] once `inner` is dropped.
480    fn unfull(&self, was_full: bool) -> bool {
481        if !was_full {
482            return false;
483        }
484        let previous = self.num_full_buffers.fetch_sub(1, Ordering::Release);
485        previous == 1
486    }
487
488    /// Wakes blocked producers and runs the [`WakeEvent::SpaceFreed`] hooks.
489    fn wake_space_freed(&self) {
490        self.produce_wait_queue.wake_all();
491        self.run_wake_hooks(WakeEvent::SpaceFreed);
492    }
493
494    /// Runs the hooks registered for `event`.
495    fn run_wake_hooks(&self, event: WakeEvent) {
496        for hook in self.wake_hooks.lock().for_event(event).values() {
497            hook();
498        }
499    }

513    /// True if at least one strong or weak observer is currently attached.
514    pub(super) fn has_observers(&self) -> bool {

516    }

519// This impl block is specific to `LifetimelessElementDescriptor<T>` (rather than generic over any
520// `D: ElementDescriptor`) because producing/consuming by value requires an owned (`'static`)
521// message.
522impl<T: Send + 'static> OQueueImplementation<LifetimelessElementDescriptor<T>> {
523    /// Produce into the OQueue. Blocking until space is available
524    pub(super) fn produce(&self, v: T) {
525        let mut v = v;
526        loop {
527            if let Err(ret) = self.try_produce(v) {
528                v = ret;
529                // TODO: PERFORMANCE: Using can_produce here and try_produce above requires 2
530                // acquires.
531                self.produce_wait_queue
532                    .wait_until(|| if self.can_produce() { Some(()) } else { None })
533            } else {
534                return;
535            }
536        }
537    }
538
539    /// Attempt to produce into the OQueue. This either succeeds or returns the value that would
540    /// have been produced.
541    pub(super) fn try_produce(&self, v: T) -> Result<(), T> {
542        let mut inner = self.inner.lock();
543        // A slow observer must never block a producer, so drop any whose ring is full.
544        inner.revoke_full_observers();
545        debug_assert_eq!(
546            inner.can_produce(),
547            self.can_produce(),
548            "num_full_buffers is out of sync with the ring buffers"
549        );
550        if inner.can_produce() {
551            // Every counted ring has room here, so a counted ring that is full afterwards has just
552            // become full.
553            let mut newly_full = 0;
554            for ObservationRingBuffer {
555                query,
556                ring_buffer,
557                revocable,
558                ..
559            } in inner.observer_ring_buffers.values_mut()
560            {
561                if let Some(ring_buffer) = ring_buffer {
562                    query.call_into(&v, ring_buffer);
563                    if !*revocable && !ring_buffer.can_produce() {
564                        newly_full += 1;
565                    }
566                }
567            }
568            for f in inner.inline_strong_observers.values() {
569                f(&v)
570            }
571            assert!(inner.consumer_ring_buffer.is_none() || inner.inline_consumer.is_none());
572            if let Some(ring_buffer) = &mut inner.consumer_ring_buffer {
573                let v = ring_buffer.try_produce(v);
574                assert!(v.is_none());
575                if !ring_buffer.can_produce() {
576                    newly_full += 1;
577                }
578            } else if let Some(consumer) = &inner.inline_consumer {
579                consumer(v);
580            } else {
581                // No consumer attached, so `v` has nowhere to land; warn instead of dropping it
582                // silently.
583                warn!(
584                    "Produced a value into a ConsumableOQueue with no consumer attached; the value is dropped"
585                );
586            }
587            self.num_full_buffers
588                .fetch_add(newly_full, Ordering::Relaxed);
589            drop(inner);
590
591            self.wake_after_produce();
592
593            Ok(())
594        } else {
595            Err(v)
596        }
597    }
598
599    /// Consume a value, blocking until it completes.
600    pub(super) fn consume(&self) -> T {
601        self.consume_wait_queue.wait_until(|| self.try_consume())
602    }
603
604    pub(super) fn can_consume(&self) -> bool {
605        let inner = self.inner.lock();
606        inner
607            .consumer_ring_buffer
608            .as_ref()
609            .expect("consume not supported")
610            .can_get_for_head(0)
611    }
612
613    /// Attempt to consume a value from the consumer ring buffer, taking ownership of the value.
614    pub(super) fn try_consume(&self) -> Option<T> {
615        let mut inner = self.inner.lock();
616        let ring_buffer = inner
617            .consumer_ring_buffer
618            .as_mut()
619            .expect("consume not supported");
620        let was_full = !ring_buffer.can_produce();
621        // SAFETY: The consumer ring buffer is never used for observation.
622        let ret = unsafe { ring_buffer.try_consume() };
623        let freed = ret.is_some() && self.unfull(was_full);
624        drop(inner);
625        *self.last_consumer.lock().deref_mut() =
626            Task::current().map(|t| Arc::downgrade(&t.cloned()));
627        if freed {
628            self.wake_space_freed();
629        }
630        ret
631    }

633    /// Attach a by-value producer to the OQueue if this supports consumers.
634    pub(super) fn attach_value_producer(
635        self: &Arc<Self>,
636    ) -> Result<super::ValueProducer<T>, OQueueError> {
637        if !self.supports_consume {
638            return super::UnsupportedSnafu.fail();
639        }
640        Ok(super::ValueProducer {
641            oqueue: self.clone(),
642        })
643    }
644
645    /// Attach a consumer to the OQueue if this does not support consumers.
646    pub(super) fn attach_consumer(self: &Arc<Self>) -> Result<super::Consumer<T>, OQueueError> {
647        if !self.supports_consume {
648            return super::UnsupportedSnafu.fail();
649        }
650        let mut inner = self.inner.lock();
651        if inner.consumer_ring_buffer.is_none() {
652            let mut ring_buffer =
653                RingBuffer::new::<T>(self.len).map_err(|_| ResourceUnavailableSnafu.build())?;
654            let id = ring_buffer.new_strong_reader();
655            assert_eq!(id, 0);
656            inner.consumer_ring_buffer = Some(ring_buffer);
657        }
658        inner.n_consumers += 1;
659        drop(inner);
660        Ok(super::Consumer {
661            oqueue: self.clone(),
662            _phantom: PhantomData,
663        })
664    }
665
666    pub(super) fn attach_inline_consumer(
667        self: &Arc<Self>,
668        f: impl Fn(T) + Send + 'static,
669    ) -> Result<(), OQueueError> {
670        let inline_consumer: Box<dyn Fn(T) + Send + 'static> = Box::new(f);
671        let mut inner = self.inner.lock();
672        ensure!(inner.inline_consumer.is_none(), ResourceUnavailableSnafu);
673        inner.inline_consumer = Some(inline_consumer);
674        Ok(())
675    }
676
677    /// True if at least one consumer is currently attached to this OQueue.
678    pub(super) fn has_consumers(&self) -> bool {
679        let inner = self.inner.lock();
680        inner.n_consumers > 0 || inner.inline_consumer.is_some()
681    }
682}

684/// The part of [`OQueueImplementation`] which is protected by a lock.
685struct OQueueInner<D: ElementDescriptor> {
686    /// The ring buffer used for consumers.
687    consumer_ring_buffer: Option<RingBuffer>,
688    /// The number of attached consumers. This is required to detect when the `consumer_ring_buffer`
689    /// can be discarded.
690    n_consumers: usize,
691    /// The ring buffers used for observers. If a ring buffer is no longer needed, due to a
692    /// detachment, its slot will be set to `None` and ignored. `None` slots are reused for later
693    /// attachments.
694    observer_ring_buffers: SlotMap<ObserverKey, ObservationRingBuffer<D>>,
695    // TODO: PERFORMANCE: There will be a bunch of cases where there is *only* a consumer ring
696    // buffer, or *only* a single observer ring. It would be nice if either could be inlined. As of
697    // now, the consumer is always inlined even when it's not in use, and the observer never is.
698    // TODO: PERFORMANCE: This needs a way to share ring buffers when multiple consumers use the
699    // same query.
700    /// Inline strong observers of the message type. These will be called during production.
701    #[expect(clippy::type_complexity)]
702    inline_strong_observers:
703        SlotMap<InlineObserverKey, Box<dyn for<'a> Fn(&'a D::Element<'a>) + Send>>,
704    /// The inline consumer if there is one.
705    ///
706    /// This is only ever populated when `MD = LifetimelessElementDescriptor<T>` (see
707    /// `OQueueImplementation<LifetimelessElementDescriptor<T>>`'s by-value impl block), since only an
708    /// owned, `'static` message can be consumed by value.
709    inline_consumer: Option<Box<dyn Fn(D::Element<'static>) + Send>>,
710}
711
712impl<D: ElementDescriptor> OQueueInner<D> {
713    /// True if a produce can succeed without blocking.
714    fn can_produce(&self) -> bool {
715        // Revoked observers are never produced to, and a full revocable observer is revoked by the
716        // produce rather than blocking it.
717        self.observer_ring_buffers
718            .values()
719            .all(|r| r.revocable || r.ring_buffer.as_ref().is_none_or(|rb| rb.can_produce()))
720            && self.consumer_ring_buffer.iter().all(|r| r.can_produce())
721    }
722
723    /// Revoke every *revocable* strong observer whose ring buffer is full (a.k.a !can_produce()).
724    ///
725    /// Revoking frees the ring buffer, so a repeatedly-revoked-and-reattached observer cannot grow
726    /// memory without bound.
727    fn revoke_full_observers(&mut self) {
728        for observer in self.observer_ring_buffers.values_mut() {
729            if observer.revocable
730                && observer
731                    .ring_buffer
732                    .as_ref()
733                    .is_some_and(|rb| !rb.can_produce())
734            {
735                observer.ring_buffer = None;
736            }
737        }
738    }
739}

741/// A partially type erased [`ObservationQuery`] with `U` (the query output type) erased, but `MD`
742/// (the message descriptor) retained statically. This is used in the OQueue machinery to dispatch
743/// to the query and produce it's result into a [`RingBuffer`].
744trait ErasedObservationQuery<D: ElementDescriptor>: Send + Any {
745    /// Call the query and then produce the value into the provided ring buffer. The ring buffer
746    /// must have a type matching the result of the query (`U` below). Returns false if the value
747    /// could not be produced into `ring_buffer`.
748    fn call_into<'a>(&self, v: &'a D::Element<'a>, ring_buffer: &mut RingBuffer) -> bool;
749}
750
751impl<D: ElementDescriptor + 'static, U: Send + 'static> ErasedObservationQuery<D>
752    for ObservationQuery<D, U>
753{
754    fn call_into<'a>(&self, v: &'a D::Element<'a>, ring_buffer: &mut RingBuffer) -> bool {
755        if let Some(v) = self.call(v) {
756            ring_buffer.try_produce(v).is_none()
757        } else {
758            true
759        }
760    }
761}

763/// A wrapper over [`RingBuffer`] which provides the information needed to perform produce and
764/// observe operations without exposing the type of the observed value (`U`).
765struct ObservationRingBuffer<D: ElementDescriptor> {
766    // TODO: PERFORMANCE: Replace the dyn ref and optimize based on the known structure of
767    // ObservationQuery. This should remove a level of pointer indirection.
768    /// The query to filter the message and place the result into the ring buffer.
769    query: Box<dyn ErasedObservationQuery<D>>,
770    /// Function to extract a value from the ring buffer with strong observer semantics and place it
771    /// in the memory pointed to by `dest`. This indirection is needed since the type `U` in the
772    /// ring buffer is not statically known.
773    ///
774    /// ## Safety
775    ///
776    /// `dest` must point to memory appropriate for the type `U` with which `ring_buffer` was
777    /// constructed.
778    try_strong_observe_into: unsafe fn(ring_buffer: &mut RingBuffer, dest: *mut ()) -> bool,
779    /// Function to extract a value from the ring buffer and place it in the memory pointed to by
780    /// `dest`. This indirection is needed since the type `U` in the ring buffer is not statically
781    /// known.
782    ///
783    /// ## Safety
784    ///
785    /// `dest` must point to memory appropriate for the type `U` with which `ring_buffer` was
786    /// constructed.
787    weak_observe_into: unsafe fn(ring_buffer: &mut RingBuffer, Cursor, dest: *mut ()) -> bool,
788    /// The actual ring buffer storing the data, or `None` once this observer has been revoked.
789    ///
790    /// A revoked observer no longer gates production and its next observe returns
791    /// [`OQueueError::Revoked`](super::OQueueError::Revoked). Revoking frees the (potentially large)
792    /// ring buffer so that an observer which is repeatedly revoked and reattached cannot grow memory
793    /// without bound. The lighter-weight `query` is retained so the slot can still report the
794    /// revoked state and be converted to an inline observer.
795    ring_buffer: Option<RingBuffer>,
796    /// Whether this observer may be revoked (dropped) when its ring fills, instead of applying
797    /// backpressure to producers. Set for userspace-facing observers where kernel liveness must
798    /// not depend on the reader keeping up.
799    revocable: bool,
800}
801
802impl<D: ElementDescriptor> ObservationRingBuffer<D> {
803    /// True if this ring is full and blocks producers, i.e. it counts in `num_full_buffers`.
804    fn blocks_producers(&self) -> bool {
805        !self.revocable
806            && self
807                .ring_buffer
808                .as_ref()
809                .is_some_and(|rb| !rb.can_produce())
810    }
811}

813impl<D: ElementDescriptor + 'static> ObservationRingBuffer<D> {
814    /// Create a new ring buffer for a message descriptor `MD` and an observed type of `U`. This
815    /// will be used to store values of type `U` for observation.
816    fn new<U: Copy + Send + 'static>(
817        query: ObservationQuery<D, U>,
818        len: usize,
819    ) -> Result<Self, AllocError> {
820        let ring_buffer = RingBuffer::new::<U>(len)?;
821
822        unsafe fn try_strong_observe_into<U: Copy + Send + 'static>(
823            r: &mut RingBuffer,
824            d: *mut (),
825        ) -> bool {
826            // TODO: Use a real head_id when we share ring buffers.
827            let head_id = 0;
828            if let Some(v) = r.try_strong_observe::<U>(head_id) {
829                // SAFETY: The safety requirements of ObservationRingBuffer::try_strong_observe_into
830                // specify that d must be appropriate for a value of type U.
831                unsafe {
832                    (d as *mut U).write(v);
833                }
834                true
835            } else {
836                false
837            }
838        }
839
840        unsafe fn weak_observe_into<U: Copy + Send + 'static>(
841            r: &mut RingBuffer,
842            c: Cursor,
843            d: *mut (),
844        ) -> bool {
845            if let Some(v) = r.try_weak_observe::<U>(c) {
846                // SAFETY: The safety requirements of ObservationRingBuffer::weak_observe_into
847                // specify that d must be appropriate for a value of type U.
848                unsafe {
849                    (d as *mut U).write(v);
850                }
851                true
852            } else {
853                false
854            }
855        }
856
857        Ok(ObservationRingBuffer {
858            query: Box::new(query),
859            try_strong_observe_into: try_strong_observe_into::<U>,
860            weak_observe_into: weak_observe_into::<U>,
861            ring_buffer: Some(ring_buffer),
862            revocable: false,
863        })
864    }
865}

867/// A dyn-compatible trait to type erase `T` from [`OQueueImplementation<T>`]. The operations here
868/// conceptually take the observed type `U` as a type parameter, but this is not possible in a
869/// dyn-compatible trait.
870///
871/// TODO(arthurp): PERFORMANCE: These methods could be lifted out of the [`OQueueImplementation`]
872/// and written directly on the untyped representation, since they are not actually going to involve
873/// `T`. This would avoid an indirect function call and allow inlining. This would require using
874/// `repr(C)` and direct manipulation of the representation.
875pub(super) trait UntypedOQueueImplementation: Sync + Send + Any {
876    /// Release any resources held by observer with the given ID. Using that `observer_id` again may
877    /// produce UB.
878    fn detach_observer(&self, observer_id: ObserverKey);
879
880    /// Release any resources held by inline observer with the given key.
881    fn detach_inline_strong_observer(&self, inline_observer_id: InlineObserverKey);
882
883    fn can_strong_observe(&self, observer_id: ObserverKey) -> bool;
884
885    /// Enqueue a waker for an observer into the wait queue. This is used to implement `Blocker` for
886    /// `StrongObserver`.
887    fn enqueue_observe_waker(&self, waker: &Arc<crate::sync::Waker>) -> WakerKey;
888
889    /// Remove a waker for an observer from the wait queue. This is used to implement `Blocker` for
890    /// `StrongObserver`.
891    fn remove_observe_waker(&self, key: WakerKey);
892
893    /// Copy the next value available to the specified observer into `dest` if it is available. This
894    /// returns `Ok(true)` if the value was copied, `Ok(false)` if there was not value available
895    /// yet, and an error if some other failure happened.
896    ///
897    /// ## Safety
898    ///
899    /// The caller must guarantee that:
900    ///
901    /// * `dest` points to memory which is of the correct size and alignment for a value of the type
902    ///   with the provided `type_id`.
903    /// * `type_id` matches the type of buffer created in `attach_strong_observer` when it returned
904    ///   this `observer_id`.
905    ///
906    /// After this returns, `dest` will be initialed iff this returns `Ok(true)`.
907    ///
908    /// NOTE: `type_id` may not actually be used in release mode (which is why it must match the
909    /// observer_id). However, in debug mode it may be checked to catch bugs in the handles.
910    unsafe fn try_strong_observe_into(
911        &self,
912        observer_id: ObserverKey,
913        type_id: TypeId,
914        dest: *mut (),
915    ) -> Result<bool, OQueueError>;
916
917    /// A blocking version of [`UntypedOQueueInner::try_strong_observe_into`]. All the safety
918    /// requirements on that apply here, except that this will never return `Ok` without filling
919    /// `dest`, since it blocks waiting for a value.
920    unsafe fn strong_observe_into(
921        &self,
922        observer_id: ObserverKey,
923        type_id: TypeId,
924        dest: *mut (),
925    ) -> Result<(), OQueueError>;
926
927    /// Copy the value at index `cursor` into `dest` if it is available. This returns `Ok(true)` if
928    /// the value was copied, `Ok(false)` if there was not value available, and an error if some
929    /// other failure happened.
930    ///
931    /// ## Safety
932    ///
933    /// The caller must guarantee that:
934    ///
935    /// * `dest` points to memory which is of the correct size and alignment for a value of the type
936    ///   with the provided `type_id`.
937    /// * `type_id` matches the type of buffer created in `attach_weak_observer` when it returned
938    ///   this `observer_id`.
939    ///
940    /// After this returns, `dest` will be initialed iff this returns `Ok(true)`.
941    ///
942    /// NOTE: `type_id` may not actually be used in release mode (which is why it must match the
943    /// observer_id). However, in debug mode it may be checked to catch bugs in the handles.
944    unsafe fn weak_observe_into(
945        &self,
946        observer_id: ObserverKey,
947        type_id: TypeId,
948        cursor: Cursor,
949        dest: *mut (),
950    ) -> Result<bool, OQueueError>;
951
952    /// Wait until new values are available for the specific observer.
953    fn wait(&self, observer_id: ObserverKey, cursor: Cursor);
954
955    /// Get the most recent cursor which will return a value.
956    fn newest_cursor(&self, observer_id: ObserverKey) -> Cursor;
957
958    /// Get the oldest cursor which will return a value.
959    fn oldest_cursor(&self, observer_id: ObserverKey) -> Cursor;
960
961    /// Register a hook to run where the OQueue wakes its waiters. This is used to implement
962    /// [`WakeHook`].
963    fn add_wake_hook(&self, event: WakeEvent, hook: Box<dyn Fn() + Send>) -> WakeHookKey;
964
965    /// Unregister a hook added by [`Self::add_wake_hook`] for `event`.
966    fn remove_wake_hook(&self, event: WakeEvent, key: WakeHookKey);
967}

971impl<D: ElementDescriptor + 'static> UntypedOQueueImplementation for OQueueImplementation<D> {
972    fn detach_observer(&self, observer_id: ObserverKey) {
973        let mut inner = self.inner.lock();
974        let freed = inner
975            .observer_ring_buffers
976            .remove(observer_id)
977            .is_some_and(|r| self.unfull(r.blocks_producers()));
978        drop(inner);
979        if freed {
980            self.wake_space_freed();
981        }
982    }
983
984    fn detach_inline_strong_observer(&self, inline_observer_id: InlineObserverKey) {
985        let mut inner = self.inner.lock();
986        inner.inline_strong_observers.remove(inline_observer_id);
987    }
988
989    unsafe fn try_strong_observe_into(
990        &self,
991        observer_id: ObserverKey,
992        _type_id: TypeId,
993        dest: *mut (),
994    ) -> Result<bool, OQueueError> {
995        let mut inner = self.inner.lock();
996        let ObservationRingBuffer {
997            try_strong_observe_into,
998            ring_buffer,
999            revocable,
1000            ..
1001        } = inner
1002            .observer_ring_buffers
1003            .get_mut(observer_id)
1004            .expect("should only be called with an id returned from new_observation_ring_buffer");
1005
1006        // The observer was dropped because it fell behind and its buffer was freed; report the
1007        // stream as revoked.
1008        let Some(ring_buffer) = ring_buffer else {
1009            return super::RevokedSnafu.fail();
1010        };
1011
1012        let was_full = !*revocable && !ring_buffer.can_produce();
1013        // SAFETY: weak_observe_into and ring_buffer where created together with the same type U.
1014        let ret = unsafe { try_strong_observe_into(ring_buffer, dest) };
1015        let freed = ret && self.unfull(was_full);
1016        drop(inner);
1017        if freed {
1018            self.wake_space_freed();
1019        }
1020        Ok(ret)
1021    }
1022
1023    unsafe fn strong_observe_into(
1024        &self,
1025        observer_id: ObserverKey,
1026        _type_id: TypeId,
1027        dest: *mut (),
1028    ) -> Result<(), OQueueError> {
1029        self.observe_wait_queue.wait_until(|| {
1030            // SAFETY: The requirements of try_strong_observe_into are the same as this function.
1031            let r = unsafe { self.try_strong_observe_into(observer_id, _type_id, dest) };
1032            if let Ok(false) = r { None } else { Some(r) }
1033        })?;
1034        Ok(())
1035    }
1036
1037    unsafe fn weak_observe_into(
1038        &self,
1039        observer_id: ObserverKey,
1040        _type_id: TypeId,
1041        cursor: Cursor,
1042        dest: *mut (),
1043    ) -> Result<bool, OQueueError> {
1044        let mut inner = self.inner.lock();
1045        let ObservationRingBuffer {
1046            weak_observe_into,
1047            ring_buffer,
1048            ..
1049        } = inner
1050            .observer_ring_buffers
1051            .get_mut(observer_id)
1052            .expect("should only be called with an id returned from new_observation_ring_buffer");
1053        // Weak observers are never revocable, so their ring buffer is never freed.
1054        let ring_buffer = ring_buffer
1055            .as_mut()
1056            .expect("weak observers are never revoked");
1057        // SAFETY: weak_observe_into and ring_buffer where created together with the same type U.
1058        Ok(unsafe { weak_observe_into(ring_buffer, cursor, dest) })
1059    }
1060
1061    fn wait(&self, observer_id: ObserverKey, cursor: Cursor) {
1062        self.observe_wait_queue.wait_until(|| {
1063            if self.newest_cursor(observer_id) > cursor {
1064                Some(())
1065            } else {
1066                None
1067            }
1068        })
1069    }
1070
1071    fn newest_cursor(&self, observer_id: ObserverKey) -> Cursor {
1072        let mut inner = self.inner.lock();
1073        let ObservationRingBuffer { ring_buffer, .. } = inner
1074            .observer_ring_buffers
1075            .get_mut(observer_id)
1076            .expect("should only be called with an id returned from new_observation_ring_buffer");
1077        // Weak observers are never revocable, so their ring buffer is never freed.
1078        ring_buffer
1079            .as_ref()
1080            .expect("weak observers are never revoked")
1081            .newest_cursor()
1082    }
1083
1084    fn oldest_cursor(&self, observer_id: ObserverKey) -> Cursor {
1085        let mut inner = self.inner.lock();
1086        let ObservationRingBuffer { ring_buffer, .. } = inner
1087            .observer_ring_buffers
1088            .get_mut(observer_id)
1089            .expect("should only be called with an id returned from new_observation_ring_buffer");
1090        // Weak observers are never revocable, so their ring buffer is never freed.
1091        ring_buffer
1092            .as_ref()
1093            .expect("weak observers are never revoked")
1094            .oldest_cursor()
1095    }
1096
1097    fn can_strong_observe(&self, observer_id: ObserverKey) -> bool {
1098        let mut inner = self.inner.lock();
1099        let ObservationRingBuffer { ring_buffer, .. } = inner
1100            .observer_ring_buffers
1101            .get_mut(observer_id)
1102            .expect("should only be called with an id returned from new_observation_ring_buffer");
1103        let head_id = 0;
1104        // A revoked observer has no buffer; report it as ready so a blocked observer wakes and
1105        // observes the revoked state (an `Err`) rather than blocking forever.
1106        ring_buffer
1107            .as_ref()
1108            .is_none_or(|rb| rb.can_get_for_head(head_id))
1109    }
1110
1111    fn enqueue_observe_waker(&self, waker: &Arc<crate::sync::Waker>) -> WakerKey {
1112        self.observe_wait_queue.enqueue(waker.clone())
1113    }
1114
1115    fn remove_observe_waker(&self, key: WakerKey) {
1116        self.observe_wait_queue.remove(key);
1117    }
1118
1119    fn add_wake_hook(&self, event: WakeEvent, hook: Box<dyn Fn() + Send>) -> WakeHookKey {
1120        self.wake_hooks.lock().for_event(event).insert(hook)
1121    }
1122
1123    fn remove_wake_hook(&self, event: WakeEvent, key: WakeHookKey) {
1124        self.wake_hooks.lock().for_event(event).remove(key);
1125    }
