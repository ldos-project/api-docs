79    /// Returns the formatted device name.
80    ///
81    /// The device name starts at "vda". The 26th device is "vdz" and the 27th is "vdaa".
82    /// The last one for two lettered suffix is "vdzz" which is followed by "vdaaa".
83    fn formatted_device_name(mut index: u32) -> String {
84        const VIRTIO_DISK_PREFIX: &str = "vd";
85
86        let mut suffix = Vec::new();
87        loop {
88            suffix.push((b'a' + (index % 26) as u8) as char);
89            index /= 26;
90            if index == 0 {
91                break;
92            }
93            index -= 1;
94        }
95        suffix.reverse();
96        let mut name = String::from(VIRTIO_DISK_PREFIX);
97        name.extend(suffix);
98        name
99    }

101    /// Creates a new VirtIO-Block driver and registers it.
102    pub(crate) fn init(device_transport: DeviceTransport) -> Result<(), VirtioDeviceError> {
103        let index = NR_BLOCK_DEVICE.fetch_add(1, Ordering::Relaxed);
104        let id = DeviceId::new(
105            VIRTIO_BLOCK_MAJOR_ID.get().unwrap().get(),
106            MinorId::new(index * aster_block::DEVICE_MINORS),
107        );
108        let name = Self::formatted_device_name(index);
109
110        // The inner device inherits this ID; it is the single source of truth
111        // and is used to tag I/O completion stats.
112        let device = DeviceInner::init(device_transport, id)?;
113
114        // Each bio request includes an additional 1 request and 1 response descriptor,
115        // therefore this upper bound is set to (QUEUE_SIZE - 2).
116
117        #[cfg(baseline_asterinas)]
118        {
119            let block_device = Arc::new(BlockDevice {
120                device,
121                queue: BioRequestSingleQueue::with_max_nr_segments_per_bio(
122                    (DeviceInner::QUEUE_SIZE - 2) as usize,
123                ),
124                name,
125                partition_manager: PartitionManager::new(),
126            });
127            aster_block::register(block_device).unwrap();
128        }
129
130        #[cfg(not(baseline_asterinas))]
131        {
132            let block_device_server = Self::new_with(|orpc_internal, _weak_self| BlockDevice {
133                device,
134                queue: BioRequestSingleQueue::with_max_nr_segments_per_bio(
135                    (DeviceInner::QUEUE_SIZE - 2) as usize,
136                ),
137                name,
138                partition_manager: PartitionManager::new(),
139                orpc_internal,
140            });
141
142            aster_block::register(block_device_server).unwrap();
143        }
144
145        bio_segment_pool_init();
146        Ok(())
147    }

149    /// Dequeues a `BioRequest` from the software staging queue and
150    /// processes the request.
151    pub fn handle_requests(&self) {
152        let request = self.queue.dequeue();
153        info!("Handle Request: {:?}", request);
154        match request.type_() {
155            BioType::Read => self.device.read(request),
156            BioType::Write => self.device.write(request),
157            BioType::Flush => self.device.flush(request),
158        }
159    }

173    fn enqueue(&self, bio: SubmittedBio) -> Result<(), BioEnqueueError> {
174        let reply_handle = self.bio_completion_oqueue().attach_ref_producer()?;
175
176        let mut bio = bio;
177        // Atomically bump the counters, capturing the pre-increment values so the
178        // outstanding counts recorded for this bio are consistent with the bump.
179        let outstanding_pages = self
180            .device
181            .num_outstanding_pages
182            .fetch_add(bio.num_pages(), Ordering::Relaxed);
183        let outstanding_requests = self
184            .device
185            .num_outstanding_requests
186            .fetch_add(1, Ordering::Relaxed);
187        bio.prepare_enqueue(reply_handle, outstanding_pages, outstanding_requests);
188        self.queue.enqueue(bio)
189    }

244    const QUEUE_SIZE: u16 = 64;

246    /// Creates and inits the device with the given `id` (assigned by the outer
247    /// [`BlockDevice`] via standard device ID allocation).
248    fn init(
249        mut device_transport: DeviceTransport,
250        id: DeviceId,
251    ) -> Result<Arc<Self>, VirtioDeviceError> {
252        let config_manager = VirtioBlockConfig::new_manager(device_transport.as_ref());
253
254        let config = config_manager.read_config();
255        debug!("virio_blk_config = {:?}", config);
256
257        let features =
258            BlockFeatures::negotiated_with_device(device_transport.read_device_features());
259
260        let block_size = if features.contains(BlockFeatures::BLK_SIZE) {
261            config_manager.block_size()
262        } else {
263            // Fall back to standard sector size SECTOR_SIZE (512 bytes)
264            // Reference:
265            // <https://docs.oasis-open.org/virtio/virtio/v1.2/virtio-v1.2.html#x1-2790004>
266            VirtioBlockConfig::sector_size()
267        };
268        if block_size != VirtioBlockConfig::sector_size() {
269            ostd::error!("block size {} is not supported yet", block_size);
270            return UnsupportedConfigSnafu.fail();
271        }
272
273        let num_queues = device_transport.num_queues();
274        if num_queues != 1 {
275            // TODO: Support Multi-Queue Block IO Queueing Mechanism
276            // (`BlkFeatures::MQ`) to accelerate multi-processor requests for
277            // block devices. When SMP is enabled on x86, the feature is on.
278            ostd::warn!(
279                "Multi-Queue Block IO Queueing Mechanism is not supported yet; using the first queue"
280            );
281        }
282
283        let nr_sectors = config_manager.capacity_sectors();
284
285        let queue = VirtQueue::new(0, Self::QUEUE_SIZE, device_transport.as_mut())?;
286
287        let block_requests = Arc::new(DmaStream::alloc(1, false).context(ResourceAllocSnafu)?);
288        let block_responses = Arc::new(DmaStream::alloc(1, false).context(ResourceAllocSnafu)?);
289        const {
290            assert!(Self::QUEUE_SIZE as usize * REQ_SIZE <= PAGE_SIZE);
291            assert!(Self::QUEUE_SIZE as usize * RESP_SIZE <= PAGE_SIZE);
292        }
293
294        let device = Arc::new(Self {
295            features,
296            queue: SpinLock::new(queue),
297            transport: SpinLock::new(device_transport),
298            block_requests,
299            block_responses,
300            id_allocator: SyncIdAlloc::with_capacity(Self::QUEUE_SIZE as usize),
301            submitted_requests: SpinLock::new(BTreeMap::new()),
302            device_id: id,
303            num_outstanding_pages: AtomicU32::new(0),
304            num_outstanding_requests: AtomicU32::new(0),
305            nr_sectors,
306        });
307
308        let cloned_device = device.clone();
309        let handle_irq = move |_: &TrapFrame| {
310            cloned_device.handle_irq();
311        };
312
313        let cloned_device = device.clone();
314        let handle_config_change = move |_: &TrapFrame| {
315            cloned_device.handle_config_change();
316        };
317
318        {
319            let mut transport = device.transport.lock();
320            transport.register_cfg_callback(Box::new(handle_config_change))?;
321            transport.register_queue_callback(0, Box::new(handle_irq), false)?;
322            transport.finish_init();
323        }
324
325        Ok(device)
326    }

328    /// Handles the IRQ issued from the device.
329    fn handle_irq(&self) {
330        info!("block device handle IRQ");
331        // When we enter the IRQs handling function,
332        // IRQs have already been disabled,
333        // so there is no need to call `disable_irq`.
334        loop {
335            // Pops the complete request
336            let complete_request = {
337                let mut queue = self.queue.lock();
338                let Ok((token, _)) = queue.pop_used_with_min_bytes(RESP_SIZE) else {
339                    return;
340                };
341                self.submitted_requests.lock().remove(&token).unwrap()
342            };
343
344            // Handles the response
345            let id = complete_request.id as usize;
346            let resp_slice =
347                Slice::new(&self.block_responses, id * RESP_SIZE..(id + 1) * RESP_SIZE);
348            resp_slice.sync_from_device().unwrap();
349            let resp: BlockResp = resp_slice.read_val(0).unwrap();
350            self.id_allocator.dealloc(id);
351            match RespStatus::try_from(resp.status) {
352                Ok(RespStatus::Ok) => {}
353                _ => {
354                    // Completes the bio request with an error
355                    complete_request.bio_request.into_bios().for_each(|bio| {
356                        bio.complete(BioStatus::IoError);
357                    });
358                    continue;
359                }
360            };
361
362            // Synchronize DMA mapping if read from the device
363            if let BioType::Read = complete_request.bio_request.type_() {
364                complete_request
365                    .bio_request
366                    .bios()
367                    .flat_map(|bio| bio.segments().iter().map(|segment| segment.dma_slice()))
368                    .for_each(|dma_slice| {
369                        dma_slice
370                            .mem_obj()
371                            .sync_from_device(dma_slice.offset().clone())
372                            .unwrap()
373                    });
374            }
375
376            // Completes the bio request
377            // let req_type = complete_request.bio_request.type_();
378            complete_request.bio_request.into_bios().for_each(|bio| {
379                #[cfg(not(baseline_asterinas))]
380                {
381                    let pages = bio
382                        .get_num_pages()
383                        .expect("num_pages was not set during prepare_enqueue");
384                    self.num_outstanding_pages
385                        .fetch_sub(pages, Ordering::Relaxed);
386                    self.num_outstanding_requests
387                        .fetch_sub(1, Ordering::Relaxed);
388                    bio.report_statistics();
389                }
390                bio.complete(BioStatus::Complete);
391            });
392        }
393    }

399    /// Reads data from the device, this function is non-blocking.
400    fn read(&self, bio_request: BioRequest) {
401        let id = self.id_allocator.alloc();
402        let req_slice = {
403            let req_slice = Slice::new(
404                self.block_requests.clone(),
405                id * REQ_SIZE..(id + 1) * REQ_SIZE,
406            );
407            let req = BlockReq {
408                type_: ReqType::In as _,
409                reserved: 0,
410                sector: bio_request.sid_range().start.to_raw(),
411            };
412            req_slice.write_val(0, &req).unwrap();
413            req_slice.sync_to_device().unwrap();
414            req_slice
415        };
416
417        let resp_slice = {
418            let resp_slice = Slice::new(
419                self.block_responses.clone(),
420                id * RESP_SIZE..(id + 1) * RESP_SIZE,
421            );
422            resp_slice.write_val(0, &BlockResp::default()).unwrap();
423            resp_slice.sync_to_device().unwrap();
424            resp_slice
425        };
426
427        let outputs = {
428            let mut outputs: Vec<&dyn DmaBuf> = Vec::with_capacity(bio_request.num_segments() + 1);
429            let dma_slices_iter = bio_request.bios().flat_map(|bio| {
430                bio.segments()
431                    .iter()
432                    .map(|segment| segment.dma_slice() as &dyn DmaBuf)
433            });
434            outputs.extend(dma_slices_iter);
435            outputs.push(&resp_slice);
436            outputs
437        };
438
439        let num_used_descs = outputs.len() + 1;
440        // FIXME: Split the request if it is too big
441        if num_used_descs > Self::QUEUE_SIZE as usize {
442            panic!("The request size surpasses the queue size");
443        }
444
445        loop {
446            let mut queue = self.queue.disable_irq().lock();
447            if num_used_descs > queue.available_desc() {
448                continue;
449            }
450            let token = queue
451                .add_dma_bufs(&[&req_slice], outputs.as_slice())
452                .expect("add queue failed");
453            if queue.should_notify() {
454                queue.notify();
455            }
456
457            // Records the submitted request
458            let submitted_request = SubmittedRequest::new(id as u16, bio_request);
459            self.submitted_requests
460                .disable_irq()
461                .lock()
462                .insert(token, submitted_request);
463            return;
464        }
465    }

467    /// Writes data to the device, this function is non-blocking.
468    fn write(&self, bio_request: BioRequest) {
469        let id = self.id_allocator.alloc();
470        let req_slice = {
471            let req_slice = Slice::new(
472                self.block_requests.clone(),
473                id * REQ_SIZE..(id + 1) * REQ_SIZE,
474            );
475            let req = BlockReq {
476                type_: ReqType::Out as _,
477                reserved: 0,
478                sector: bio_request.sid_range().start.to_raw(),
479            };
480            req_slice.write_val(0, &req).unwrap();
481            req_slice.sync_to_device().unwrap();
482            req_slice
483        };
484
485        let resp_slice = {
486            let resp_slice = Slice::new(
487                self.block_responses.clone(),
488                id * RESP_SIZE..(id + 1) * RESP_SIZE,
489            );
490            resp_slice.write_val(0, &BlockResp::default()).unwrap();
491            resp_slice.sync_to_device().unwrap();
492            resp_slice
493        };
494
495        let inputs = {
496            let mut inputs: Vec<&dyn DmaBuf> = Vec::with_capacity(bio_request.num_segments() + 1);
497            inputs.push(&req_slice);
498            for dma_slice in bio_request
499                .bios()
500                .flat_map(|bio| bio.segments().iter().map(|segment| segment.dma_slice()))
501            {
502                dma_slice
503                    .mem_obj()
504                    .sync_to_device(dma_slice.offset().clone())
505                    .unwrap();
506                inputs.push(dma_slice as &dyn DmaBuf);
507            }
508            inputs
509        };
510
511        let num_used_descs = inputs.len() + 1;
512        // FIXME: Split the request if it is too big
513        if num_used_descs > Self::QUEUE_SIZE as usize {
514            panic!("The request size surpasses the queue size");
515        }
516        loop {
517            let mut queue = self.queue.disable_irq().lock();
518            if num_used_descs > queue.available_desc() {
519                continue;
520            }
521            let token = queue
522                .add_dma_bufs(inputs.as_slice(), &[&resp_slice])
523                .expect("add queue failed");
524            if queue.should_notify() {
525                queue.notify();
526            }
527
528            // Records the submitted request
529            let submitted_request = SubmittedRequest::new(id as u16, bio_request);
530            self.submitted_requests
531                .disable_irq()
532                .lock()
533                .insert(token, submitted_request);
534            return;
535        }
536    }

538    /// Flushes any cached data from the guest to the persistent storage on the host.
539    /// This will be ignored if the device doesn't support the `VIRTIO_BLK_F_FLUSH` feature.
540    fn flush(&self, bio_request: BioRequest) {
541        if !self.features.contains(BlockFeatures::FLUSH) {
542            bio_request.into_bios().for_each(|bio| {
543                bio.complete(BioStatus::Complete);
544            });
545            return;
546        }
547
548        let id = self.id_allocator.alloc();
549        let req_slice = {
550            let req_slice = Slice::new(&self.block_requests, id * REQ_SIZE..(id + 1) * REQ_SIZE);
551            let req = BlockReq {
552                type_: ReqType::Flush as _,
553                reserved: 0,
554                sector: bio_request.sid_range().start.to_raw(),
555            };
556            req_slice.write_val(0, &req).unwrap();
557            req_slice.sync_to_device().unwrap();
558            req_slice
559        };
560
561        let resp_slice = {
562            let resp_slice =
563                Slice::new(&self.block_responses, id * RESP_SIZE..(id + 1) * RESP_SIZE);
564            resp_slice.write_val(0, &BlockResp::default()).unwrap();
565            resp_slice.sync_to_device().unwrap();
566            resp_slice
567        };
568
569        // One descriptor for the input `req_slice`, one for the output `resp_slice`.
570        let num_used_descs = 2;
571        loop {
572            let mut queue = self.queue.disable_irq().lock();
573            if num_used_descs > queue.available_desc() {
574                continue;
575            }
576            let token = queue
577                .add_dma_bufs(&[&req_slice], &[&resp_slice])
578                .expect("add queue failed");
579            if queue.should_notify() {
580                queue.notify();
581            }
582
583            // Records the submitted request
584            let submitted_request = SubmittedRequest::new(id as u16, bio_request);
585            self.submitted_requests
586                .disable_irq()
587                .lock()
588                .insert(token, submitted_request);
589            return;
590        }
591    }

607/// VirtIOBlock request.

612    reserved: u32,

615
616const REQ_SIZE: usize = size_of::<BlockReq>();
617
618/// Response of a VirtIOBlock request.
619#[repr(C)]
620#[derive(Clone, Copy, Debug, Pod)]
621struct BlockResp {
622    pub status: u8,
623}
624
625const RESP_SIZE: usize = size_of::<BlockResp>();
626
627impl Default for BlockResp {
628    fn default() -> Self {
629        Self {
630            status: RespStatus::_NotReady as _,
631        }
