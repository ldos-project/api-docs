48#[ostd_error]
49#[derive(Debug, Snafu)]
50#[snafu(visibility(pub))]
51pub enum VirtioDeviceError {
52    #[snafu(transparent)]
53    Transport {
54        source: VirtioTransportError,
55    },
56    #[ostd(context(source))]
57    ResourceAlloc {
58        source: ostd::Error,
59    },
60    InvalidQueueArgs,
61    UnsupportedConfig,
62    /// The ORPC Errors
63    #[cfg(not(baseline_asterinas))]
64    #[snafu(transparent)]
65    #[ostd(context(source))]
66    RPCError {
67        /// Source ORPC error
68        source: RPCError,
69    },
70    /// The OQueue errors
71    #[cfg(not(baseline_asterinas))]
72    #[snafu(transparent)]
73    #[ostd(context(source))]
74    OQueueError {
75        /// Source OQueue error
76        source: OQueueError,
77    },
78}
79
80impl From<queue::CreationError> for VirtioDeviceError {
81    fn from(value: queue::CreationError) -> Self {
82        match value {
83            queue::CreationError::InvalidArgs => InvalidQueueArgsSnafu.build(),
84            queue::CreationError::ResourceAlloc(e) => {
85                Err::<(), _>(e).context(ResourceAllocSnafu).unwrap_err()
86            }
87            queue::CreationError::Transport(e) => e.into(),
88        }
89    }
90}
